```html
<!DOCTYPE html>
<html lang="en">
<head>
    <meta charset="UTF-8">
    <meta name="viewport" content="width=device-width, initial-scale=1.0">
    <title>Vivid Night Sky Animation</title>
    <style>
        body {
            background-color: #0a0a2a; /* Very dark indigo/blue for night sky */
            overflow: hidden;
            margin: 0;
            padding: 0;
            height: 100vh;
            cursor: none; /* Hides default cursor for immersion */
        }
        canvas {
            display: block;
        }
    </style>
</head>
<body>
    <canvas id="nightSkyCanvas"></canvas>

    <script>
        const canvas = document.getElementById('nightSkyCanvas');
        const ctx = canvas.getContext('2d');

        let width = window.innerWidth;
        let height = window.innerHeight;

        canvas.width = width;
        canvas.height = height;

        window.addEventListener('resize', () => {
            width = window.innerWidth;
            height = window.innerHeight;
            canvas.width = width;
            canvas.height = height;
        });

        // --- Star System Management ---
        const numStars = 300;
        let stars = [];

        class Star {
            constructor() {
                this.x = Math.random() * width;
                this.y = Math.random() * height;
                this.radius = Math.random() * 1.5 + 0.5;
                // Opacity changes based on a 'twinkle' cycle
                this.baseOpacity = Math.random() * 0.7 + 0.3;
                this.twinkleSpeed = Math.random() * 0.08 + 0.02;
                this.currentAlpha = 1;
            }

            update() {
                // Simple twinkling effect using sine wave for smooth brightness changes
                this.currentAlpha = this.baseOpacity + Math.abs(Math.sin(Date.now() / (1000 / this.twinkleSpeed) + this.x / 100)) * 0.4;
                this.currentAlpha = Math.max(0.1, Math.min(1, this.currentAlpha));
            }

            draw() {
                ctx.beginPath();
                ctx.arc(this.x, this.y, this.radius, 0, Math.PI * 2);
                ctx.fillStyle = `rgba(255, 255, 200, ${this.currentAlpha})`; // Bright yellowish-white
                ctx.fill();
            }
        }

        function initializeStars() {
            for (let i = 0; i < numStars; i++) {
                stars.push(new Star());
            }
        }

        // --- Shooting Star (Meteor) Management ---
        const meteorCount = 5;
        let meteors = [];

        class Meteor {
            constructor() {
                this.reset();
            }

            reset() {
                // Start position: random point near the bottom/edges
                this.x = Math.random() * width;
                this.y = height + Math.random() * 50; // Start just below screen
                
                // Target velocity/angle calculation
                let angle = Math.random() * 0.4 + Math.PI * 1.2 + Math.random() * 0.2; // Aim upwards/diagonally
                let speed = Math.random() * 8 + 5;
                
                this.vx = Math.cos(angle) * speed * (0.8 + Math.random() * 0.4);
                this.vy = Math.sin(angle) * speed * (0.8 + Math.random() * 0.4);
                
                this.size = Math.random() * 2 + 1;
                this.opacity = 1;
                this.trailLength = 10;
                this.trail = [];
            }

            update() {
                // Keep track of the previous positions for the trail effect
                this.trail.unshift({ x: this.x, y: this.y, opacity: 1 });
                if (this.trail.length > this.trailLength) {
                    this.trail.pop();
                }
                
                this.x += this.vx;
                this.y += this.vy;
                this.opacity -= 0.02; // Fade out over time
            }

            draw() {
                // 1. Draw the trail (fading dots)
                for (let i = 0; i < this.trail.length; i++) {
                    const point = this.trail[i];
                    const alpha = (this.trailLength - i) / this.trailLength * 0.7; // Fade based on age
                    
                    ctx.fillStyle = `rgba(150, 255, 255, ${alpha * 0.8})`;
                    ctx.beginPath();
                    ctx.arc(point.x, point.y, (this.size / (this.trailLength - i) * 1.5), 0, Math.PI * 2);
                    ctx.fill();
                }

                // 2. Draw the head (brightest part)
                ctx.fillStyle = `rgba(255, 255, 255, ${this.opacity})`;
                ctx.beginPath();
                ctx.arc(this.x, this.y, this.size * 0.8, 0, Math.PI * 2);
                ctx.fill();
            }
        }

        function initializeMeteors() {
            for (let i = 0; i < meteorCount; i++) {
                meteors.push(new Meteor());
            }
        }
        
        // --- Animation Loop ---
        function animate() {
            // Clear the canvas slightly less aggressively to leave a fading 'afterglow'
            ctx.fillStyle = 'rgba(10, 10, 42, 0.15)'; // Dark blue with slight opacity for ghosting trail effect
            ctx.fillRect(0, 0, width, height);

            // 1. Draw Stars (Twinkling)
            stars.forEach(star => {
                star.update();
                star.draw();
            });

            // 2. Draw Meteors
            meteors.forEach(meteor => {
                meteor.update();
                meteor.draw();
            });

            // 3. Randomly spawn new meteors occasionally
            if (Math.random() < 0.03) {
                meteors[Math.floor(Math.random() * meteorCount)].reset();
            }

            requestAnimationFrame(animate);
        }

        // --- Initialization ---
        initializeStars();
        initializeMeteors();
        
        // Start the show!
        animate();

    </script>
</body>
</html>
```
