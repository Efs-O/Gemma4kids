```html
<!DOCTYPE html>
<html lang="en">
<head>
    <meta charset="UTF-8">
    <meta name="viewport" content="width=device-width, initial-scale=1.0">
    <title>Post-Rain Rainbow</title>
    <style>
        body {
            margin: 0;
            overflow: hidden;
            background-color: #0a0a20; /* Deep night blue */
            perspective: 1000px;
            height: 100vh;
            display: flex;
            justify-content: center;
            align-items: center;
            font-family: 'Arial', sans-serif;
        }

        /* --- Rain Simulation --- */
        .rain-container {
            position: absolute;
            top: 0;
            left: 0;
            width: 100%;
            height: 100%;
            pointer-events: none;
            z-index: 1;
        }

        .rain-drop {
            position: absolute;
            width: 1px;
            height: 30px;
            background: linear-gradient(to bottom, rgba(255, 255, 255, 0.6), rgba(173, 216, 230, 0.8)); /* Light blue/white streaks */
            animation: fall linear infinite;
            opacity: 0;
        }

        @keyframes fall {
            0% {
                transform: translateY(-100vh);
                opacity: 0.5;
            }
            100% {
                transform: translateY(100vh);
                opacity: 0.5;
            }
        }

        /* --- Rainbow Effect --- */
        .rainbow-layer {
            position: absolute;
            top: 0;
            left: 0;
            width: 100%;
            height: 100%;
            z-index: 10; /* Above the rain */
            opacity: 0;
            transition: opacity 2s ease-in-out; /* Smooth fade in */
            background: linear-gradient(
                135deg, 
                #ff0000, /* Red */
                #ff7f50, /* Orange */
                #ffff00, /* Yellow */
                #00ff00, /* Green */
                #0000ff, /* Blue */
                #4b0082 /* Indigo */
            );
            background-size: 400% 400%; /* For animation */
            animation: rainbowShift 10s ease-in-out forwards;
        }

        @keyframes rainbowShift {
            0% {
                background-position: 0% 50%;
            }
            50% {
                background-position: 100% 50%;
            }
            100% {
                background-position: 0% 50%;
            }
        }

        /* --- Text Overlay (for context) --- */
        .message {
            position: relative;
            z-index: 20;
            color: #ffffff;
            text-align: center;
            padding: 30px;
            background: rgba(0, 0, 0, 0.5);
            border-radius: 15px;
        }

    </style>
</head>
<body>

    <!-- Rain Layer -->
    <div class="rain-container" id="rainLayer">
        <!-- Generate 100 rain drops -->
        <script>
            const rainLayer = document.getElementById('rainLayer');
            for (let i = 0; i < 100; i++) {
                const drop = document.createElement('div');
                drop.className = 'rain-drop';
                // Stagger starting positions and animation delays
                drop.style.left = `${Math.random() * 100}%`;
                drop.style.animationDelay = `${Math.random() * 5}s`;
                drop.style.animationDuration = `${5 + Math.random() * 10}s`;
                drop.style.top = `${Math.random() * -200}px`; // Start above screen
                rainLayer.appendChild(drop);
            }
        </script>
    </div>

    <!-- Rainbow Layer (Appears after rain effect begins) -->
    <div class="rainbow-layer" id="rainbowLayer"></div>

    <!-- Message -->
    <div class="message">
        <h1>The Aftermath</h1>
        <p>Rain fades, the vibrant colors emerge.</p>
    </div>

    <script>
        // Trigger the rainbow to fade in slightly later, simulating the rain clearing and the light appearing.
        document.addEventListener('DOMContentLoaded', () => {
            const rainbow = document.getElementById('rainbowLayer');
            // Start the rainbow fade in after 1 second, ensuring the initial rain is visible first.
            setTimeout(() => {
                rainbow.style.opacity = 1;
            }, 1000);
        });
    </script>

</body>
</html>
```
