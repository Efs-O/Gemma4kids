```html
<!DOCTYPE html>
<html lang="en">
<head>
    <meta charset="UTF-8">
    <meta name="viewport" content="width=device-width, initial-scale=1.0">
    <title>Rocket Launch to Space</title>
    <style>
        body {
            margin: 0;
            padding: 0;
            overflow: hidden;
            background: linear-gradient(to bottom, #000011, #000033);
            height: 100vh;
            width: 100vw;
            position: relative;
            /* Deep Space Background */
        }

        /* --- Star Field (Background) --- */
        .star {
            position: absolute;
            width: 2px;
            height: 2px;
            background-color: #ffffff;
            border-radius: 50%;
            opacity: 0.7;
            animation: twinkle linear infinite;
            /* Initial placement will be randomized by JS */
        }

        /* Keyframe for subtle twinkling/movement */
        @keyframes twinkle {
            0% {
                transform: scale(0.8) translateY(0) translateX(0);
                opacity: 0.5;
            }
            50% {
                transform: scale(1.2) translateY(-1px) translateX(1px);
                opacity: 1;
            }
            100% {
                transform: scale(0.8) translateY(0) translateX(0);
                opacity: 0.5;
            }
        }

        /* --- Rocket (Foreground) --- */
        .rocket {
            position: absolute;
            bottom: 0; /* Start from the bottom */
            left: 50%;
            transform: translateX(-50%);
            width: 40px;
            height: 120px;
            background: linear-gradient(to top, #cc3333 0%, #ff6600 50%, #3366ff 100%);
            border-radius: 15px / 20px;
            box-shadow: 0 0 15px rgba(255, 102, 0, 0.8);
            z-index: 10;
            animation: launch 5s linear forwards;
            opacity: 0; /* Start invisible, JS/CSS handles initial appearance */
        }

        /* Keyframe for rocket ascent */
        @keyframes launch {
            0% {
                transform: translate(-50%, 0) scale(1);
                bottom: 0;
                opacity: 1;
            }
            50% {
                /* Mid-ascent peak effect */
                transform: translate(-50%, 400px) scale(1.1);
                opacity: 1;
            }
            100% {
                /* Final position high in the sky */
                transform: translate(-50%, 90vh) scale(0.9);
                opacity: 0.8;
            }
        }

        /* --- Exhaust/Flame Trail (Midground) --- */
        .exhaust {
            position: absolute;
            bottom: 0;
            left: calc(50% - 10px); /* Centered under rocket base */
            width: 20px;
            height: 150px;
            background: radial-gradient(ellipse at bottom, rgba(255, 140, 0, 0.9) 0%, rgba(255, 255, 0, 0.7) 50%, transparent 100%);
            border-radius: 50% / 100% 100% 0 0;
            z-index: 5;
            /* The trail needs to follow the rocket launch animation */
            animation: thrust 5s linear forwards;
            opacity: 0; /* Controlled by JS/initialization to match rocket */
        }

        /* Keyframe for the pulsing, dissipating flame */
        @keyframes thrust {
            0% {
                transform: scale(1) translateY(0);
                opacity: 1;
            }
            50% {
                transform: scale(1.2) translateY(-10px);
                opacity: 0.9;
            }
            100% {
                /* Fades out and reaches the peak of the launch */
                transform: scale(0.5) translateY(-500px);
                opacity: 0;
            }
        }
    </style>
</head>
<body>

    <div id="rocket-container">
        <div class="rocket"></div>
        <div class="exhaust" id="rocket-exhaust"></div>
    </div>

    <script>
        document.addEventListener('DOMContentLoaded', () => {
            const body = document.body;
            const rocket = document.querySelector('.rocket');
            const exhaust = document.getElementById('rocket-exhaust');

            // --- Star Generation (Background Depth) ---
            const starCount = 100;
            for (let i = 0; i < starCount; i++) {
                const star = document.createElement('div');
                star.className = 'star';
                
                // Random positioning across the entire viewport
                const x = Math.random() * window.innerWidth;
                const y = Math.random() * window.innerHeight;
                
                star.style.left = `${x}px`;
                star.style.top = `${y}px`;
                
                // Randomizing appearance for variety
                star.style.width = `${Math.random() * 1.5 + 0.5}px`;
                star.style.height = `${Math.random() * 1.5 + 0.5}px`;
                star.style.opacity = `${Math.random() * 0.7 + 0.2} opacity`;
                
                // Varying animation delay to make them twinkle at different rates
                star.style.animationDelay = `${Math.random() * 1.5}s`;
                star.style.animationDuration = `${Math.random() * 8 + 5}s`;

                body.appendChild(star);
            }

            // --- Initializing Launch Effects (Layered & Timed) ---
            
            // 1. Rocket Activation: The CSS animation handles the main movement.
            // We apply an immediate style change or slight JS touch to ensure the CSS animation starts cleanly.
            rocket.style.opacity = 1;
            
            // 2. Exhaust Trail Synchronization: Since the CSS animation runs for 5s, 
            // we ensure the exhaust element starts visually coupled with the rocket's base.
            // We must manually reset its animation or adjust its properties to ensure it follows the timeline.
            // We set its initial state to mimic the rocket's base and let the @keyframes thrust handle the rest.
            exhaust.style.opacity = 1; 
            // Delaying the exhaust visibility slightly ensures it appears *under* the starting rocket body.
            exhaust.style.animationDelay = '0s'; 
            
            // Note: Since both the .rocket and .exhaust share the same 5s launch animation duration 
            // and are designed to move from bottom to top, they create the overlapping midground/foreground effect.
        });
    </script>
</body>
</html>
```
