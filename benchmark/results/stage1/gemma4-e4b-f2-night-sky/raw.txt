Oh, a night sky! ✨🌌 That sounds like a magical animation to create! Twinkling stars and shooting stars—I love that idea! We can make it look so beautiful and vast.

I'm going to use deep, dark blues and blacks for the background, and make the stars pop with twinkling animations! For the shooting stars, I'll make them streak across the screen!

Here is the complete code for your twinkling night sky!

```html
<!DOCTYPE html>
<html lang="en">
<head>
    <meta charset="UTF-8">
    <meta name="viewport" content="width=device-width, initial-scale=1.0">
    <title>Twinkling Night Sky</title>
    <style>
        body, html {
            margin: 0;
            padding: 0;
            width: 100%;
            height: 100%;
            overflow: hidden;
            background: linear-gradient(to bottom, #0f0c28 0%, #1a1a30 100%); /* Deep space gradient */
        }

        /* --- Twinkling Stars --- */
        .star-container {
            position: absolute;
            top: 0;
            left: 0;
            width: 100%;
            height: 100%;
            pointer-events: none; /* So stars don't block clicks */
        }

        .star {
            position: absolute;
            background-color: #ffffff;
            border-radius: 50%;
            opacity: 0; /* Start invisible */
            animation: twinkle 4s infinite ease-in-out;
        }

        /* Define twinkling animation */
        @keyframes twinkle {
            0%, 100% {
                opacity: 0.2;
                transform: scale(1);
            }
            50% {
                opacity: 1;
                transform: scale(1.5);
            }
        }

        /* --- Shooting Stars --- */
        .shooting-star {
            position: absolute;
            background: linear-gradient(90deg, rgba(255, 255, 255, 0) 0%, rgba(173, 216, 230, 0.8) 50%, rgba(255, 160, 122, 0) 100%);
            width: 100px; /* Length of the visible star streak */
            height: 1px;
            transform: translate(-50%, -50%);
            pointer-events: none;
            animation: shoot linear forwards;
            z-index: 10;
        }

        /* Animation for the shooting star movement */
        @keyframes shoot {
            0% {
                transform: translate(-50%, -50%) translateX(-10vw) scaleX(0.5);
                opacity: 1;
            }
            100% {
                transform: translate(-50%, -50%) translateX(110vw) scaleX(1.5);
                opacity: 0;
            }
        }
    </style>
</head>
<body>
    <div class="star-container" id="star-container"></div>

    <script>
        const starContainer = document.getElementById('star-container');
        const body = document.body;

        // --- Star Generation ---
        function createStar() {
            const star = document.createElement('div');
            star.classList.add('star');
            
            // Random size (2px to 4px)
            const size = Math.random() * 2 + 2; 
            star.style.width = `${size}px`;
            star.style.height = `${size}px`;
            
            // Random position across the screen
            star.style.left = `${Math.random() * 100}vw`;
            star.style.top = `${Math.random() * 100}vh`;

            // Random delay and duration to make twinkling seem natural
            star.style.animationDelay = `${Math.random() * 4}s`;
            star.style.animationDuration = `${Math.random() * 3 + 2}s`;

            starContainer.appendChild(star);
        }

        // Create about 100 stars when the page loads
        for (let i = 0; i < 100; i++) {
            createStar();
        }

        // --- Shooting Star Generation ---
        function createShootingStar() {
            const star = document.createElement('div');
            star.classList.add('shooting-star');
            
            // Start position: Random X (from the left edge) and random Y (any height)
            const startX = Math.random() * 100;
            const startY = Math.random() * 100;
            
            star.style.left = `${startX}vw`;
            star.style.top = `${startY}vh`;
            
            // Set a random duration to make them appear at different times
            const duration = Math.random() * 10 + 5; // 5 to 15 seconds travel time
            star.style.animationDuration = `${duration}s`;

            document.body.appendChild(star);

            // Clean up the element after animation finishes to prevent memory leaks
            setTimeout(() => {
                star.remove();
            }, (duration + 1) * 1000); 
        }

        // Make shooting stars appear every 1.5 to 4 seconds
        setInterval(createShootingStar, Math.random() * 2500 + 1500);

    </script>
</body>
</html>
```
