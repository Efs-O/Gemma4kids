```html
<!DOCTYPE html>
<html lang="en">
<head>
    <meta charset="UTF-8">
    <meta name="viewport" content="width=device-width, initial-scale=1.0">
    <title>Vivid Bouncing Ball Animation</title>
    <style>
        /* Global Reset and Fullscreen Setup */
        body {
            margin: 0;
            overflow: hidden; /* Important to prevent scrollbars */
            background-color: #1a0033; /* Deep space background */
            display: flex;
            justify-content: center;
            align-items: center;
            height: 100vh;
            width: 100vw;
        }

        /* Canvas Styling */
        #bouncingCanvas {
            display: block;
            /* The JS will handle sizing, but setting it up ensures it fills the body */
        }
    </style>
</head>
<body>

    <canvas id="bouncingCanvas"></canvas>

    <script>
        // --- Setup Variables ---
        const canvas = document.getElementById('bouncingCanvas');
        const ctx = canvas.getContext('2d');

        // Set canvas dimensions to fill the entire viewport
        canvas.width = window.innerWidth;
        canvas.height = window.innerHeight;

        // Ball Properties
        let ball = {
            x: canvas.width * 0.1, // Start 10% from left
            y: canvas.height * 0.1, // Start 10% from top
            radius: 30,
            color: '#FF00FF', // Vivid Magenta
            
            // Initial Velocities (Speed)
            dx: 8, // X velocity
            dy: 7  // Y velocity
        };

        // Gravity/Damping factor for bounciness (controls energy loss)
        const DAMPING_FACTOR = 0.99;
        
        // --- Animation Core Functions ---

        /**
         * Draws the ball on the canvas context.
         */
        function drawBall() {
            ctx.beginPath();
            ctx.arc(ball.x, ball.y, ball.radius, 0, Math.PI * 2);
            ctx.fillStyle = ball.color;
            ctx.fill();
            ctx.strokeStyle = '#FFFFFF';
            ctx.lineWidth = 3;
            ctx.stroke();
            ctx.closePath();
        }

        /**
         * Updates the ball's position and handles boundary collisions.
         */
        function updateBall() {
            // 1. Update position
            ball.x += ball.dx;
            ball.y += ball.dy;

            // 2. Collision Detection (Horizontal Walls)
            if (ball.x + ball.radius > canvas.width || ball.x - ball.radius < 0) {
                ball.dx = -ball.dx; // Reverse direction
                // Keep ball within bounds to prevent sticking
                ball.x = Math.max(ball.radius, Math.min(canvas.width - ball.radius, ball.x));
            }

            // 3. Collision Detection (Vertical Walls)
            if (ball.y + ball.radius > canvas.height || ball.y - ball.radius < 0) {
                ball.dy = -ball.dy; // Reverse direction
                // Keep ball within bounds
                ball.y = Math.max(ball.radius, Math.min(canvas.height - ball.radius, ball.y));
            }

            // 4. Apply Damping (Makes the bounce lose energy)
            ball.dx *= DAMPING_FACTOR;
            ball.dy *= DAMPING_FACTOR;
        }

        /**
         * The main animation loop controlled by requestAnimationFrame.
         */
        function animate() {
            // 1. Clear the entire canvas for the next frame
            // This creates the illusion of movement
            ctx.clearRect(0, 0, canvas.width, canvas.height);

            // Optional: Draw a faint background element that moves slowly
            drawBackgroundGradient();

            // 2. Update the physics and position
            updateBall();

            // 3. Draw the ball at its new position
            drawBall();

            // Request the browser to call animate() again before the next repaint
            requestAnimationFrame(animate);
        }

        /**
         * Adds a subtle, colorful gradient background effect that shifts slowly.
         */
        let startTime = Date.now();
        function drawBackgroundGradient() {
            const elapsed = (Date.now() - startTime) / 1000;
            
            // Create a linear gradient that shifts based on time
            const gradient = ctx.createLinearGradient(0, 0, canvas.width, canvas.height);
            
            // Vivid color mapping based on time offset
            const colorStart = `hsl(${180 + Math.sin(elapsed * 0.1) * 50}, 80%, 40%)`; // Cyan/Blue shift
            const colorEnd = `hsl(${280 + Math.cos(elapsed * 0.08) * 50}, 90%, 30%)`;   // Magenta/Purple shift
            
            gradient.addColorStop(0, colorStart);
            gradient.addColorStop(1, colorEnd);

            ctx.fillStyle = gradient;
            ctx.fillRect(0, 0, canvas.width, canvas.height);
        }


        // --- Initialization & Event Listeners ---

        // Handle window resizing to keep the animation covering the screen
        window.addEventListener('resize', () => {
            canvas.width = window.innerWidth;
            canvas.height = window.innerHeight;
            // Reset ball position slightly to account for new size if needed, 
            // but for simplicity, we just let it continue bouncing from its current relative position.
        });

        // Start the animation loop when the script loads
        animate();

    </script>
</body>
</html>
```
