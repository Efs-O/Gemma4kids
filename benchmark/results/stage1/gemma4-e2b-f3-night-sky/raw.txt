```html
<!DOCTYPE html>
<html lang="en">
<head>
    <meta charset="UTF-8">
    <meta name="viewport" content="width=device-width, initial-scale=1.0">
    <title>Vivid Starfield</title>
    <style>
        body {
            margin: 0;
            overflow: hidden;
            background: radial-gradient(circle at center, #0a001a 0%, #1a0033 100%); /* Deep space gradient */
        }
        canvas {
            display: block;
        }
    </style>
</head>
<body>
    <canvas id="nightSkyCanvas"></canvas>

    <script>
        const canvas = document.getElementById('nightSkyCanvas');
        const ctx = canvas.getContext('2d');

        let stars = [];
        const NUM_STARS = 150;
        const shootingStars = [];
        const MAX_SHOOTING_STARS = 5;

        // Resize canvas to fill the viewport
        function resizeCanvas() {
            canvas.width = window.innerWidth;
            canvas.height = window.innerHeight;
        }
        window.addEventListener('resize', resizeCanvas);
        resizeCanvas();

        // --- Star Setup ---
        class Star {
            constructor() {
                this.x = Math.random() * canvas.width;
                this.y = Math.random() * canvas.height;
                this.size = Math.random() * 2 + 0.5; // Size between 0.5 and 2.5
                this.twinkleSpeed = Math.random() * 0.01 + 0.005;
                this.opacity = Math.random() * 0.8 + 0.2; // Base opacity
            }
        }

        function initStars() {
            for (let i = 0; i < NUM_STARS; i++) {
                stars.push(new Star());
            }
        }

        // --- Shooting Star Setup ---
        class ShootingStar {
            constructor() {
                this.x = Math.random() * canvas.width;
                this.y = -50;
                this.size = Math.random() * 2 + 1;
                this.speed = Math.random() * 3 + 4; // Fast speed
                this.opacity = 1.0;
                this.life = 100; // Lifetime counter
            }

            update() {
                this.y += this.speed;
                this.life -= 1;
                
                // Loop the star back to the top if it falls off the screen
                if (this.y > canvas.height + 50) {
                    this.reset();
                }
            }

            draw() {
                ctx.fillStyle = `rgba(255, 255, 255, ${this.opacity})`;
                ctx.beginPath();
                ctx.arc(this.x, this.y, this.size, 0, Math.PI * 2);
                ctx.fill();
            }

            reset() {
                this.x = Math.random() * canvas.width;
                this.y = -50;
                this.speed = Math.random() * 3 + 4;
                this.life = 100;
            }
        }

        function initShootingStars() {
            for (let i = 0; i < MAX_SHOOTING_STARS; i++) {
                shootingStars.push(new ShootingStar());
            }
        }

        // --- Animation Loop ---
        function animate() {
            // Clear canvas for smooth redraw
            ctx.clearRect(0, 0, canvas.width, canvas.height);

            // 1. Draw Stars (Twinkling Effect)
            stars.forEach(star => {
                // Twinkle logic: subtly change opacity over time
                star.opacity = Math.min(1.0, star.opacity + star.twinkleSpeed * 0.05);
                
                ctx.fillStyle = `rgba(255, 255, 255, ${star.opacity})`;
                ctx.beginPath();
                ctx.arc(star.x, star.y, star.size, 0, Math.PI * 2);
                ctx.fill();
            });

            // 2. Draw Shooting Stars
            shootingStars.forEach(star => {
                star.update();
                star.draw();
            });

            requestAnimationFrame(animate);
        }

        // Initialization
        initStars();
        initShootingStars();
        animate();

    </script>
</body>
</html>
```
