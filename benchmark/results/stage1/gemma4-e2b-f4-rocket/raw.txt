    <script>
        const starBackground = document.getElementById('starBackground');
        const viewportWidth = window.innerWidth;
        const viewportHeight = window.innerHeight;

        // Function to generate random stars
        function generateStars() {
            const numStars = 200;
            for (let i = 0; i < numStars; i++) {
                const star = document.createElement('div');
                star.className = 'star';

                // Position randomly across the screen
                star.style.left = `${Math.random() * 100}%`;
                star.style.top = `${Math.random() * 100}%`;

                // Vary size slightly
                const size = Math.random() * 3 + 1;
                star.style.width = `${size}px`;
                star.style.height = `${size}px`;
                
                // Vary animation timing/delay for depth effect
                const delay = Math.random() * 10; // Slow drift variation
                const duration = 10 + Math.random() * 10; // Slow movement variation
                const animationDelay = delay * 0.5;
                const animationDuration = duration;

                star.style.animationDelay = `${animationDelay}s`;
                star.style.animationDuration = `${animationDuration}s`;

                starBackground.appendChild(star);
            }
        }

        // Initialize the star field
        generateStars();

        // Handle resizing if necessary (though CSS handles most positioning)
        window.addEventListener('resize', () => {
            // Re-generate stars if size changes drastically, ensuring coverage
            generateStars();
        });
    </script>

</body>
