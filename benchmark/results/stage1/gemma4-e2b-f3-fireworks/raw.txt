```html
<!DOCTYPE html>
<html lang="en">
<head>
    <meta charset="UTF-8">
    <meta name="viewport" content="width=device-width, initial-scale=1.0">
    <title>Vivid Fireworks Night Sky</title>
    <style>
        body {
            margin: 0;
            overflow: hidden;
            /* Vivid Night Sky Background */
            background: linear-gradient(to bottom, #0a001a 0%, #1c003d 50%, #000011 100%);
            height: 100vh;
            width: 100vw;
        }

        /* --- Firework Base Style --- */
        .firework {
            position: absolute;
            border-radius: 50%;
            pointer-events: none;
            animation: explode 2s ease-out forwards;
        }

        @keyframes explode {
            0% {
                transform: scale(0);
                opacity: 1;
            }
            100% {
                transform: scale(20);
                opacity: 0;
            }
        }

        /* --- Particle / Trail Styling (Simulating the burst) --- */
        .particle {
            position: absolute;
            width: 2px;
            height: 2px;
            border-radius: 50%;
            animation: fadeOut 1.5s forwards;
        }

        @keyframes fadeOut {
            0% {
                opacity: 1;
            }
            100% {
                opacity: 0;
            }
        }

        /* --- Staggered Firework Definitions (Simulating different explosion shapes/colors) --- */
        .firework-1 {
            /* Red/Gold Burst */
            background-color: #ff4400;
            animation-delay: 0.5s;
        }
        .firework-2 {
            /* Bright Cyan Burst */
            background-color: #00ffff;
            animation-delay: 1.0s;
        }
        .firework-3 {
            /* Vivid Magenta Burst */
            background-color: #ff00ff;
            animation-delay: 1.5s;
        }
        .firework-4 {
            /* Bright Green Burst */
            background-color: #00ff99;
            animation-delay: 2.0s;
        }
        .firework-5 {
            /* Deep Blue Burst */
            background-color: #4444ff;
            animation-delay: 2.5s;
        }

        /* --- Generating Actual Explosions via JS for dynamic placement --- */
        /* Note: Since CSS cannot easily generate arbitrary radiating shapes based on position in a clean way without complex SVG/Canvas, we generate particle divs via JS for precise radial burst effects. */

    </style>
</head>
<body>

    <div id="fireworks-container">
        <!-- Fireworks will be injected here by JavaScript -->
    </div>

    <script>
        const container = document.getElementById('fireworks-container');
        const totalFireworks = 10; // How many simultaneous bursts to aim for

        function createFirework(x, y) {
            const fireworkDiv = document.createElement('div');
            fireworkDiv.className = 'firework';
            
            // Randomly assign a base color style based on the firework type
            const colors = ['#ff4400', '#00ffff', '#ff00ff', '#00ff99', '#4444ff'];
            const randomColor = colors[Math.floor(Math.random() * colors.length)];
            fireworkDiv.style.backgroundColor = randomColor;
            
            // Position the initial explosion point near the top center
            fireworkDiv.style.left = `${x}px`;
            fireworkDiv.style.top = `${y}px`;
            fireworkDiv.style.transform = 'translate(-50%, -50%) scale(0)'; // Start small

            container.appendChild(fireworkDiv);

            // --- Create radiating particles for the burst effect ---
            const numParticles = 30;
            for (let i = 0; i < numParticles; i++) {
                const particle = document.createElement('div');
                particle.className = 'particle';
                
                // Calculate random angle and distance for the particle to fly out
                const angle = (i / numParticles) * 360;
                const distance = Math.random() * 150 + 50; // Particles travel 50px to 200px distance
                
                // Convert polar coordinates to Cartesian coordinates for positioning
                const radians = angle * (Math.PI / 180);
                const offsetX = Math.cos(radians) * distance;
                const offsetY = Math.sin(radians) * distance;

                particle.style.backgroundColor = randomColor;
                
                // Position the particle relative to the firework's center point
                particle.style.left = `${x + offsetX}px`;
                particle.style.top = `${y + offsetY}px`;
                
                // Apply a staggered animation delay to create the burst effect
                particle.style.animationDelay = `${Math.random() * 0.5}s`; 
                
                container.appendChild(particle);
            }

            // Trigger the main explosion animation after a small delay
            setTimeout(() => {
                fireworkDiv.style.animation = 'explode 1.5s ease-out forwards';
            }, 50); 
        }

        function launchFireworks() {
            // Launch fireworks randomly across the upper half of the screen
            for (let i = 0; i < totalFireworks; i++) {
                const x = Math.random() * window.innerWidth;
                // Ensure the launch point is not too low, simulating an explosion high up
                const y = Math.random() * (window.innerHeight * 0.4); 
                createFirework(x, y);
            }
        }

        // Launch initial fireworks immediately
        launchFireworks();

        // Set up an infinite loop for continuous display
        setInterval(launchFireworks, 3000); // Launch a new set every 3 seconds
    </script>
</body>
</html>
```
