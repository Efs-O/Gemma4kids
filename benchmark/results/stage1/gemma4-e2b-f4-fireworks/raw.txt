        body {
            margin: 0;
            height: 100vh;
            display: flex;
            justify-content: center;
            align-items: center;
            background: radial-gradient(circle at center, #0a0a20 0%, #1a0a33 100%); /* Deep night sky background */

        .firework-container {
            position: absolute;
            top: 0;
            left: 0;

        /* --- General Particle Keyframes --- */
        @keyframes explode {
            0% {
                transform: scale(0) translate(0, 0);
                opacity: 1;
                filter: blur(0px);
            }
            100% {
                transform: scale(1.5) translate(var(--dx), var(--dy));
                opacity: 0;
                filter: blur(15px);
            }
        }

        /* --- Specific Firework Styles and Animations --- */

        /* Firework 1: Bright Red Cascade (Midground) */
        #fw1 {
            width: 50px;

            border-radius: 50%;
            position: absolute;
            top: 50%;
            left: 50%;
            transform: translate(-50%, -50%);
            animation: explode 2.5s ease-out forwards;

<body>

    <div class="firework-container">
        <!-- Multiple fireworks representing different explosions -->
        <div id="fw1"></div>
        <div id="fw2"></div>
        <div id="fw3"></div>
        <div id="fw4"></div>
        <div id="fw5"></div>
    </div>

    <script>
        // JavaScript is used here to dynamically position the explosions across the screen
        // simulating a realistic spread effect, using window dimensions for positioning.

        const container = document.querySelector('.firework-container');
        const width = window.innerWidth;
        const height = window.innerHeight;

        // Function to spawn a firework at a random location and time
        function launchFirework(elementId, delay, color) {
            const fw = document.getElementById(elementId);
            if (!fw) return;

            // Calculate random position within the screen boundaries
            const startX = Math.random() * width * 0.8 + (width * 0.1); // Keep near center slightly off-center
            const startY = Math.random() * height * 0.8 + (height * 0.1);

            // Set initial position (off-screen top/left)
            fw.style.left = `${startX}px`;
            fw.style.top = `${startY}px`;

            // Set custom CSS variables for the 'explode' keyframes to define the spread direction
            // We use JS to define the final destination coordinates for the CSS animation targets (dx, dy)
            // Since CSS handles the actual transformation in this setup, we simulate the spread effect by slightly altering the starting point if we were doing complex trajectory, but for simplicity in this CSS-driven setup, we rely on the CSS definitions alone for the burst effect.
        }

        // Event listener to trigger fireworks periodically based on time
        function triggerFireworks() {
            // Trigger a new firework every 1.5 seconds, cycling through the defined elements
            const t = Date.now() / 1000;

            // Launching FW1 (Red)
            launchFirework('fw1', t + 0.5, '#ff4444');

            // Launching FW2 (Blue)
            launchFirework('fw2', t + 1.0, '#44ffff');

            // Launching FW3 (Gold)
            launchFirework('fw3', t + 1.5, '#ffff66');

            // Launching FW4 (Purple)
            launchFirework('fw4', t + 2.0, '#9933ff');

            // Launching FW5 (Green)
            launchFirework('fw5', t + 2.5, '#00ff99');
        }

        // Start the animation loop
        // We use requestAnimationFrame for smooth timing, although the CSS handles the visual timing here.
        let lastLaunchTime = 0;
        const interval = 1500; // Time between major bursts

        function animate() {
            const currentTime = Date.now() / 1000;

            if (currentTime - lastLaunchTime > interval) {
                triggerFireworks();
                lastLaunchTime = currentTime;
            }
            requestAnimationFrame(animate);
        }

        // Initial setup call
        animate();

    </script>
</body>
