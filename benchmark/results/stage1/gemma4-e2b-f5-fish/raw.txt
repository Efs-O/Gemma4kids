```html
<!DOCTYPE html>
<html lang="en">
<head>
    <meta charset="UTF-8">
    <meta name="viewport" content="width=device-width, initial-scale=1.0">
    <title>Colorful Ocean Animation</title>
    <style>
        body {
            margin: 0;
            padding: 0;
            background: linear-gradient(to bottom, #0077be, #00bfff, #00ffff); /* Vivid blue sea gradient */
            overflow: hidden;
            height: 100vh;
            font-family: sans-serif;
        }

        /* --- Fish Styling --- */
        .fish {
            position: absolute;
            width: 60px;
            height: 30px;
            border-radius: 50% 50% 0 0;
            background-color: var(--fish-color);
            box-shadow: 0 0 10px var(--fish-color), 0 0 20px rgba(255, 255, 255, 0.5);
            animation: swim var(--speed) ease-in-out infinite alternate;
            top: 50%;
            transform: translateY(-50%);
        }

        /* Define the swimming animation */
        @keyframes swim {
            0% {
                transform: translateY(-50%) translateX(0);
            }
            100% {
                transform: translateY(-50%) translateX(100vw);
            }
        }

        /* --- Bubble Styling --- */
        .bubble {
            position: absolute;
            border-radius: 50%;
            background-color: rgba(255, 255, 255, 0.8);
            opacity: 0; /* Start invisible */
            animation: float var(--bubble-duration) linear infinite;
            pointer-events: none; /* Ignore mouse events */
        }

        /* Define the floating animation */
        @keyframes float {
            0% {
                transform: translateY(0) scale(1);
                opacity: 0.8;
            }
            100% {
                transform: translateY(-100vh) scale(1.5);
                opacity: 0;
            }
        }
    </style>

    <!-- Container for the animation -->
    <div id="ocean-container"></div>

</head>
<body>

    <div id="ocean-container">
        <!-- Fish Elements will be dynamically populated by JS -->
    </div>

    <script>
        const container = document.getElementById('ocean-container');
        const numberOfFish = 15;
        const numberOfBubbles = 50;

        // Vivid color palette for fish
        const fishColors = [
            '#ff6f61', // Coral Red
            '#4dd0e1', // Bright Cyan
            '#a1c4e2', // Light Blue
            '#fff176', // Bright Yellow
            '#81c784', // Lime Green
            '#f06292', // Hot Pink
            '#64b5f6'  // Sky Blue
        ];

        // 1. Create Fish
        for (let i = 0; i < numberOfFish; i++) {
            const fish = document.createElement('div');
            fish.className = 'fish';
            
            // Assign random color and speed for variation
            const randomColor = fishColors[Math.floor(Math.random() * fishColors.length)];
            const randomSpeed = 1 + Math.random() * 2; // Speed between 1 and 3

            fish.style.setProperty('--fish-color', randomColor);
            fish.style.setProperty('--speed', `${randomSpeed}s`);
            
            // Random starting horizontal position
            fish.style.left = `${Math.random() * 100}%`;
            
            // Stagger the start time to ensure staggered movement
            fish.style.animationDelay = `${Math.random() * 3}s`;
            
            container.appendChild(fish);
        }

        // 2. Create Bubbles
        for (let i = 0; i < numberOfBubbles; i++) {
            const bubble = document.createElement('div');
            bubble.className = 'bubble';
            
            // Random size and duration for variation
            const bubbleSize = 10 + Math.random() * 30;
            const bubbleDuration = 5 + Math.random() * 10; // Duration between 5 and 15 seconds
            
            bubble.style.width = `${bubbleSize}px`;
            bubble.style.height = `${bubbleSize}px`;
            bubble.style.setProperty('--bubble-duration', `${bubbleDuration}s`);
            
            // Random starting position
            bubble.style.left = `${Math.random() * 100}%`;
            bubble.style.top = `${Math.random() * 100}%`;
            
            // Random starting delay
            bubble.style.animationDelay = `${Math.random() * 5}s`;
            
            container.appendChild(bubble);
        }
    </script>

</body>
</html>
```
