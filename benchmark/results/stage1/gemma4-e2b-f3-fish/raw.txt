    <script>
        // JavaScript to dynamically spawn more bubbles to ensure continuous flow
        const container = document.querySelector('.scene-container');
        const numberOfBubbles = 50;

        for (let i = 1; i <= numberOfBubbles; i++) {
            const bubble = document.createElement('div');
            bubble.classList.add('bubble');
            
            // Randomize starting position, size, and animation timing for a natural effect
            const startX = Math.random() * 100;
            const startY = Math.random() * 100;
            const duration = 6 + Math.random() * 6; // Duration between 6s and 12s
            const delay = Math.random() * 15; // Stagger delay

            bubble.style.left = `${startX}%`;
            bubble.style.top = `${startY}%`;
            bubble.style.animationDuration = `${duration}s`;
            bubble.style.animationDelay = `${delay}s`;
            
            // Vary bubble appearance colors
            const colorHue = Math.floor(Math.random() * 360);
            bubble.style.backgroundColor = `hsl(${colorHue}, 80%, 70%)`;

            container.appendChild(bubble);
        }
    </script>
