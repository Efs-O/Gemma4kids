Wow, a rocket launch into space sounds super exciting! I can make a beautiful animation for you with CSS and a little bit of JavaScript to make the stars twinkle. Get ready for liftoff! 🚀✨

```html
<!DOCTYPE html>
<html lang="en">
<head>
    <meta charset="UTF-8">
    <meta name="viewport" content="width=device-width, initial-scale=1.0">
    <title>Rocket Launch</title>
    <style>
        :root {
            --space-blue: hsl(210, 100%, 50%);
            --space-dark: hsl(220, 10%, 20%);
            --rocket-color: #ff4500; /* Bright orange/red for the rocket */
            --star-color-1: #ffffaa;
            --star-color-2: #ffffff;
            --animation-duration: 8s;
        }

        body {
            background: linear-gradient(to bottom, var(--space-dark), var(--space-blue));
            margin: 0;
            overflow: hidden;
            height: 100vh;
            font-family: sans-serif;
        }

        .scene {
            width: 100%;
            height: 100%;
            position: relative;
        }

        /* --- Stars Background --- */
        .star {
            position: absolute;
            background: var(--star-color-2);
            border-radius: 50%;
            opacity: 0; /* Start hidden, JS will make them visible */
        }

        /* --- Rocket Styling --- */
        .rocket {
            position: absolute;
            bottom: 50px; /* Start near the bottom */
            left: 50%;
            transform: translateX(-50%);
            width: 50px;
            height: 100px;
            background: var(--rocket-color);
            border-radius: 10px 10px 0 0;
            box-shadow: 0 0 15px rgba(255, 69, 0, 0.8);
            animation: launch 8s ease-in-out forwards;
        }

        /* Rocket Body */
        .rocket::before {
            content: '🚀'; /* Use an emoji for a fun rocket look */
            font-size: 40px;
            position: absolute;
            top: -20px;
            left: 50%;
            transform: translateX(-50%);
        }

        /* --- Animation Keyframes --- */
        @keyframes launch {
            0% {
                bottom: 50px;
                transform: translate(-50%, 0);
                opacity: 1;
            }
            100% {
                bottom: 80vh; /* Launching into space */
                transform: translate(-50%, 0);
                opacity: 0.5; /* Fades slightly as it goes high */
            }
        }

        /* --- Star Animation (Twinkling) --- */
        .star:nth-child(odd) {
            animation: twinkle var(--animation-duration) linear infinite;
        }
        .star:nth-child(even) {
            animation: twinkle var(--animation-duration) linear infinite;
        }

        @keyframes twinkle {
            0%, 100% { opacity: 0.8; }
            50% { opacity: 0.3; }
        }

    </style>
</head>
<body>
    <div class="scene">
        <!-- Stars will be generated by JavaScript -->
        <div class="rocket"></div>
    </div>

    <script>
        const scene = document.querySelector('.scene');
        const numberOfStars = 100;

        // 1. Generate Stars
        for (let i = 0; i < numberOfStars; i++) {
            const star = document.createElement('div');
            star.classList.add('star');
            
            // Random position across the screen
            const x = Math.random() * 100;
            const y = Math.random() * 100;
            
            star.style.left = `${x}vw`;
            star.style.top = `${y}vh`;
            
            // Random size for variety
            const size = Math.random() * 3;
            star.style.width = `${size}px`;
            star.style.height = `${size}px`;
            
            scene.appendChild(star);
        }
    </script>
</body>
</html>
```
