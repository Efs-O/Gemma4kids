```html
<!DOCTYPE html>
<html lang="en">
<head>
    <meta charset="UTF-8">
    <meta name="viewport" content="width=device-width, initial-scale=1.0">
    <title>Gentle Falling Snowflakes Animation</title>
    <style>
        :root {
            --sky-color: #0a0f2c; /* Deep twilight blue */
            --flake-color: #ffffff;
            --glow-color: rgba(173, 216, 230, 0.8); /* Light blue glow */
        }

        body {
            background-color: var(--sky-color);
            overflow: hidden; /* Crucial to keep flakes contained */
            margin: 0;
            padding: 0;
            height: 100vh;
            display: flex;
            align-items: center;
            justify-content: center;
        }

        /* Container for all flakes to manage the spread */
        .snowfall-container {
            position: absolute;
            width: 100%;
            height: 100%;
            pointer-events: none; /* Allows interaction with underlying content if added */
        }

        /* The Snowflake Base Style */
        .snowflake {
            position: absolute;
            width: 10px;
            height: 10px;
            background: var(--flake-color);
            border-radius: 50%;
            box-shadow: 0 0 5px var(--glow-color), 0 0 10px var(--glow-color);
            opacity: 0.8;
            animation: fall linear infinite;
            /* Ensure initial position is randomized via JS or by applying transforms */
        }

        /* Keyframe Animation for Falling */
        @keyframes fall {
            0% {
                transform: translateY(-10vh) translateX(0) scale(1);
                opacity: 0.8;
            }
            100% {
                /* Moves significantly down and drifts slightly horizontally */
                transform: translateY(110vh) translateX(100vw) scale(0.5);
                opacity: 0.2;
            }
        }

        /* --- Randomization Classes (Used by JavaScript to set unique animation properties) --- */

        /* Speed variations (Duration) */
        .s-fast { animation-duration: 15s; }
        .s-medium { animation-duration: 25s; }
        .s-slow { animation-duration: 40s; }

        /* Horizontal Drift & Size variations (Scale/Spread) */
        .s-drift-left { left: 0%; animation-name: fall-left; }
        .s-drift-right { left: 100%; animation-name: fall-right; }
        .s-center { left: 50%; animation-name: fall; }

        /* Specific animation paths to simulate drift */
        @keyframes fall-left {
            0% { transform: translateY(-10vh) translateX(0) scale(1); }
            100% { transform: translateY(110vh) translateX(-150vw) scale(0.5); }
        }
        @keyframes fall-right {
            0% { transform: translateY(-10vh) translateX(0) scale(1); }
            100% { transform: translateY(110vh) translateX(150vw) scale(0.5); }
        }

    </style>
</head>
<body>
    <div class="snowfall-container" id="snowfall">
        <!-- Snowflakes will be generated here by JavaScript -->
    </div>

    <script>
        const container = document.getElementById('snowfall');
        const numberOfSnowflakes = 100;

        /**
         * Creates and animates multiple snowflakes to simulate a continuous snowfall.
         */
        function createSnowflakes() {
            for (let i = 0; i < numberOfSnowflakes; i++) {
                const flake = document.createElement('div');
                flake.classList.add('snowflake');

                // 1. Randomize Size (Variation)
                const size = Math.random() * 6 + 5; // Between 5px and 11px
                flake.style.width = `${size}px`;
                flake.style.height = `${size}px`;

                // 2. Randomize Starting Position (Spread across the screen)
                const startX = Math.random() * 100; // Percentage left
                const startY = Math.random() * -20;  // Start slightly above the viewport
                flake.style.left = `${startX}vw`;
                flake.style.top = `${startY}vh`;
                
                // 3. Randomize Animation Properties (Speed and Drift)
                const speedClasses = ['s-fast', 's-medium', 's-slow'];
                const driftClasses = ['s-center', 's-drift-left', 's-drift-right'];

                // Assign random classes
                flake.classList.add(speedClasses[Math.floor(Math.random() * speedClasses.length)]);
                flake.classList.add(driftClasses[Math.floor(Math.random() * driftClasses.length)]);

                // 4. Randomize Delay (Creates the continuous, staggered effect)
                const delay = Math.random() * 10; // Delay up to 10 seconds
                flake.style.animationDelay = `${delay}s`;

                container.appendChild(flake);
            }
        }

        // Initialize the snowfall animation
        createSnowflakes();

    </script>
</body>
</html>
```
