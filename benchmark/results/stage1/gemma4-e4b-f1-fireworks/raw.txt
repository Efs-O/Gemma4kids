```html
<!DOCTYPE html>
<html lang="en">
<head>
    <meta charset="UTF-8">
    <meta name="viewport" content="width=device-width, initial-scale=1.0">
    <title>Vivid Fireworks Display</title>
    <style>
        body {
            margin: 0;
            padding: 0;
            overflow: hidden; /* Crucial to keep the animation contained */
            background-color: #000011; /* Very dark, deep night sky blue/black */
        }
        canvas {
            display: block;
            width: 100vw;
            height: 100vh;
        }
    </style>
</head>
<body>
    <canvas id="fireworksCanvas"></canvas>

    <script>
        const canvas = document.getElementById('fireworksCanvas');
        const ctx = canvas.getContext('2d');

        // Set canvas dimensions to full window size
        let W = window.innerWidth;
        let H = window.innerHeight;
        canvas.width = W;
        canvas.height = H;

        // --- Color Palette ---
        const COLORS = [
            '#ff006e', // Vivid Pink/Magenta
            '#ff8c00', // Bright Orange
            '#00ffff', // Cyan
            '#ffff00', // Bright Yellow
            '#ff4136', // Vivid Red
            '#7f00ff'  // Deep Violet
        ];

        // --- Particle System Base Class ---
        class Particle {
            constructor(x, y, color, velocityX, velocityY) {
                this.x = x;
                this.y = y;
                this.color = color;
                this.vx = velocityX;
                this.vy = velocityY;
                this.gravity = 0.03;
                this.friction = 0.99;
                this.alpha = 1; // Opacity
                this.decay = 0.015 + Math.random() * 0.01; // Fade rate
                this.size = Math.random() * 3 + 1;
            }

            update() {
                // Apply friction and gravity
                this.vx *= this.friction;
                this.vy *= this.friction;
                this.vy += this.gravity;

                this.x += this.vx;
                this.y += this.vy;
                
                // Fade out
                this.alpha -= this.decay;
            }

            draw() {
                ctx.save();
                ctx.globalAlpha = this.alpha;
                ctx.fillStyle = this.color;
                ctx.beginPath();
                ctx.arc(this.x, this.y, this.size, 0, Math.PI * 2);
                ctx.fill();
                ctx.restore();
            }
        }

        // --- Firework Rocket Class (The ascent phase) ---
        class Rocket {
            constructor(startX, startY, targetX, targetY) {
                this.x = startX;
                this.y = startY;
                this.targetX = targetX;
                this.targetY = targetY;
                this.color = COLORS[Math.floor(Math.random() * COLORS.length)];
                this.speed = 5 + Math.random() * 3;
                this.angle = Math.atan2(targetY - startY, targetX - startX);
                this.vx = Math.cos(this.angle) * this.speed;
                this.vy = Math.sin(this.angle) * this.speed;
                this.trail = []; // For drawing the ascent trail
            }

            update() {
                // Store previous position for trail effect
                this.trail.unshift({ x: this.x, y: this.y });
                if (this.trail.length > 5) {
                    this.trail.pop();
                }

                // Move
                this.x += this.vx;
                this.y += this.vy;
                
                // Check if target reached (or slightly passed)
                const distance = Math.sqrt(Math.pow(this.targetX - this.x, 2) + Math.pow(this.targetY - this.y, 2));
                return distance < 15; // Explode when close enough to target
            }

            draw() {
                // Draw trail
                ctx.strokeStyle = this.color;
                ctx.lineWidth = 1.5;
                ctx.beginPath();
                ctx.moveTo(this.trail[0].x, this.trail[0].y);
                for (let i = 1; i < this.trail.length; i++) {
                    ctx.lineTo(this.trail[i].x, this.trail[i].y);
                }
                ctx.stroke();

                // Draw current point
                ctx.fillStyle = this.color;
                ctx.beginPath();
                ctx.arc(this.x, this.y, 2, 0, Math.PI * 2);
                ctx.fill();
            }
        }

        // --- Global Animation Arrays ---
        let fireworks = [];
        let particles = [];
        
        // --- Firework Generation ---
        function createFirework() {
            const startX = W * (0.1 + Math.random() * 0.8); // Launch from a random spot on the bottom
            const startY = H; 
            
            // Set target high up in the sky
            const targetX = W * (0.1 + Math.random() * 0.8); 
            const targetY = H * (0.1 + Math.random() * 0.4); // Keep explosion above the lower quarter of the screen

            fireworks.push(new Rocket(startX, startY, targetX, targetY));
        }

        function explode(rocket) {
            const count = 60 + Math.floor(Math.random() * 50);
            const explosionColor = rocket.color;

            for (let i = 0; i < count; i++) {
                // Calculate random angle and velocity for explosion spread
                const angle = Math.random() * Math.PI * 2;
                const speed = Math.random() * 8 + 2;
                
                const vx = Math.cos(angle) * speed;
                const vy = Math.sin(angle) * speed;

                particles.push(new Particle(
                    rocket.x, 
                    rocket.y, 
                    explosionColor, 
                    vx, 
                    vy
                ));
            }
        }

        // --- Animation Loop ---
        function animate() {
            // 1. Fading Background Effect (Creates the "afterglow" look)
            // Draw a semi-transparent black rectangle over the whole canvas each frame
            ctx.globalAlpha = 0.1; 
            ctx.fillStyle = '#000011'; 
            ctx.fillRect(0, 0, W, H);
            ctx.globalAlpha = 1; // Reset alpha for drawing objects

            // 2. Update and Draw Fireworks (Rockets)
            for (let i = fireworks.length - 1; i >= 0; i--) {
                const firework = fireworks[i];
                const exploded = firework.update();

                firework.draw();

                if (exploded) {
                    explode(firework);
                    fireworks.splice(i, 1); // Remove the spent rocket
                }
            }

            // 3. Update and Draw Particles
            for (let i = particles.length - 1; i >= 0; i--) {
                const p = particles[i];
                p.update();
                p.draw();

                // Remove particle if it fades out or falls too far
                if (p.alpha <= 0 || p.y > H + 20) {
                    particles.splice(i, 1);
                }
            }

            requestAnimationFrame(animate);
        }

        // --- Event Listeners & Initial Run ---
        window.addEventListener('resize', () => {
            W = window.innerWidth;
            H = window.innerHeight;
            canvas.width = W;
            canvas.height = H;
        });

        // Launch a firework every 700ms to 1500ms
        let fireworkTimer = setInterval(() => {
            createFirework();
        }, 700);

        // Start the animation loop
        animate();
        
    </script>
</body>
</html>
```
