```html
<!DOCTYPE html>
<html lang="en">
<head>
    <meta charset="UTF-8">
    <meta name="viewport" content="width=device-width, initial-scale=1.0">
    <title>Colorful Fireworks</title>
    <style>
        body {
            background-color: #00001a; /* Deep dark blue/night sky */
            margin: 0;
            overflow: hidden; /* Hide scrollbars */
            height: 100vh;
            font-family: sans-serif;
        }

        .firework {
            position: absolute;
            border-radius: 50%;
            pointer-events: none; /* Ensure clicks pass through */
            opacity: 0;
            animation: explode 2s ease-out forwards;
        }

        @keyframes explode {
            0% {
                transform: scale(0.1);
                opacity: 1;
            }
            100% {
                transform: scale(4);
                opacity: 0;
            }
        }
    </style>
</head>
<body>

    <div id="fireworks-container">
        <!-- Fireworks will be injected here by JavaScript -->
    </div>

    <script>
        const container = document.getElementById('fireworks-container');
        const numFireworks = 15; // Number of fireworks to simulate initially

        /**
         * Generates a random color string (vivid hues)
         */
        function getRandomColor() {
            const colors = [
                '#ff0000', // Red
                '#ff7f50', // Orange
                '#ffff00', // Yellow
                '#00ff00', // Green
                '#00ffff', // Cyan
                '#ff00ff', // Magenta
                '#ffffff'  // White
            ];
            return colors[Math.floor(Math.random() * colors.length)];
        }

        /**
         * Creates and animates a single firework explosion
         * @param {number} x - Center X position
         * @param {number} y - Center Y position
         * @param {string} color - The color of the explosion
         */
        function createFirework(x, y, color) {
            const fireworkElement = document.createElement('div');
            fireworkElement.className = 'firework';
            
            // Set initial position relative to the screen
            fireworkElement.style.left = `${x}px`;
            fireworkElement.style.top = `${y}px`;
            fireworkElement.style.backgroundColor = color;

            container.appendChild(fireworkElement);

            // Trigger the animation immediately
            // We need a slight delay to ensure the element is rendered before the animation starts cleanly
            requestAnimationFrame(() => {
                fireworkElement.style.animation = 'explode 2s ease-out forwards';
            });

            // Clean up the element after animation finishes
            fireworkElement.addEventListener('animationend', () => {
                fireworkElement.remove();
            });
        }

        /**
         * Main loop to generate fireworks periodically
         */
        function launchFireworks() {
            const width = window.innerWidth;
            const height = window.innerHeight;

            // Launch fireworks at random positions across the screen
            for (let i = 0; i < numFireworks; i++) {
                const x = Math.random() * width;
                const y = Math.random() * height;
                const color = getRandomColor();
                
                createFirework(x, y, color);
            }

            // Set up a recurring launch timer for continuous effect
            // This ensures fireworks keep appearing after the initial burst
            setTimeout(launchFireworks, 2000); // Launch a new batch every 2 seconds
        }

        // Start the show when the page loads
        launchFireworks();

    </script>
</body>
</html>
```
