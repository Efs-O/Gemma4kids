<!DOCTYPE html>
<html lang="en">
<head>
    <meta charset="UTF-8">
    <meta name="viewport" content="width=device-width, initial-scale=1.0">
    <title>Colorful Fireworks</title>
    <style>
        :root {
            --sky-color-dark: hsl(240, 10%, 20%); /* Deep Navy/Dark Blue */
            --firework-size: 50px;
        }

        body {
            background: linear-gradient(to bottom, #1a1a2e, #16213e); /* Night Sky Gradient */
            margin: 0;
            height: 100vh;
            display: flex;
            justify-content: center;
            align-items: center;
            overflow: hidden;
        }

        .firework {
            position: absolute;
            border-radius: 50%;
            animation: explode var(--duration) ease-out forwards;
            opacity: 0;
        }

        /* New CSS for falling confetti */
        .confetti {
            position: absolute;
            width: 10px;
            height: 10px;
            border-radius: 50%;
            animation: fall var(--duration) linear forwards;
        }

        @keyframes explode {
            0% {
                transform: scale(0.1);
                opacity: 1;
            }
            100% {
                transform: scale(3);
                opacity: 0;
            }
        }

        @keyframes fall {
            0% {
                transform: translateY(-100vh) rotate(0deg);
                opacity: 1;
            }
            100% {
                transform: translateY(100vh) rotate(720deg);
                opacity: 0;
            }
        }
    </style>
</head>
<body>

    <div id="fireworks-container">
        <!-- Fireworks will be added here by JavaScript -->
    </div>

    <script>
        const container = document.getElementById('fireworks-container');

        function createFirework() {
            const firework = document.createElement('div');
            firework.classList.add('firework');

            // Random colors
            const colors = ['hsl(200, 100%, 50%)', 'hsl(300, 100%, 50%)', 'hsl(330, 100%, 50%)', 'hsl(120, 100%, 50%)', 'hsl(240, 100%, 70%)'];
            const randomColor = colors[Math.floor(Math.random() * colors.length)];
            
            // Random size and duration for a more varied effect
            const size = Math.random() * 80 + 40; // Size between 40px and 120px
            const duration = Math.random() * 1 + 1.5; // Duration between 1s and 2.5s

            firework.style.width = `${size}px`;
            firework.style.height = `${size}px`;
            firework.style.backgroundColor = randomColor;
            firework.style.left = `${Math.random() * 100}%`;
            firework.style.top = `${Math.random() * 100}%`;
            firework.style.animationDuration = `${duration}s`;

            container.appendChild(firework);
            
            // Remove the element after animation finishes to keep the DOM clean
            firework.addEventListener('animationend', () => {
                firework.remove();
            });
        }

        function createConfetti() {
            const confetti = document.createElement('div');
            confetti.classList.add('confetti');

            // Random color for confetti
            const colors = ['#ff0000', '#ff7f00', '#ffff00', '#00ff00', '#0000ff', '#4b0082', '#ee82ee'];
            const randomColor = colors[Math.floor(Math.random() * colors.length)];
            
            // Random starting position (left side) and initial delay for staggered start
            confetti.style.backgroundColor = randomColor;
            confetti.style.left = `${Math.random() * 100}%`;
            confetti.style.top = `0px`; // Start from the top
            confetti.style.animationDuration = `${Math.random() * 2 + 2}s`; // Duration between 2s and 4s
            confetti.style.transform = `rotate(${Math.random() * 360}deg)`;
            
            container.appendChild(confetti);
            
            // Remove the element after animation finishes
            confetti.addEventListener('animationend', () => {
                confetti.remove();
            });
        }

        // Start creating fireworks repeatedly!
        // We'll create a new firework every 500 milliseconds!
        setInterval(createFirework, 500);

        // Start creating confetti repeatedly!
        setInterval(createConfetti, 300);

    </script>

</body>
</html>
