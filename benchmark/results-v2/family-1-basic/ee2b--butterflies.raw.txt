        const butterflyContainer = document.body;
        const numButterflies = 20;

        for (let i = 0; i < numButterflies; i++) {
            const butterfly = document.createElement('div');
            butterfly.className = 'butterfly';
            
            // Assign random starting positions
            const startX = Math.random() * window.innerWidth;
            const startY = Math.random() * window.innerHeight;
            
            // Assign random initial color for vividness

            butterfly.style.top = `${startY}px`;

            butterfly.style.animationDuration = `${10 + Math.random() * 10}s`; // Vary flight speed
