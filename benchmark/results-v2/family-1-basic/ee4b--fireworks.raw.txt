```html
<!DOCTYPE html>
<html lang="en">
<head>
    <meta charset="UTF-8">
    <meta name="viewport" content="width=device-width, initial-scale=1.0">
    <title>Colorful Fireworks Display</title>
    <style>
        body {
            margin: 0;
            padding: 0;
            background-color: #000020; /* Deep night sky blue-black */
            overflow: hidden;
            height: 100vh;
            position: relative;
        }

        /* Container for all fireworks */
        .fireworks-container {
            position: absolute;
            top: 0;
            left: 0;
            width: 100%;
            height: 100%;
        }

        /* Keyframes for the overall burst effect */
        @keyframes explode {
            0% {
                transform: translate(0, 0) scale(0.1);
                opacity: 1;
            }
            100% {
                transform: translate(var(--end-x), var(--end-y)) scale(2);
                opacity: 0;
            }
        }

        /* Particle keyframes for fading sparks */
        @keyframes fadeout {
            0% {
                opacity: 1;
                transform: scale(1);
            }
            100% {
                opacity: 0;
                transform: scale(0.1);
            }
        }

        /* Styling for a single firework instance */
        .firework {
            position: absolute;
            width: 10px;
            height: 10px;
            border-radius: 50%;
            background: radial-gradient(circle, var(--color) 0%, rgba(255,255,255,0) 70%);
            pointer-events: none; /* Ensures it doesn't block clicks */
            opacity: 0;
            z-index: 10;
            animation: explode var(--duration) ease-out forwards;
        }
        
        /* Generates the explosion effect by creating multiple fading sparks per firework element */
        .spark {
            position: absolute;
            width: 4px;
            height: 4px;
            border-radius: 50%;
            background-color: var(--color);
            opacity: 1;
            animation: fadeout var(--spark-duration) ease-out forwards;
        }

        /* --- Firework Launch Elements (Simulating the burst effect) --- */
        
        /* Firework 1: Top Left */
        .fw1 {
            left: 10%;
            top: 10%;
            --color: #ff3366;
            --duration: 1.5s;
            --end-x: 15vw;
            --end-y: 20vh;
            animation: explode var(--duration) ease-out forwards;
        }

        /* Firework 2: Center High */
        .fw2 {
            left: 40%;
            top: 5%;
            --color: #33ccff;
            --duration: 1.8s;
            --end-x: -10vw;
            --end-y: 30vh;
            animation: explode var(--duration) ease-out forwards;
        }

        /* Firework 3: Bottom Right */
        .fw3 {
            left: 85%;
            top: 30%;
            --color: #ccff33;
            --duration: 2.0s;
            --end-x: 20vw;
            --end-y: 40vh;
            animation: explode var(--duration) ease-out forwards;
        }

        /* Firework 4: Random Mid-Sky */
        .fw4 {
            left: 60%;
            top: 40%;
            --color: #ff9933;
            --duration: 1.6s;
            --end-x: -25vw;
            --end-y: 25vh;
            animation: explode var(--duration) ease-out forwards;
        }

        /* Firework 5: Bottom Left (Slower, wider) */
        .fw5 {
            left: 5%;
            top: 70%;
            --color: #9933ff;
            --duration: 2.2s;
            --end-x: 40vw;
            --end-y: 50vh;
            animation: explode var(--duration) ease-out forwards;
        }
        
        /* Helper class to render the explosion sparks upon activation */
        .spark-explosion {
            position: absolute;
            width: 6px;
            height: 6px;
            border-radius: 50%;
            animation: fadeout 1.2s ease-out forwards;
        }
    </style>
</head>
<body>
    <div class="fireworks-container" id="fireworks-container">
        <!-- Firework elements are dynamically generated or styled to explode -->
    </div>

    <script>
        const container = document.getElementById('fireworks-container');
        const colors = ['#ff3366', '#33ccff', '#ccff33', '#ff9933', '#9933ff', '#ffffff'];

        /**
         * Generates the visual sparks for a firework explosion at a given position.
         * @param {number} x - The horizontal position (relative to container).
         * @param {number} y - The vertical position (relative to container).
         * @param {string} color - The primary color of the firework.
         */
        function createSpark(x, y, color) {
            const spark = document.createElement('div');
            spark.className = 'spark-explosion';
            
            // Position the spark exactly where the explosion "bursts"
            spark.style.left = `${x}px`;
            spark.style.top = `${y}px`;
            
            // Randomize size slightly
            const size = Math.random() * 6 + 4;
            spark.style.width = `${size}px`;
            spark.style.height = `${size}px`;

            // Set custom properties for CSS animation delay and color
            spark.style.setProperty('--color', color);
            spark.style.setProperty('--spark-duration', `${Math.random() * 0.8 + 0.8}s`); // 0.8s to 1.6s fade
            
            container.appendChild(spark);

            // Clean up the spark element after animation ends
            setTimeout(() => {
                spark.remove();
            }, 1500);
        }

        /**
         * Simulates a firework burst sequence for visual richness.
         * We attach the particle generation to a timed loop rather than relying purely on CSS classes,
         * as the CSS classes only animate the *initial point* of explosion, not the resulting sparks.
         */
        function triggerFirework(startX, startY, endX, endY, color, duration) {
            // 1. Create a visual "launch point" indicator that animates
            const fw = document.createElement('div');
            fw.className = 'firework';
            fw.style.setProperty('--color', color);
            fw.style.setProperty('--duration', `${duration}s`);
            fw.style.setProperty('--end-x', `${endX}px`);
            fw.style.setProperty('--end-y', `${endY}px`);
            
            // Position the initial 'shell' at the bottom of the screen
            fw.style.left = `${startX}px`;
            fw.style.top = `${startY}px`;
            
            container.appendChild(fw);

            // 2. Wait for the CSS animation to reach its peak (just before it fades)
            setTimeout(() => {
                // Calculate the final pixel coordinates where the burst happens
                const burstX = parseFloat(fw.style.left) + parseFloat(fw.style.transform.getAttribute('translate-x') || '0');
                const burstY = parseFloat(fw.style.top) + parseFloat(fw.style.transform.getAttribute('translate-y') || '0');
                
                // Create many individual sparks radiating from the burst point
                for (let i = 0; i < 30; i++) {
                    // Use the center of the explosion point as the origin
                    createSpark(burstX, burstY, color);
                }
                
                // Clean up the shell element
                fw.remove();
            }, duration * 700); // Trigger spark generation slightly before the animation finishes
        }

        /**
         * Main animation loop to create continuous fireworks.
         */
        function startFireworkShow() {
            // Firework 1: Low Left to Mid Sky
            setTimeout(() => {
                triggerFirework(50, 95, 200, 200, '#ff3366', 1.5);
            }, 0);

            // Firework 2: Center High (Rapid)
            setTimeout(() => {
                triggerFirework(350, 95, -100, 150, '#33ccff', 1.8);
            }, 800);
            
            // Firework 3: Right Side (Slow build)
            setTimeout(() => {
                triggerFirework(850, 95, 200, 350, '#ccff33', 2.0);
            }, 2000);

            // Firework 4: Random Mid-Sky
            setTimeout(() => {
                triggerFirework(600, 95, -250, 250, '#ff9933', 1.6);
            }, 3500);
            
            // Firework 5: Bottom Left (Grand finale feel)
            setTimeout(() => {
                triggerFirework(50, 95, 400, 500, '#9933ff', 2.2);
            }, 5000);
            
            // Keep repeating the show randomly
            setInterval(() => {
                const randomColor = colors[Math.floor(Math.random() * colors.length)];
                const startX = Math.random() * 100 + 50; // 50% to 150% width position
                const startY = 90;
                
                // Target coordinates relative to the screen size for a random burst
                const endX = (Math.random() - 0.5) * 600; // -300px to 300px horizontal spread
                const endY = Math.random() * 400 + 100;   // 100px to 500px vertical spread
                const duration = Math.random() * 1.5 + 1.2;
                
                triggerFirework(startX, startY, endX, endY, randomColor, duration);
            }, 2000 + Math.random() * 3000);
        }

        // Start the initial sequence
        startFireworkShow();
    </script>
</body>
</html>
```
