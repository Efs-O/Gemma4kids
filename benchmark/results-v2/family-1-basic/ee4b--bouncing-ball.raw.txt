<!DOCTYPE html>
<html lang="en">
<head>
    <meta charset="UTF-8">
    <meta name="viewport" content="width=device-width, initial-scale=1.0">
    <title>Bouncing Ball Animation</title>
    <style>
        body {
            margin: 0;
            padding: 0;
            background: linear-gradient(135deg, #ff00cc, #33ccff); /* Vivid background gradient */
            overflow: hidden; /* Hides potential scrollbars from the animation */
            height: 100vh;
            display: flex;
            justify-content: center;
            align-items: center;
        }

        #bouncingBall {
            width: 80px;
            height: 80px;
            background-color: #ffcc00; /* Vivid yellow ball color */
            border-radius: 50%;
            position: absolute; /* Allows us to control movement via top/left */
            box-shadow: 0 10px 20px rgba(0, 0, 0, 0.3);
            
            /* Apply the complex animation */
            animation: bounce 10s linear infinite; 
            
            /* Initial position setting (important for keyframe starting point) */
            top: 0;
            left: 0;
        }

        /* 
         * Keyframe Animation for Bouncing Effect 
         * This combines horizontal movement with sinusoidal vertical oscillation 
         * to simulate realistic bouncing physics.
         */
        @keyframes bounce {
            0% {
                transform: translate(-10vw, 0) scale(1);
                /* Initial state */
            }
            10% {
                /* Slightly higher bounce start */
                transform: translate(20vw, 0) scale(1.05);
            }
            20% {
                /* Peak high point */
                transform: translate(50vw, -15vh) scale(1.1);
            }
            30% {
                /* Low point, slight drift right */
                transform: translate(70vw, 20vh) scale(0.95);
            }
            50% {
                /* Highest bounce/peak center */
                transform: translate(100vw, -10vh) scale(1.2);
            }
            60% {
                 /* Coming down, reaching the other side */
                transform: translate(90vw, 15vh) scale(1.1);
            }
            80% {
                /* Near bottom, starting rise */
                transform: translate(30vw, 10vh) scale(0.9);
            }
            100% {
                /* Returns towards the start, wrapping around the visual area */
                transform: translate(0vw, 0) scale(1);
            }
        }

        /* 
         * Fallback/Refinement: Because we are using 'transform' in the keyframes, 
         * we need to adjust the width/height/positioning slightly to ensure the keyframes 
         * translate across the entire visible area without needing to constantly update 'left' and 'top'.
         * By using 'transform: translate()' we are offsetting the position relative to the original (0,0) anchor.
         * We ensure the animation lasts long enough (10s) to traverse the screen width.
         */

    </style>
</head>
<body>

    <div id="bouncingBall"></div>

    <script>
        // JavaScript is used here only to ensure the animation loop feels continuous 
        // or to adjust the total duration if CSS alone proves too rigid.
        // For this setup, the CSS keyframes are robust enough.

        // Optional: To make it truly cyclical, we can slightly adjust the animation timing

    </script>
</body>
</html>
