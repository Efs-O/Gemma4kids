```html
<!DOCTYPE html>
<html lang="en">
<head>
    <meta charset="UTF-8">
    <meta name="viewport" content="width=device-width, initial-scale=1.0">
    <title>Twinkling Night Sky</title>
    <style>
        :root {
            --sky-color-dark: #0a0a2a;
            --star-color: #ffdd00;
            --shooting-star-color: #ff6347;
        }

        body {
            margin: 0;
            height: 100vh;
            background: linear-gradient(to bottom, var(--sky-color-dark), #16213e);
            overflow: hidden;
        }

        .sky {
            position: relative;
            width: 100%;
            height: 100%;
        }

        /* --- Stars --- */
        .star {
            position: absolute;
            background-color: var(--star-color);
            border-radius: 50%;
            opacity: 0; /* Start hidden */
            animation: twinkle 3s infinite ease-in-out;
        }

        @keyframes twinkle {
            0%, 100% { opacity: 0.8; }
            50% { opacity: 0.3; }
        }

        /* --- Shooting Stars --- */
        .shooting-star {
            position: absolute;
            width: 2px;
            height: 80px;
            background-color: var(--shooting-star-color);
            border-radius: 1px;
            animation: shoot 3s ease-in-out forwards;
        }

        @keyframes shoot {
            0% {
                transform: translateX(-10vw) translateY(10vh);
                opacity: 1;
            }
            100% {
                transform: translateX(110vw) translateY(-10vh);
                opacity: 0;
            }
        }
    </style>
</head>
<body>
    <div class="sky">
        <!-- Stars will be added by JavaScript -->
        <!-- Shooting stars will be added by JavaScript -->
    </div>

    <script>
        const sky = document.querySelector('.sky');
        const numStars = 100;
        const shootingStarCount = 5;
        
        // Define a palette of bright rainbow colors
        const rainbowColors = [
            '#ff6347', // Tomato (Red)
            '#ffa500', // Orange
            '#ffff00', // Yellow
            '#32cd32', // Lime Green
            '#0000ff', // Blue
            '#1e90ff', // Dodger Blue
            '#ee82ee'  // Violet
        ];

        // --- Create Stars ---
        for (let i = 0; i < numStars; i++) {
            const star = document.createElement('div');
            star.classList.add('star');
            
            // Random position
            star.style.left = `${Math.random() * 100}%`;
            star.style.top = `${Math.random() * 100}%`;
            
            // Random size for variety
            const size = Math.random() * 3 + 1; // size between 1px and 4px
            star.style.width = `${size}px`;
            star.style.height = `${size}px`;
            
            // Assign a random bright rainbow color
            const randomColor = rainbowColors[Math.floor(Math.random() * rainbowColors.length)];
            star.style.backgroundColor = randomColor;
            
            // Stagger animation start times slightly for a better twinkle effect
            star.style.animationDelay = `${Math.random() * 3}s`; 

            sky.appendChild(star);
        }

        // --- Create Shooting Stars ---
        const shootingStars = [];
        
        function createShootingStar() {
            const star = document.createElement('div');
            star.classList.add('shooting-star');
            
            // Random starting position (left side)
            star.style.left = '0%';
            star.style.top = `${Math.random() * 100}%`;
            
            // Random destination (right side)
            const endX = 100 + Math.random() * 50; // Go across the screen plus a bit more
            star.style.setProperty('--end-x', `${endX}%`);
            star.style.animation = `shoot 3s ease-in-out forwards`;
            
            // Make the shooting star move across the screen
            // We need a bit of a delay before triggering the animation class if we want precise control, 
            // but for simplicity, let's just spawn them randomly and let the CSS handle the movement based on the animation definition.
            
            // We need to ensure the animation runs for each star, so we'll just append it immediately.
            sky.appendChild(star);
            shootingStars.push(star);
        }

        // Spawn shooting stars periodically
        const starInterval = setInterval(createShootingStar, 700); // Spawn a new star every 700ms

    </script>
</body>
</html>
```
