    <script>
        const body = document.body;

        // Helper function to get a random vivid color
        function getRandomColor() {
            const colors = ['#FF5733', '#33FF57', '#3357FF', '#FFBD33', '#A933FF', '#FF33A1'];
            return colors[Math.floor(Math.random() * colors.length)];
        }

        // Function to create the explosion (the main effect)
        function createExplosion(x, y) {
            const colors = ['#FF144B', '#FF8C00', '#00FFFF', '#ADFF2F', '#FFD700', '#EE82EE', '#0000FF', '#39FF14', '#FF00FF'];
            
            for (let i = 0; i < 30; i++) {
                const spark = document.createElement('div');
                spark.className = 'spark';
                
                const color = colors[Math.floor(Math.random() * colors.length)];
                
                spark.style.setProperty('--spark-color', color);
                
                // Random spread directions (simulating an explosion)
                const angle = Math.random() * 2 * Math.PI;
                const distance = Math.random() * 150 + 50; // Explosion radius between 50px and 200px
                
                // Calculate offsets for CSS animation positioning
                const xOffset = Math.cos(angle) * distance;
                const yOffset = Math.sin(angle) * distance;

                spark.style.setProperty('--x-offset', `${xOffset}px`);
                spark.style.setProperty('--y-offset', `${yOffset}px`);
                
                // Set initial position at the target point
                spark.style.left = `${x}px`;
                spark.style.top = `${y}px`;
                
                body.appendChild(spark);

                // Clean up the spark after its animation ends (1.5s + little buffer)
                setTimeout(() => {
                    spark.remove();
                }, 1600);
            }
        }

        // Event listener for mouse clicks (user interaction)
        body.addEventListener('click', (e) => {
            createExplosion(e.clientX, e.clientY);
        });
        
        // Auto-fire fireworks periodically
        function launchFirework() {
            const x = Math.random() * window.innerWidth;
            const y = Math.random() * (window.innerHeight * 0.8); // Keep them mostly in the upper area
            
            // 1. Simulate the rocket rising (Optional visual flair)
            const rocket = document.createElement('div');
            rocket.className = 'rocket-trail';
            rocket.style.left = `${x}px`;
            rocket.style.top = `${window.innerHeight}px`; // Start from bottom
            body.appendChild(rocket);

            // Animate the rocket up to the explosion point
            setTimeout(() => {
                rocket.style.transition = 'top 2s ease-out, opacity 1s ease-out';
                rocket.style.top = `${y}px`;
                rocket.style.opacity = '0';
            }, 10);

            // 2. Create the explosion when the rocket reaches the peak
            setTimeout(() => {
                createExplosion(x, y);
                rocket.remove(); // Remove the trail element
            }, 2100); 
        }

        // Fire a firework every 1.5 to 3 seconds
        setInterval(launchFirework, Math.random() * 1500 + 1500);

    </script>
