    <!-- Butterflies will be added and animated using JavaScript for dynamic movement -->
    <script>
        const body = document.querySelector('body');
        const starsContainer = document.getElementById('stars');
        const butterflyColors = [
            { wing1: '#FF6347', wing2: '#FFD700' }, // Orange/Yellow
            { wing1: '#9370DB', wing2: '#FF69B4' }, // Purple/Pink
            { wing1: '#FF1493', wing2: '#DDA0DD' }  // Deep Pink/Plum
        ];

        const flowerElements = document.querySelectorAll('.flower');
        
        // Function to create a star element
        function createStar(size, duration) {
            const star = document.createElement('div');
            star.className = 'star';
            star.style.width = `${size}px`;
            star.style.height = `${size}px`;
            star.style.left = `${Math.random() * 100}%`;
            star.style.top = `${Math.random() * 100}%`;
            star.style.animationDuration = `${duration}s`;
            star.style.animationDelay = `${Math.random() * 10}s`;
            starsContainer.appendChild(star);
        }

        // Populate stars (creating 100 stars of varied sizes and twinkles)
        for (let i = 0; i < 100; i++) {
            const size = Math.random() * 3 + 1; // Size between 1px and 4px
            const duration = Math.random() * 15 + 10; // Duration between 10s and 25s
            createStar(size, duration);
        }


        // Function to create a butterfly element
        function createButterfly(colorScheme) {
            const butterfly = document.createElement('div');
            butterfly.className = 'butterfly';
            
            // Use the standard wing class structure, but inject dynamic colors for realism
            butterfly.innerHTML = `
                <div class="butterfly-body"></div>
                <div class="wing" style="background: radial-gradient(circle at 50% 20%, ${colorScheme.wing1}, ${colorScheme.wing2}); transform: rotate(-15deg) translate(-15px, 0); left: 15px; top: 5px;"></div>
                <div class="wing" style="background: radial-gradient(circle at 50% 20%, ${colorScheme.wing2}, ${colorScheme.wing1}); transform: rotate(15deg) translate(15px, 0); right: 15px; top: 5px;"></div>
            `;
            
            // Random starting position (spread across the screen)
            const startX = Math.random() * window.innerWidth;
            const startY = Math.random() * (window.innerHeight * 0.7); // Keep them above the flowers
            
            butterfly.style.cssText = `
                left: ${startX}px;
                top: ${startY}px;
                opacity: 1;
                animation: none; /* Remove CSS keyframe animation initially */
            `;

            // Set random target location for animation
            const endX = startX + (Math.random() * 300 - 150); // Move slightly left or right
            const endY = startY - (Math.random() * 100 + 100); // Move slightly up
            
            // Apply a custom animation style that fades them out after a period
            butterfly.style.setProperty('--start-x', startX + 'px');
            butterfly.style.setProperty('--start-y', startY + 'px');
            butterfly.style.setProperty('--end-x', endX + 'px');
            butterfly.style.setProperty('--end-y', endY + 'px');
            butterfly.style.setProperty('--movement-x', (Math.random() * 100) + 'px');
            butterfly.style.setProperty('--movement-y', (Math.random() * 100) + 'px');

            // Use a CSS animation keyframe definition dynamically for smooth flying movement
            const styleSheet = document.styleSheets[document.styleSheets.length];
            styleSheet.insertRule(`
                @keyframes fly-custom {
                    0% { transform: translate(${startX}px, ${startY}px); opacity: 0.8; }
                    10% { transform: translate(${startX + 50}px, ${startY - 20}px); opacity: 1; }
                    50% { transform: translate(${endX + 20}px, ${endY + 30}px); opacity: 1; }
                    100% { transform: translate(${endX + 50}px, ${endY - 10}px); opacity: 0; }
                }
            `, styleSheet.cssRules.length);
            
            butterfly.style.animation = 'fly-custom 15s linear forwards';
            
            return butterfly;
        }

        // Spawn 8 butterflies with different colors and starting positions
        for (let i = 0; i < 8; i++) {
            const randomColor = butterflyColors[Math.floor(Math.random() * butterflyColors.length)];
            const bee = createButterfly(randomColor);
            body.appendChild(bee);
        }
    </script>
