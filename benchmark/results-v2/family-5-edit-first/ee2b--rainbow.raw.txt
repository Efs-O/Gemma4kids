```html
<!DOCTYPE html>
<html lang="en">
<head>
    <meta charset="UTF-8">
    <meta name="viewport" content="width=device-width, initial-scale=1.0">
    <title>Rain to Rainbow</title>
    <style>
        :root {
            --sky-color: #2c3e50;
            --rain-color: #42b883; /* Bright green/blue for rain effect */
            --rainbow-start: 0;
            --rainbow-end: 360deg;
        }

        body {
            margin: 0;
            height: 100vh;
            display: flex;
            justify-content: center;
            align-items: center;
            background-color: var(--sky-color);
            overflow: hidden;
            font-family: 'Arial', sans-serif;
        }

        /* --- Rainbow Container --- */
        #rainbow-container {
            position: absolute;
            top: 0;
            left: 0;
            width: 100%;
            height: 100%;
            pointer-events: none;
            opacity: 0; /* Start hidden */
            transition: opacity 2s ease-in;
        }

        #rainbow-container.visible {
            opacity: 1;
        }

        /* --- Rain Simulation --- */
        .rain-drop {
            position: absolute;
            width: 2px;
            height: 40px;
            background-color: #ffffff; /* White rain */
            animation: fall linear infinite;
            opacity: 0.7;
            box-shadow: 0 0 5px rgba(255, 255, 255, 0.8);
        }

        @keyframes fall {
            from {
                transform: translateY(-100vh);
            }
            to {
                transform: translateY(100vh);
            }
        }

        /* --- Rain Layer Styling (for depth) --- */
        #rain-layer {
            position: absolute;
            top: 0;
            left: 0;
            width: 100%;
            height: 100%;
            z-index: 10;
        }

    </style>
</head>
<body>

    <!-- The layer where the rain falls -->
    <div id="rain-layer"></div>

    <!-- The rainbow effect container -->
    <div id="rainbow-container"></div>

    <script>
        const rainLayer = document.getElementById('rain-layer');
        const rainbowContainer = document.getElementById('rainbow-container');
        const numDrops = 150;

        // 1. Simulate Rain Generation
        function createRain() {
            for (let i = 0; i < numDrops; i++) {
                const drop = document.createElement('div');
                drop.classList.add('rain-drop');
                
                // Random starting position and animation delay for varied flow
                drop.style.left = `${Math.random() * 100}%`;
                drop.style.animationDuration = `${2 + Math.random() * 5}s`; // Vary falling speed
                drop.style.top = `${Math.random() * -100}px`; // Start above the screen

                // Stagger the start time slightly
                drop.style.animationDelay = `${Math.random() * 2}s`;

                rainLayer.appendChild(drop);
            }
        }

        // 2. Sequence the Appearance
        
        // Wait for rain simulation to start (Rain is visible immediately)
        createRain();
        
        // Wait 3 seconds, then trigger the rainbow appearance
        setTimeout(() => {
            rainbowContainer.classList.add('visible');
            
            // Optional: Make the rain fade out slightly to emphasize the rainbow
            // Note: We keep the rain running for the full effect, but the focus shifts.
        }, 3000); 

    </script>
</body>
</html>
```
