        // Optional: Adding a small JavaScript touch to generate sporadic, floating bubbles 
        // to ensure the animation is rich and visible immediately without relying only on static HTML elements.
        const oceanScene = document.querySelector('.ocean-scene');

        function createBubble() {
            const bubble = document.createElement('div');
            bubble.classList.add('bubble');
            
            // Random size and initial position
            const size = Math.random() * 20 + 10; // 10px to 30px
            const leftPos = Math.random() * 90; // 0% to 90%
            const topPos = Math.random() * 50 + 50; // 50% to 100% from bottom
            
            bubble.style.width = `${size}px`;
            bubble.style.height = `${size}px`;
            bubble.style.left = `${leftPos}%`;
            bubble.style.top = `${topPos}%`;

            // Randomize animation properties slightly for natural look
            const duration = Math.random() * 4 + 4; // 4s to 8s
            bubble.style.animationDuration = `${duration}s`;
            bubble.style.animationDelay = `${Math.random() * 1}s`;

            oceanScene.appendChild(bubble);

            // Clean up the element after its animation finishes
            setTimeout(() => {
                bubble.remove();
            }, duration * 1000 + 1000);
        }

        // Trigger bubble generation every 800ms to keep the sea active
        setInterval(createBubble, 800);
