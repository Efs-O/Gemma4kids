```html
<!DOCTYPE html>
<html lang="en">
<head>
    <meta charset="UTF-8">
    <meta name="viewport" content="width=device-width, initial-scale=1.0">
    <title>Vivid Night Sky Animation</title>
    <style>
        body {
            background-color: #0a001a; /* Deep night blue/purple */
            margin: 0;
            overflow: hidden;
            min-height: 100vh;
            position: relative;
        }

        /* --- Star Styling --- */
        .star {
            position: absolute;
            width: 2px;
            height: 2px;
            background-color: white;
            border-radius: 50%;
            box-shadow: 0 0 1px #ffffaa;
            animation: twinkle 3s infinite ease-in-out;
        }

        @keyframes twinkle {
            0%, 100% {
                opacity: 0.8;
                transform: scale(1);
            }
            50% {
                opacity: 1.5;
                transform: scale(1.5);
            }
        }

        /* --- Shooting Star Styling (Base Class) --- */
        .shooting-star {
            position: absolute;
            width: 2px;
            height: 150px;
            background: linear-gradient(to bottom, rgba(255, 255, 0, 1), rgba(255, 100, 255, 0)); /* Bright yellow to pink trail */
            opacity: 0; /* Start invisible */
            animation: shoot 3s linear forwards;
        }

        @keyframes shoot {
            0% {
                transform: translateX(-10vw) translateY(0);
                opacity: 1;
            }
            100% {
                transform: translateX(110vw) translateY(50vh);
                opacity: 0;
            }
        }
    </style>
</head>
<body>

    <div id="sky-container">
        <!-- Stars will be dynamically generated by JavaScript -->
    </div>

    <script>
        const skyContainer = document.getElementById('sky-container');
        const NUM_STARS = 300;

        // 1. Generate Stars
        function generateStars() {
            for (let i = 0; i < NUM_STARS; i++) {
                const star = document.createElement('div');
                star.className = 'star';
                
                // Random position across the viewport
                const xPos = Math.random() * window.innerWidth;
                const yPos = Math.random() * window.innerHeight;
                
                star.style.left = `${xPos}px`;
                star.style.top = `${yPos}px`;

                // Make some stars slightly brighter/larger for variation
                if (Math.random() < 0.1) {
                    star.style.backgroundColor = '#ffddaa'; // Brighter stars
                } else {
                    star.style.backgroundColor = 'white';
                }

                skyContainer.appendChild(star);
            }
        }

        // 2. Generate Shooting Stars
        function generateShootingStars() {
            const NUM_SHOOTERS = 15;
            const shooters = [];

            for (let i = 0; i < NUM_SHOOTERS; i++) {
                const shooter = document.createElement('div');
                shooter.className = 'shooting-star';
                
                // Random starting position (left edge)
                const startX = Math.random() * -20; // Start slightly off-screen left
                shooter.style.left = `${startX}px`;
                
                // Random trajectory/timing
                const duration = 3 + Math.random() * 2; // 3 to 5 seconds
                const delay = Math.random() * 5; // Stagger the start time
                const animationDuration = duration;

                // Set specific animation properties for this star
                shooter.style.animationDuration = `${animationDuration}s`;
                shooter.style.animationDelay = `-${delay}s`; // Start some already running

                // Determine end position based on random trajectory
                const endX = window.innerWidth + (Math.random() * 500); // End far right
                
                // Apply the custom shoot animation properties (We rely on the global @keyframes shoot for the trail effect)
                // Note: Since the keyframe handles the movement, we just need to place the element.
                
                skyContainer.appendChild(shooter);
                shooters.push(shooter);
            }
        }

        // Initialize the scene
        generateStars();
        generateShootingStars();

    </script>
</body>
</html>
```
