```html
<!DOCTYPE html>
<html lang="en">
<head>
    <meta charset="UTF-8">
    <meta name="viewport" content="width=device-width, initial-scale=1.0">
    <title>Autumn Leaves Fall</title>
    <style>
        /* Vivid Autumn Background */
        body {
            background: linear-gradient(to bottom, #ff7e5f, #feb47b, #8c564b, #a3d9a5);
            margin: 0;
            overflow: hidden; /* Keep leaves contained */
            height: 100vh;
            position: relative;
        }

        /* Style for the falling leaves container (optional, mainly for context) */
        .leaf-container {
            position: absolute;
            top: 0;
            left: 0;
            width: 100%;
            height: 100%;
            pointer-events: none; /* Allows interaction with elements underneath if any */
        }

        /* Base style for a single falling leaf */
        .autumn-leaf {
            position: absolute;
            width: 30px;
            height: 30px;
            opacity: 0.8;
            animation-name fall;
            animation-timing-function linear;
            animation-iteration-count infinite;
            will-change: transform, opacity;
        }

        /* Define the gentle falling animation */
        @keyframes fall {
            0% {
                transform: translateY(-100vh) translateX(0) rotate(0deg);
                opacity: 0.9;
            }
            100% {
                transform: translateY(100vh) translateX(calc(var(--drift, 0px))) rotate(720deg);
                opacity: 0.3;
            }
        }
    </style>
</head>
<body>

    <div class="leaf-container" id="leafContainer">
        <!-- Leaves will be injected here by JavaScript -->
    </div>

    <script>
        const container = document.getElementById('leafContainer');
        const numberOfLeaves = 50; // Ensure at least 30
        const fallDuration = 15 + Math.random() * 10; // Vary the fall time slightly
        const leafElements = [];

        // Function to create a single leaf element
        function createLeaf(index) {
            const leaf = document.createElement('div');
            leaf.className = 'autumn-leaf';

            // Randomize size and color for vivid effect
            const size = 20 + Math.random() * 40;
            const colors = ['#ff6b6b', '#ffa500', '#ffeb3b', '#4caf50', '#2ecc71', '#f9a825'];
            const randomColor = colors[Math.floor(Math.random() * colors.length)];
            
            leaf.style.width = `${size}px`;
            leaf.style.height = `${size}px`;
            leaf.style.backgroundColor = randomColor;
            
            // Position randomly across the screen width
            const startX = Math.random() * 100;
            leaf.style.left = `${startX}vw`;
            
            // Calculate horizontal drift based on index for staggered movement
            const drift = (index % 5 === 0 ? 20 : -10) * (Math.random() * 0.5 + 0.5); // Small, gentle drift
            leaf.style.setProperty('--drift', `${drift}px`);

            // Set animation parameters
            leaf.style.animationDuration = `${fallDuration + Math.random() * 5}s`;
            leaf.style.animationDelay = `${Math.random() * 2}s`; // Stagger the start time
            
            container.appendChild(leaf);
            leafElements.push(leaf);
        }

        // Populate the leaves
        for (let i = 0; i < numberOfLeaves; i++) {
            createLeaf(i);
        }
    </script>

</body>
</html>
```
