<!DOCTYPE html>
<html lang="en">
<head>
    <meta charset="UTF-8">
    <meta name="viewport" content="width=device-width, initial-scale=1.0">
    <title>Vivid Bouncing Ball</title>
    <style>
        :root {
            /* Vivid background gradient for the "floor" */
            --bg-start: #1a0333; /* Deep Indigo/Violet */
            --bg-end: #4d0085;   /* Rich Purple */
        }

        body {
            margin: 0;
            padding: 0;
            height: 100vh;
            width: 100vw;
            background: linear-gradient(135deg, var(--bg-start) 0%, var(--bg-end) 100%);
            overflow: hidden;
            display: flex;
            justify-content: center;
            align-items: flex-end; /* Keep action near the floor */
            position: relative;
        }

        /* The floor area for bouncing physics context */
        .bouncing-floor {
            position: absolute;
            bottom: 0;
            left: 0;
            width: 100%;
            height: 5%; /* Simulating the ground level */
            background: radial-gradient(ellipse at bottom, rgba(255, 255, 255, 0.1) 0%, transparent 70%);
            z-index: 1;
        }

        #bouncingBall {
            width: 100px; /* Minimum 80px */
            height: 100px;
            background-color: #FF007A; /* Bright Magenta */
            border-radius: 50%;
            position: fixed; /* Fixed positioning to stay in view area */
            bottom: 0; /* Start at the bottom */
            left: 50%;
            transform: translateX(-50%);
            box-shadow: 0 10px 30px rgba(255, 0, 122, 0.9);
            /* Animation applied: Repeats indefinitely */
            animation: bounce 2.5s cubic-bezier(0.42, 0, 0.58, 1.1) infinite;
            z-index: 10;
        }

        /* Keyframes defining the bounce physics simulation */
        @keyframes bounce {
            0%, 100% {
                /* Bottom position (touching the simulated floor) */
                transform: translate(-50%, 0) scale(1);
                animation-timing-function: cubic-bezier(0.42, 0, 0.58, 1.1); /* Springy feel */
            }
            50% {
                /* Peak bounce position (highest point) */
                transform: translate(-50%, -350px) scale(1.1);
                animation-timing-function: cubic-bezier(0.4, 0, 0.6, 1.1); /* Flattens slightly at the peak */
            }
        }

    </style>
</head>
<body>

    <div id="bouncingBall"></div>
    <div class="bouncing-floor"></div>

    <script>
        // Optional JS touch: Slightly alters the bounce effect on click for added dynamism
        const ball = document.getElementById('bouncingBall');
        
        document.body.addEventListener('click', () => {
            // Temporarily change animation properties on click to give a responsive feel
            // Note: To reset the physics accurately, we typically need to re-apply the animation 
            // or use a requestAnimationFrame loop, but for simple interaction, we'll just trigger a visual 'hop'.
            
            // Quick visual 'hop' by resetting the transform slightly
            ball.style.transition = 'transform 0.2s ease-out';
            ball.style.transform = 'translate(-50%, -10px) scale(1.05)';
            
            // Reset after a short delay to let the animation kick back in
            setTimeout(() => {
                 ball.style.transition = 'none'; // Temporarily remove transition for immediate keyframe restart
                 // Forcing a reflow/repaint helps the keyframe animation restart cleanly
                 ball.offsetHeight; 
                 ball.style.animationPlayState = 'running';
            }, 200);
        });
    </script>

</body>
</html>
