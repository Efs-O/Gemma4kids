```html
<!DOCTYPE html>
<html lang="en">
<head>
    <meta charset="UTF-8">
    <meta name="viewport" content="width=device-width, initial-scale=1.0">
    <title>Rocket Launch to Space</title>
    <style>
        body {
            margin: 0;
            padding: 0;
            height: 100vh;
            overflow: hidden;
            /* Deep space gradient background */
            background: radial-gradient(circle at 50% 0%, #0a0a2a 0%, #1c0c44 70%, #000011 100%);
        }

        /* Starfield Background Effect */
        .starfield {
            position: absolute;
            top: 0;
            left: 0;
            width: 100%;
            height: 100%;
            pointer-events: none;
            background-image: radial-gradient(white 1px, transparent 1px),
                              radial-gradient(white 1px, transparent 1px);
            background-size: 50px 50px;
            animation: twinkle 60s linear infinite;
        }

        @keyframes twinkle {
            from { background-position: 0 0; }
            to { background-position: 5000px 5000px; }
        }

        /* Rocket Styling */
        .rocket-container {
            position: absolute;
            bottom: 0;
            left: 50%;
            transform: translateX(-50%);
            z-index: 10;
        }

        .rocket {
            width: 60px;
            height: 120px;
            background: linear-gradient(to top, #ff4500, #ff6347, #ffa500); /* Vivid orange/red body */
            border-radius: 30px 30px 0 0;
            position: relative;
            box-shadow: 0 0 15px #ff8c00, inset 0 0 10px rgba(255, 255, 0, 0.8);
            transform-origin: bottom center;
            /* Initial position for animation start */
            opacity: 1;
        }

        .rocket::before {
            content: '';
            position: absolute;
            top: -15px;
            left: 10px;
            width: 0px;
            height: 0px;
            background: #1e90ff; /* Blue cockpit window */
            border-radius: 5px;
        }

        /* Exhaust/Trail Effect */
        .exhaust {
            position: absolute;
            bottom: -10px;
            left: 50%;
            transform: translateX(-50%);
            width: 50px;
            height: 40px;
            background: radial-gradient(circle, rgba(255, 165, 0, 0.9) 0%, rgba(255, 69, 0, 0.5) 60%, rgba(255, 255, 0, 0) 100%);
            border-radius: 50% / 100% 100% 0 0;
            filter: blur(10px);
            animation: exhaustFire 0.2s linear infinite;
            pointer-events: none;
        }

        /* Animation Keyframes */

        /* Rocket Launch Animation */
        @keyframes launch {
            0% {
                transform: translateX(-50%) translateY(0);
                opacity: 1;
            }
            100% {
                /* Launch far up and slightly off center */
                transform: translateX(-50%) translateY(-120vh);
                opacity: 0.5;
            }
        }

        /* Exhaust Fire Animation (Creates pulsing look) */
        @keyframes exhaustFire {
            0%, 100% {
                opacity: 0.7;
                transform: scale(1);
                box-shadow: 0 0 15px 5px rgba(255, 69, 0, 0.8);
            }
            50% {
                opacity: 0.3;
                transform: scale(1.2);
                box-shadow: 0 0 25px 10px rgba(255, 140, 0, 0.6);
            }
        }

        /* Apply animation to the rocket body */
        .rocket-launch {
            animation-name: launch;
            animation-duration: 4s; /* Controls the visible launch time */
            animation-timing-function: cubic-bezier(0.1, 0.5, 0.3, 1); /* Slight acceleration feel */
            animation-delay: 0.5s; /* Wait slightly before launch */
        }
    </style>
</head>
<body>

    <!-- Starfield Background Layer -->
    <div class="starfield"></div>

    <!-- Rocket Structure -->
    <div class="rocket-container">
        <div class="rocket" id="rocket">
            <!-- The ::before pseudo-element acts as the cockpit -->
        </div>
        <!-- Exhaust element - attached to the container for correct positioning -->
        <div class="exhaust" id="exhaust"></div>
    </div>

    <script>
        // Since the animation is purely CSS keyframe based on the .rocket-launch class,
        // we just need a small JS script to add the class after a delay to trigger the visual effect.
        
        window.onload = function() {
            const rocketElement = document.getElementById('rocket');
            const container = document.querySelector('.rocket-container');

            // Add the launch class after a short delay to ensure CSS rendering picks it up
            setTimeout(() => {
                rocketElement.classList.add('rocket-launch');
                container.style.animationDuration = '4s'; // Ensure JS confirms the animation duration
            }, 100);
        };
    </script>
</body>
</html>
```
