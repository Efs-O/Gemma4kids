```html
<!DOCTYPE html>
<html lang="en">
<head>
    <meta charset="UTF-8">
    <meta name="viewport" content="width=device-width, initial-scale=1.0">
    <title>Rocket Launch to Space</title>
    <style>
        body {
            margin: 0;
            overflow: hidden;
            /* Vivid space background */
            background: radial-gradient(circle at center, #00001a 0%, #050033 50%, #000088 100%);
            background-image: radial-gradient(circle, rgba(255, 255, 255, 0.1) 1px, transparent 1px);
            background-size: 50px 50px;
        }

        #spaceCanvas {
            position: absolute;
            top: 0;
            left: 0;
            width: 100vw;
            height: 100vh;
            z-index: 1;
        }

        #rocket {
            position: absolute;
            bottom: 10%; /* Start near the bottom */
            left: 50%;
            transform: translateX(-50%);
            width: 80px;
            height: 150px;
            background: linear-gradient(to top, #ff4d4d, #ffaa00); /* Vivid rocket color */
            border-radius: 10px 10px 20px 20px;
            box-shadow: 0 0 20px #ffaa00, 0 0 40px #ff4d4d;
            z-index: 10;
            /* Initial state for animation */
            transition: bottom 3s ease-out; /* Smooth launch path */
        }

        .exhaust {
            position: absolute;
            bottom: -30px;
            left: 50%;
            transform: translateX(-50%);
            width: 20px;
            height: 50px;
            background-color: #ff8800;
            border-radius: 50%;
            box-shadow: 0 0 15px #ffaa00;
            animation: thrust 0.1s infinite alternate;
            z-index: 11;
        }

        @keyframes thrust {
            from { transform: translate(-50%, 0) scaleY(1); opacity: 1; }
            to { transform: translate(-50%, -10px) scaleY(1.1); opacity: 0.9; }
        }
    </style>
</head>
<body>

    <canvas id="spaceCanvas"></canvas>
    <div id="rocket">
        <div class="exhaust"></div>
    </div>

    <script>
        const canvas = document.getElementById('spaceCanvas');
        const ctx = canvas.getContext('2d');
        const rocket = document.getElementById('rocket');
        
        let width = window.innerWidth;
        let height = window.innerHeight;
        
        canvas.width = width;
        canvas.height = height;

        // --- Starfield Implementation ---
        const starfield = [];
        const numStars = 60;

        function createStar() {
            return {
                x: Math.random() * width,
                y: Math.random() * height,
                size: Math.random() * 2 + 0.5, // Size between 0.5 and 2.5
                opacity: Math.random() * 0.8 + 0.2, // Opacity between 0.2 and 1.0
                twinkleSpeed: Math.random() * 0.05 // For subtle twinkling
            };
        }

        function initializeStars() {
            for (let i = 0; i < numStars; i++) {
                starfield.push(createStar());
            }
        }

        function drawStars() {
            ctx.fillStyle = 'rgba(255, 255, 255, 1)';
            
            starfield.forEach(star => {
                // Twinkling effect based on time/frame for subtle animation
                const twinkle = Math.sin(Date.now() * 0.001 + star.twinkleSpeed) * 0.2;
                ctx.globalAlpha = Math.max(0.2, star.opacity + twinkle);
                
                ctx.beginPath();
                ctx.arc(star.x, star.y, star.size, 0, Math.PI * 2);
                ctx.fill();
            });
            ctx.globalAlpha = 1.0; // Reset global alpha
        }

        // --- Rocket Launch Animation ---
        
        // Trigger the launch animation after a short delay
        setTimeout(() => {
            // Set the final position (near the top)
            rocket.style.bottom = `${height * 0.9}px`; 
            
            // Start the continuous animation loop
            requestAnimationFrame(gameLoop);
        }, 500);


        function gameLoop() {
            // 1. Clear the canvas for fresh drawing (background remains dark space)
            ctx.clearRect(0, 0, width, height);
            
            // 2. Draw the Starfield (stars are drawn directly on canvas for performance and control)
            drawStars();

            // 3. Update Rocket Position (if not using DOM transition for primary move)
            // Since the rocket is primarily controlled by CSS transition for the main launch effect, 
            // we focus the canvas drawing on the backdrop.

            // Request next frame
            requestAnimationFrame(gameLoop);
        }

        // Handle resizing
        window.addEventListener('resize', () => {
            width = window.innerWidth;
            height = window.innerHeight;
            canvas.width = width;
            canvas.height = height;
            // Re-initialize starfield positions if needed (optional, but good practice)
        });

        // Initial setup
        initializeStars();
        // Start the drawing loop immediately to establish the static starfield background
        requestAnimationFrame(gameLoop);
    </script>
</body>
</html>
```
