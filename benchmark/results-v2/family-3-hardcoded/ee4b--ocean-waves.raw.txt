<!DOCTYPE html>
<html lang="en">
<head>
    <meta charset="UTF-8">
    <meta name="viewport" content="width=device-width, initial-scale=1.0">
    <title>Animated Ocean Waves</title>
    <style>
        /* Global Reset and Body Setup */
        body, html {
            margin: 0;
            padding: 0;
            width: 100%;
            height: 100%;
            overflow: hidden;
            font-family: sans-serif;
        }

        /* Background: Sky and Beach Blend */
        .scene {
            position: relative;
            width: 100%;
            height: 100%;
            background: linear-gradient(to bottom, 
                        #87CEEB 0%, /* Bright Sky Blue */
                        #1E90FF 50%, /* Mid Ocean Blue */
                        #0077be 100% /* Deep Water Blue */);
            /* Adding a sandy beach effect near the bottom */
        }

        .beach-sand {
            position: absolute;
            bottom: 0;
            left: 0;
            width: 100%;
            height: 20vh; /* Covers the bottom 20% */
            background: linear-gradient(to bottom, 
                        #FFD700 0%, /* Bright Gold/Yellow Sand */
                        #F0E68C 100%); /* Lighter sand tone */
            z-index: 10;
        }

        /* --- Wave Styling and Animation --- */

        .wave-container {
            position: relative;
            width: 100%;
            height: 40vh; /* Wave area height */
            bottom: 0;
            z-index: 20; /* Must be above the sand */
            overflow: hidden; /* Keeps the wave animation contained */
        }

        .wave-layer {
            position: absolute;
            width: 100%;
            height: 100%;
            animation-name: wave-motion;
            animation-timing-function: linear;
            animation-iteration-count: infinite;
            background-size: 200% 100%; /* Allows pattern to repeat horizontally */
            background-repeat: repeat-x;
            /* The actual wave pattern is defined by gradients */
        }

        /* Layer 1: Deep Water Base (Slower, larger movement) */
        #wave1 {
            background: linear-gradient(
                120deg, 
                rgba(30, 144, 255, 0.7) 0%, 
                rgba(0, 139, 139, 0.7) 50%, 
                rgba(30, 144, 255, 0.7) 100%
            );
            transform: translateX(0);
            animation-duration: 15s;
            animation-delay: 0s;
        }

        /* Layer 2: Mid Water (Medium speed, brighter cyan) */
        #wave2 {
            background: linear-gradient(
                120deg, 
                rgba(0, 191, 255, 0.6) 0%, 
                rgba(0, 200, 255, 0.6) 50%, 
                rgba(0, 191, 255, 0.6) 100%
            );
            transform: translateX(0);
            animation-duration: 10s;
            animation-delay: 3s; /* Offset delay */
        }

        /* Layer 3: Cresting Foam/Crash (Fastest, brightest, simulating impact) */
        #wave3 {
            background: linear-gradient(
                120deg, 
                rgba(255, 255, 255, 0.8) 0%, 
                rgba(255, 255, 255, 0.9) 50%, 
                rgba(255, 255, 255, 0.7) 100%
            );
            transform: translateX(0);
            animation-duration: 8s;
            animation-delay: 1.5s; /* Another offset delay */
        }


        /* KEYFRAMES DEFINITION */
        @keyframes wave-motion {
            0% {
                background-position: -20% 0;
                transform: translateX(0);
            }
            100% {
                /* Shift pattern completely across the width */
                background-position: 20% 0;
                transform: translateX(0);
            }
        }

    </style>
</head>
<body>

    <div class="scene">
        
        <!-- Wave Animation Layers (Bottom up) -->
        <div class="wave-container">
            
            <!-- Layer 1: Deep Water -->
            <div id="wave1" class="wave-layer"></div>
            
            <!-- Layer 2: Mid Water -->
            <div id="wave2" class="wave-layer"></div>
            
            <!-- Layer 3: Foam/Crash (Topmost wave effect) -->
            <div id="wave3" class="wave-layer"></div>

        </div>

        <!-- Beach Sand Layer -->
        <div class="beach-sand"></div>

    </div>

    <script>
        // JavaScript is not strictly required for this CSS animation, 
        // but it's good practice to ensure dynamic elements are accounted for if the scene were complex.
        // For this wave animation, CSS handles the cyclical movement perfectly.
    </script>
</body>
</html>
