<!DOCTYPE html>
<html lang="en">
<head>
    <meta charset="UTF-8">
    <meta name="viewport" content="width=device-width, initial-scale=1.0">
    <title>Colorful Underwater Scene</title>
    <style>
        body {
            margin: 0;
            height: 100vh;
            display: flex;
            justify-content: center;
            align-items: center;
            background: linear-gradient(to bottom, #004d99 0%, #0077be 50%, #00aaff 100%);
            overflow: hidden;
        }

        /* --- Water/Scene Elements --- */

        .scene-container {
            position: relative;
            width: 100vw;
            height: 100vh;
            display: flex;
            justify-content: space-around;
            align-items: flex-end; /* Keep creatures near the 'bottom' of the visual field */
        }

        /* --- Fish Styling --- */
        .fish {
            position: absolute;
            bottom: 10%;
            width: 100px;
            height: 50px;
            border-radius: 50% 50% 50% 50% / 60% 60% 40% 40%; /* Stylized fish body */
            background: linear-gradient(135deg, #ff6b6b, #4ecdc4); /* Vivid colors */
            box-shadow: 0 5px 15px rgba(0, 0, 0, 0.3);
            animation: swim 8s linear infinite alternate;
        }

        /* Specific Fish Placement and Animation */
        .fish:nth-child(1) {
            left: 5%;
            animation-duration: 10s;
        }

        .fish:nth-child(2) {
            left: 30%;
            animation-duration: 12s;
        }

        .fish:nth-child(3) {
            left: 55%;
            animation-duration: 9s;
        }

        /* --- Bubble Styling --- */
        .bubble {
            position: absolute;
            width: 20px;
            height: 20px;
            background: radial-gradient(circle, rgba(255, 255, 255, 0.8) 0%, rgba(255, 255, 255, 0) 100%);
            border-radius: 50%;
            animation: rise 4s ease-out infinite;
            pointer-events: none; /* Ensure bubbles don't interfere with clicks */
        }

        /* Bubble specific colors */
        .bubble:nth-child(odd) {
            background-color: rgba(255, 255, 255, 0.7);
        }
        .bubble:nth-child(even) {
            background-color: rgba(173, 216, 230, 0.7);
        }

        /* --- Keyframe Animations --- */

        /* Fish Swimming Motion */
        @keyframes swim {
            0% {
                transform: translateX(0) translateY(0);
            }
            100% {
                transform: translateX(100vw) translateY(-50px); /* Simulate swimming across the screen */
            }
        }

        /* Bubble Rising Motion */
        @keyframes rise {
            0% {
                transform: translateY(0);
                opacity: 1;
            }
            100% {
                transform: translateY(-100vh); /* Rise up and out of view */
                opacity: 0;
            }
        }
    </style>
</head>
<body>
    <div class="scene-container">
        <!-- Fish 1 -->
        <div class="fish"></div>
        <!-- Fish 2 -->
        <div class="fish"></div>
        <!-- Fish 3 -->
        <div class="fish"></div>

        <!-- Bubbles (Spawned dynamically for visual effect) -->
        <div class="bubble" style="left: 15%; bottom: 20%;"></div>
        <div class="bubble" style="left: 35%; bottom: 30%;"></div>
        <div class="bubble" style="left: 60%; bottom: 15%;"></div>
        <div class="bubble" style="left: 80%; bottom: 40%;"></div>
        <div class="bubble" style="left: 5%; bottom: 50%;"></div>
        <div class="bubble" style="left: 50%; bottom: 5%; animation-delay: 0.5s;"></div>
        <div class="bubble" style="left: 70%; bottom: 25%;"></div>
    </div>

    <script>
        // Simple JS to add more bubbles dynamically to ensure a dense, continuous effect
        const container = document.querySelector('.scene-container');
        const numBubbles = 30;

        for (let i = 1; i <= numBubbles; i++) {
            const bubble = document.createElement('div');
            bubble.classList.add('bubble');
            
            // Random starting position spread across the container
            const randomLeft = Math.random() * 100;
            const randomBottom = Math.random() * 100;
            
            bubble.style.left = `${randomLeft}vw`;
            bubble.style.bottom = `${randomBottom}vh`;
            
            // Randomize the rise time and delay for staggered effect
            bubble.style.animationDuration = `${3 + Math.random() * 3}s`;
            bubble.style.animationDelay = `${Math.random() * 4}s`;
            
            container.appendChild(bubble);
        }
    </script>
</body>
