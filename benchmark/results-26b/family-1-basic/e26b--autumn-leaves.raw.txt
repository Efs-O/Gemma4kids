```html
<!DOCTYPE html>
<html lang="en">
<head>
    <meta charset="UTF-8">
    <meta name="viewport" content="width=device-width, initial-scale=1.0">
    <title>Autumn Leaves Falling</title>
    - <style>
        body, html {
            margin: 0;
            padding: 0;
            width: 100%;
            height: 100%;
            overflow: hidden;
            background: linear-gradient(to bottom, #ff9d5c, #ff5e62, #2c3e50);
        }

        canvas {
            display: block;
        }

        /* Simple overlay to give a warm atmosphere */
        .overlay {
            position: absolute;
            top: 0;
            left: 0;
            width: 100%;
            height: 100%;
            background: radial-gradient(circle, transparent 40%, rgba(0,0,0,0.3) 100%);
            pointer-events: none;
        }
    </style>
</head>
<body>
    <div class="overlay"></div>
    <canvas id="canvas"></canvas>

    <script>
        const canvas = document.getElementById('canvas');
        const ctx = canvas.getContext('2d');

        let width, height;
        let leaves = [];
        const leafCount = 80;
        const colors = [
            '#e67e22', // Orange
            '#d35400', // Burnt Orange
            '#f1c40f', // Yellow
            '#c0392b', // Red
            '#e74c3c', // Bright Red
            '#f39c12', // Gold
            '#7e5109'  // Brown
        ];

        function init() {
            resize();
            createLeaves();
            animate();
        }

        function resize() {
            width = window.innerWidth;
            height = window.innerHeight;
            canvas.width = width;
            canvas.height = height;
        }

        class Leaf {
            constructor() {
                this.reset();
                // Start leaves at random heights so they don't all fall at once
                this.y = Math.random() * height;
            }

            reset() {
                this.x = Math.random() * width;
                this.y = -50;
                this.size = Math.random() * 15 + 10;
                this.color = colors[Math.floor(Math.random() * colors.length)];
                this.speedY = Math.random() * 1.5 + 0.5;
                this.speedX = Math.random() * 2 - 1;
                this.rotation = Math.random() * Math.PI * 2;
                this.rotationSpeed = Math.random() * 0.05 - 0.025;
                this.swing = Math.random() * 2;
                this.swingSpeed = Math.random() * 0.02 + 0.01;
                this.offset = Math.random() * 100;
            }

            update() {
                this.y += this.speedY;
                this.x += Math.sin(this.y * this.swingSpeed + this.offset) * this.swing;
                this.rotation += this.rotationSpeed;

                // If leaf goes off bottom, reset to top
                if (this.y > height + 50) {
                    this.reset();
                }
                // If leaf goes off sides, wrap it
                if (this.x < -50) this.x = width + 50;
                if (this.x > width + 50) this.x = -50;
            }

            draw() {
                ctx.save();
                ctx.translate(this.x, this.y);
                ctx.rotate(this.rotation);
                ctx.beginPath();
                
                // Drawing a simple leaf shape (an ellipse/almond shape)
                ctx.moveTo(0, -this.size);
                ctx.quadraticCurveTo(this.size, 0, 0, this.size);
                ctx.quadraticCurveTo(-this.size, 0, 0, -this.size);
                
                ctx.fillStyle = this.color;
                ctx.fill();
                
                // Add a vein for detail
                ctx.beginPath();
                ctx.moveTo(0, -this.size);
                ctx.lineTo(0, this.size);
                ctx.strokeStyle = 'rgba(0,0,0,0.1)';
                ctx.lineWidth = 1;
                ctx.stroke();
                
                ctx.restore();
            }
        }

        function createLeaves() {
            leaves = [];
            for (let i = 0; i < leafCount; i++) {
                leaves.push(new Leaf());
            }
        }

        function drawTrees() {
            // Draw some dark silhouettes of trees/branches at the bottom
            ctx.fillStyle = '#1a0f00';
            
            // Left tree trunk
            ctx.beginPath();
            ctx.moveTo(0, height);
            ctx.lineTo(width * 0.2, height);
            ctx.lineTo(width * 0.1, height * 0.4);
            ctx.lineTo(-width * 0.1, height * 0.5);
            ctx.fill();

            // Right tree trunk
            ctx.beginPath();
            ctx.moveTo(width, height);
            ctx.lineTo(width * 0.8, height);
            ctx.lineTo(width * 0.9, height * 0.3);
            ctx.lineTo(width * 1.1, height * 0.4);
            ctx.fill();

            // Some branches
            ctx.lineWidth = 15;
            ctx.strokeStyle = '#1a0f00';
            ctx.beginPath();
            ctx.moveTo(width * 0.1, height * 0.4);
            ctx.lineTo(width * 0.3, height * 0.2);
            ctx.stroke();

            ctx.beginPath();
            ctx.moveTo(width * 0.9, height * 0.3);
            ctx.lineTo(width * 0.7, height * 0.15);
            ctx.stroke();
        }

        function animate() {
            // Clear background with a slight trail effect
            ctx.clearRect(0, 0, width, height);
            
            // Draw static background elements first
            drawTrees();

            // Update and draw leaves
            leaves.forEach(leaf => {
                leaf.update();
                leaf.draw();
            });

            requestAnimationFrame(animate);
        }

        window.addEventListener('resize', () => {
            resize();
            createLeaves();
        });

        init();
    </script>
</body>
</html>
```
