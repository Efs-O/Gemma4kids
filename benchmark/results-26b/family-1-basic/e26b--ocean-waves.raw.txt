```html
<!DOCTYPE html>
<html lang="en">
<head>
    <meta charset="UTF-8">
    <meta name="viewport" content="width=device-width, initial-scale=1.0">
    <title>Animated Ocean Waves</title>
    <style>
        body, html {
            margin: 0;
            padding: 0;
            width: 100%;
            height: 100%;
            overflow: hidden;
        }

        canvas {
            display: block;
        }

        /* Overlaying a subtle vignette for depth */
        .overlay {
            position: absolute;
            top: 0;
            left: 0;
            width: 100%;
            height: 100%;
            pointer-events: none;
            background: radial-gradient(circle, transparent 50%, rgba(0,0,0,0.1) 100%);
        }
    </style>
</head>
<body>
    <canvas id="oceanCanvas"></canvas>
    <div class="overlay"></div>

    <script>
        const canvas = document.getElementById('oceanCanvas');
        const ctx = canvas.getContext('2d');

        let width, height;
        let waves = [];
        let particles = [];
        let time = 0;

        // Configuration
        const waveCount = 5;
        const particleCount = 40;
        const colors = [
            'rgba(0, 119, 182, 0.7)',  // Deep Blue
            'rgba(0, 180, 216, 0.6)',  // Ocean Blue
            'rgba(72, 202, 228, 0.5)', // Light Blue
            'rgba(144, 224, 239, 0.4)',// Cyan
            'rgba(255, 255, 255, 0.8)' // Foam/White
        ];

        function resize() {
            width = canvas.width = window.innerWidth;
            height = canvas.height = window.innerHeight;
            initWaves();
            initParticles();
        }

        class Wave {
            constructor(index) {
                this.index = index;
                this.reset();
            }

            reset() {
                this.amplitude = 20 + Math.random() * 30;
                this.frequency = 0.005 + Math.random() * 0.01;
                this.speed = 0.02 + Math.random() * 0.03;
                this.yOffset = height * 0.5 + (this.index * 25);
                this.color = colors[this.index % colors.length];
                this.offset = Math.random() * 100;
            }

            draw() {
                ctx.beginPath();
                ctx.moveTo(0, height);
                
                for (let x = 0; x <= width; x += 10) {
                    // Sine wave math: y = sin(x * freq + time) * amplitude
                    const y = this.yOffset + Math.sin(x * this.frequency + time * this.speed + this.offset) * this.amplitude;
                    ctx.lineTo(x, y);
                }

                ctx.lineTo(width, height);
                ctx.lineTo(0, height);
                ctx.fillStyle = this.color;
                ctx.fill();
                ctx.closePath();
            }
        }

        class Particle {
            constructor() {
                this.reset();
            }

            reset() {
                this.x = Math.random() * width;
                this.y = height * 0.5 + Math.random() * (height * 0.5);
                this.size = Math.random() * 3 + 1;
                this.speedX = Math.random() * 1 - 0.5;
                this.speedY = Math.random() * -1 - 0.5;
                this.alpha = Math.random() * 0.5 + 0.2;
            }

            update() {
                this.x += this.speedX;
                this.y += this.speedY;
                this.alpha -= 0.005;

                if (this.alpha <= 0 || this.y < height * 0.4) {
                    this.reset();
                }
            }

            draw() {
                ctx.beginPath();
                ctx.arc(this.x, this.y, this.size, 0, Math.PI * 2);
                ctx.fillStyle = `rgba(255, 255, 255, ${this.alpha})`;
                ctx.fill();
            }
        }

        function initWaves() {
            waves = [];
            for (let i = 0; i < waveCount; i++) {
                waves.push(new Wave(i));
            }
        }

        function initParticles() {
            particles = [];
            for (let i = 0; i < particleCount; i++) {
                particles.push(new Particle());
            }
        }

        function drawBackground() {
            // Sky Gradient
            const skyGrad = ctx.createLinearGradient(0, 0, 0, height * 0.6);
            skyGrad.addColorStop(0, '#4facfe'); // Bright Blue
            skyGrad.addColorStop(1, '#00f2fe'); // Cyan
            ctx.fillStyle = skyGrad;
            ctx.fillRect(0, 0, width, height * 0.6);

            // Sun
            ctx.beginPath();
            ctx.arc(width * 0.8, height * 0.15, 50, 0, Math.PI * 2);
            const sunGrad = ctx.createRadialGradient(
                width * 0.8, height * 0.15, 10, 
                width * 0.8, height * 0.15, 50
            );
            sunGrad.addColorStop(0, '#fff700');
            sunGrad.addColorStop(1, '#ff9900');
            ctx.fillStyle = sunGrad;
            ctx.shadowBlur = 40;
            ctx.shadowColor = '#ffcc00';
            ctx.fill();
            ctx.shadowBlur = 0; // Reset shadow

            // Sand
            ctx.fillStyle = '#f4d03f'; // Vivid Sand
            ctx.fillRect(0, height * 0.6, width, height * 0.4);
            
            // Texture for sand
            ctx.fillStyle = 'rgba(0,0,0,0.05)';
            for(let i=0; i<width; i+=20) {
                ctx.fillRect(i, height * 0.6, 2, height * 0.4);
            }
        }

        function animate() {
            ctx.clearRect(0, 0, width, height);
            
            drawBackground();

            // Draw waves from back to front
            waves.forEach(wave => wave.draw());

            // Update and draw sparkling particles (sea spray/glitter)
            particles.forEach(p => {
                p.update();
                p.draw();
            });

            time += 1;
            requestAnimationFrame(animate);
        }

        window.addEventListener('resize', resize);

        // Initial start
        resize();
        animate();

    </script>
</body>
</html>
```
