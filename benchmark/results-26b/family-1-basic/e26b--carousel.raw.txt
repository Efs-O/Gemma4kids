<!DOCTYPE html>

<head>

    <title>Magic Neon Carousel</title>
    <style>
        :root {
            --bg-color: #0a0a2e;
            --gold: #ffd700;
            --accent: #ff00ff;
            --cyan: #00ffff;
        }

        body, html {
            margin: 0;
            padding: 0;
            width: 100%;
            height: 100%;
            overflow: hidden;
            background-color: var(--bg-color);
            display: flex;
            justify-content: center;
            align-items: center;
            perspective: 1200px;
        }

        /* Starry Background */
        .stars {
            position: absolute;
            width: 100%;
            height: 100%;
            background: radial-gradient(circle at center, #1b1b4d 0%, #050510 100%);
            z-index: -2;
        }

        .star {
            position: absolute;
            background: white;
            border-radius: 50%;
            opacity: 0.5;
            animation: twinkle var(--d) infinite ease-in-out;
        }

        @keyframes twinkle {
            0%, 100% { opacity: 0.3; transform: scale(1); }
            50% { opacity: 1; transform: scale(1.2); }
        }

        /* The Carousel Stage */
        .scene {
            position: relative;
            width: 100vw;
            height: 100vh;
            display: flex;
            justify-content: center;
            align-items: center;
            transform-style: preserve-3d;
            transform: rotateX(-10deg);
        }

        /* The Central Pole */
        .pole {
            position: absolute;
            width: 20px;
            height: 600px;
            background: linear-gradient(to right, #ffd700, #fff, #ffd700);
            transform: translateZ(0px);
            z-index: 5;
            border-radius: 10px;
        }

        /* The Rotating Ring */
        .carousel-ring {
            position: absolute;
            width: 0;
            height: 0;
            transform-style: preserve-3d;
            animation: spin 15s linear infinite;
        }

        @keyframes spin {
            from { transform: rotateY(0deg); }
            to { transform: rotateY(360deg); }
        }

        /* The Horse Container */
        .horse-wrapper {
            position: absolute;
            transform-style: preserve-3d;
            display: flex;
            justify-content: center;
            align-items: center;
            animation: bob 2s ease-in-out infinite;
        }

        @keyframes bob {
            0%, 100% { transform: translateY(0px); }
            50% { transform: translateY(-30px); }
        }

        /* The Horse Visuals (CSS Art) */
        .horse {
            position: relative;
            width: 60px;
            height: 40px;
            background: var(--horse-color, #ff00ff);
            border-radius: 50% 50% 40% 40%;
            box-shadow: 0 0 20px var(--horse-color);

        }

        /* Horse Head */

<body>

    <div class="stars" id="starfield"></div>

    <div class="scene">
        <div class="floor"></div>
        <div class="pole"></div>
        <div class="roof-top"></div>
        <div class="roof"></div>
        
        <div class="carousel-ring" id="carouselRing">
            <!-- Horses will be injected here by JS -->
        </div>
    </div>

    <script>
        const ring = document.getElementById('carouselRing');
        const starfield = document.getElementById('starfield');
        const horseCount = 12;
        const radius = 250;
        const colors = ['#ff00ff', '#00ffff', '#ffff00', '#ff3300', '#33ff00', '#ff0066', '#00ffcc'];

        // Create Stars
        for (let i = 0; i < 150; i++) {
            const star = document.createElement('div');
            star.className = 'star';
            const size = Math.random() * 3 + 'px';
            star.style.width = size;
far           star.style.height = size;
            star.style.top = Math.random() * 100 + '%';
            star.style.left = Math.random() * 100 + '%';
            star.style.setProperty('--d', (Math.random() * 3 + 2) + 's');
            starfield.appendChild(star);
        }

        // Create Horses
        for (let i = 0; i < horseCount; i++) {
            const angle = (i / horseCount) * 360;
            const color = colors[i % colors.length];
            
            const wrapper = document.createElement('div');
            wrapper.className = 'horse-wrapper';
            
            // Position horse in a circle
            // We rotate the wrapper, then translate it out from the center
            wrapper.style.transform = `rotateY(${angle}deg) translateZ(${radius}px)`;
            
            // Add a small delay to the bobbing for a wave effect
            wrapper.style.animationDelay = (i * 0.1) + 's';

            wrapper.innerHTML = `
                <div class="horse-pole"></div>
                <div class="horse" style="--horse-color: ${color}; --horse-string: ${color};"></div>
            `;
            
            ring.appendChild(wrapper);
        }
    </script>
</body>
