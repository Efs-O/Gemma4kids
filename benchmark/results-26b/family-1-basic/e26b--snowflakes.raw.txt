```html
<!DOCTYPE html>
<html lang="en">
<head>
    <meta charset="UTF-8">
    <meta name="viewport" content="width=device-width, initial-scale=1.0">
    <title>Vivid Falling Snowflakes</title>
    <style>
        body, html {
            margin: 0;
            padding: 0;
            width: 100%;
            height: 100%;
            overflow: hidden;
            background-color: #000814; /* Deep dark space blue */
        }

        canvas {
            display: block;
        }

        /* Overlay a subtle glow to the background */
        .vignette {
            position: absolute;
            top: 0;
            left: 0;
            width: 100%;
            height: 100%;
            background: radial-gradient(circle, transparent 40%, rgba(0, 0, 0, 0.8) 100%);
            pointer-events: none;
        }
    </style>
</head>
<body>
    <div class="vignette"></div>
    <canvas id="snowCanvas"></canvas>

    <script>
        const canvas = document.getElementById('snowCanvas');
        const ctx = canvas.getContext('2d');

    	let width, height, snowflakes;

        // Configuration
        const snowflakeCount = 150;
        const colors = [
            '#00f5ff', // Cyan
            '#ff00ff', // Magenta
            '#7fff00', // Chartreuse
            '#ffffff', // White
            '#0077ff', // Bright Blue
            '#ffea00'  // Yellow
        ];

        class Snowflake {
            constructor() {
                this.init();
            }

            init() {
                this.x = Math.random() * width;
                this.y = Math.random() * height - height; // Start above viewport
                this.radius = Math.random() * 4 + 1;
                this.speedY = Math.random() * 1 + 0.5; // Falling speed
                this.speedX = (Math.random() - 0.5) * 1; // Horizontal drift
                this.drift = Math.random() * 2; // Oscillation amplitude
                this.color = colors[Math.floor(Math.random() * colors.length)];
                this.opacity = Math.random() * 0.5 + 0.5;
                this.angle = Math.random() * Math.PI * 2; // For sine wave movement
            }

            update() {
                this.y += this.speedY;
                this.angle += 0.02;
                // Add a gentle swaying motion using sine
                this.x += Math.sin(this.angle) * 0.5 + this.speedX;

                // If snowflake goes off bottom, reset to top
                if (this.y > height) {
                    this.y = -10;
                    this.x = Math.random() * width;
                }

                // If snowflake goes too far left or right, wrap around
                if (this.x < 0) this.x = width;
                if (this.x > width) this.x = 0;
            }

            draw() {
                ctx.beginPath();
                ctx.arc(this.x, this.y, this.radius, 0, Math.PI * 2);
                ctx.fillStyle = this.color;
                ctx.globalAlpha = this.opacity;
                ctx.shadowBlur = 10;
                ctx.shadowColor = this.color;
                ctx.fill();
                ctx.closePath();
            }
        }

        function resize() {
            width = canvas.width = window.innerWidth;
            height = canvas.height = window.innerHeight;
            
            // Re-initialize snowflakes on resize to ensure they fill the screen
            snowflakes = [];
            for (let i = 0; i < snowflakeCount; i++) {
                snowflakes.push(new Snowflake());
            }
        }

        function animate() {
            // Create a slight trail effect by not clearing fully
            ctx.fillStyle = 'rgba(0, 8, 20, 0.2)';
            ctx.fillRect(0, 0, width, height);

            snowflakes.forEach(flake => {
                flake.update();
                flake.draw();
            });

            requestAnimationFrame(animate);
        }

        // Initialization
        window.addEventListener('resize', resize);
        resize();
        animate();

    </script>
</body>
</html>
```
