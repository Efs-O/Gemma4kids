```html
<!DOCTYPE html>
<html lang="en">
<head>
    <meta charset="UTF-8">
    <meta name="viewport" content="width=device-width, initial-scale=1.0">
    <title>Fun Clown Juggling Animation</title>
    <style>
        body, html {
            margin: 0;
            padding: 0;
            width: 100%;
            height: 100%;
            overflow: hidden;
            background-color: #ffefd5;
        }
        canvas {
            display: block;
        }
    </style>
</head>
<body>

<canvas id="canvas"></canvas>

<script>
    const canvas = document.getElementById('canvas');
    const ctx = canvas.getContext('2d');

    let width, height;
    let balls = [];
    let confetti = [];
    const colors = ['#FF0055', '#00FFCC', '#FFD700', '#FF4500', '#9D00FF', '#00FF00', '#FF1493'];

    // Resize handler
    function resize() {
        width = canvas.width = window.innerWidth;
        height = canvas.height = window.innerHeight;
        initScene();
    }

    class Ball {
        constructor() {
            this.reset();
        }

        reset() {
            this.radius = Math.random() * 15 + 15;
            this.x = width / 2;
            this.y = height - 150;
            this/this.color = colors[Math.floor(Math.random() * colors.length)];
            // Juggling physics: velocity
            this.vx = (Math.random() - 0.5) * 10;
            this.vy = -Math.random() * 15 - 10;
            this.gravity = 0.25;
            this.friction = 0.98;
            this.bounce = -0.8;
        }

        update() {
            this.vy += this.gravity;
            this.x += this.vx;
            this.y += this.vy;

            // Bounce off walls
            if (this.x - this.radius < 0 || this.x + this.radius > width) {
                this.vx = -this.vx;
            }

            // Re-throw logic (The "Juggling" effect)
            // When the ball falls below the clown's hands, push it back up
            if (this.y > height - 100) {
                this.y = height - 110;
                this.vy = -(Math.random() * 10 + 12);
                this.vx = (Math.random() - 0.5) * 14;
            }

            // Boundary bounce (floor)
            if (this.y + this.radius > height) {
                this.y = height - this.radius;
                this.vy *= this.bounce;
            }
        }

        draw() {
            ctx.beginPath();
            ctx.arc(this.x, this.y, this.radius, 0, Math.PI * 2);
            ctx.fillStyle = this.color;
            ctx.fill();
            // Shine effect
            ctx.beginPath();
            ctx.arc(this.x - this.radius*0.3, this.y - this.radius*0.3, this.radius*0.2, 0, Math.PI * 2);
            ctx.fillStyle = 'rgba(255,255,255,0.5)';
            ctx.fill();
            ctx.strokeStyle = 'black';
            ctx.lineWidth = 2;
            ctx.stroke();
            ctx.closePath();
        }
    }

    class Confetti {
        constructor() {
            this.x = Math.random() * width;
            this.y = Math.random() * height;
            this.size = Math.random() * 5 + 2;
            this.color = colors[Math.floor(Math.random() * colors.length)];
            this.speedY = Math.random() * 2 + 1;
            this.speedX = (Math.random() - 0.5) * 2;
        }
        update() {
            this.y += this.speedY;
            this.x += this.speedX;
            if (this.y > height) this.y = -10;
        }
        draw() {
            ctx.fillStyle = this.color;
            ctx.fillRect(this.x, this.y, this.size, this.size);
        }
    }

    function initScene() {
        balls = [];
        for (let i = 0; i < 8; i++) {
            balls.push(new Ball());
        }
        confetti = [];
        for (let i = 0; i < 100; i++) {
            confetti.push(new Confetti());
        }
    }

    function drawBackground() {
        // Circus Stripes
        const stripeWidth = width / 10;
        for (let i = 0; i < 10; i++) {
            ctx.fillStyle = i % 2 === 0 ? '#ff6b6b' : '#ffffff';
            ctx.fillRect(i * stripeWidth, 0, stripeWidth, height);
        }
        // Overlay gradient for depth
        const grad = ctx.createRadialGradient(width/2, height/2, 10, width/2, height/2, width);
        grad.addColorStop(0, 'transparent');
        grad.addColorStop(1, 'rgba(0,0,0,0.2)');
        ctx.fillStyle = grad;
        ctx.fillRect(0,0,width,height);
    }

    function drawClown() {
        const centerX = width / 2;
        const centerY = height - 120;

        // Hair (Big, messy, colorful)
        for(let i=0; i<20; i++) {
            ctx.beginPath();
            ctx.arc(centerX + Math.cos(i)*60, centerY - 60 + Math.sin(i)*60, 40, 0, Math.PI*2);
            ctx.fillStyle = colors[i % colors.length];
            ctx.fill();
        }

        // Face
        ctx.beginPath();
        ctx.arc(centerX, centerY, 70, 0, Math.PI * 2);
        ctx.fillStyle = 'white';
        ctx.fill();
        ctx.strokeStyle = '#333';
        ctx.lineWidth = 3;
        ctx.stroke();

        // Eyes
        ctx.fillStyle = 'black';
        ctx.beginPath();
        ctx.arc(centerX - 25, centerY - 20, 10, 0, Math.PI * 2);
        ctx.arc(centerX + 25, centerY - 20, 10, 0, Math.PI * 2);
        ctx.fill();
        
        // Eye shine
        ctx.fillStyle = 'white';
        ctx.beginPath();
        ctx.arc(centerX - 28, centerY - 23, 4, 0, Math.PI * 2);
        ctx.arc(centerX + 22, centerY - 23, 4, 0, Math.string(0, Math.PI * 2)); // Fix error in logic: just use 0, Math.PI*2
        ctx.fill();
        // Note: Re-writing the eye shine correctly
        ctx.beginPath();
        ctx.arc(centerX + 22, centerY - 23, 4, 0, Math.PI * 2);
        ctx.fill();

        // Red Nose
        ctx.beginPath();
        ctx.arc(centerX, centerY + 5, 20, 0, Math.PI * 2);
        ctx.fillStyle = '#FF0000';
        ctx.fill();
        ctx.stroke();

        // Smile
        ctx.beginPath();
        ctx.arc(centerX, centerY + 20, 40, 0.2 * Math.PI, 0.8 * Math.PI);
        ctx.strokeStyle = '#333';
        ctx.lineWidth = 4;
        ctx.lineCap = 'round';
        ctx.stroke();

        // Clown Body/Shirt (simplified)
        ctx.beginPath();
        ctx.moveTo(centerX - 80, height);
        ctx.lineTo(centerX - 40, centerY + 70);
        ctx.lineTo(centerX + 40, centerY + 70);
        ctx.lineTo(centerX + 80, height);
        ctx.fillStyle = '#FFD700';
        ctx.fill();
        ctx.stroke();
    }

    function animate() {
        ctx.clearRect(0, 0, width, height);
        
        drawBackground();

        confetti.forEach(c => {
            c.update();
            c.draw();
        });

        balls.forEach(ball => {
            ball.update();
            ball.draw();
        });

        drawClown();

        requestAnimationFrame(animate);
    }

    window.addEventListener('resize', resize);
    
    // Initial call
    resize();
    animate();

</script>
</body>
</html>
```
