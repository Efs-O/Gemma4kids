```html
<!DOCTYPE html>
<html lang="en">
<head>
    <meta charset="UTF-8">
    <meta name="viewport" content="width=device-width, initial-scale=1.0">
    <title>Colorful Underwater World</title>
    <style>
        body, html {
            margin: 0;
            padding: 0;
            width: 100%;
            height: 100%;
            overflow: hidden;
            background-color: #001f3f;
        }

        canvas {
            display: block;
        }

        /* Decorative background gradient to simulate deep sea to sunlight */
        .ocean-bg {
            position: absolute;
            top: 0;
            left: 0;
            width: 100%;
            height: 100%;
            background: linear-gradient(to bottom, #4da6ff 0%, #0059b3 50%, #001f3f 100%);
            z-index: -1;
        }

        /* Sunlight rays effect */
        .sunlight {
            position: absolute;
            top: 0;
            left: 0;
            width: 100%;
            height: 100%;
            background: radial-gradient(circle at 50% -20%, rgba(255, 255, 255, 0.3) 0%, transparent 60%);
            pointer-events: none;
        }
    </style>
</head>
<body>

    <div class="ocean-bg"></div>
    <div class="sunlight"></div>
    <canvas id="canvas"></canvas>

    <script>
        const canvas = document.getElementById('canvas');
        const ctx = canvas.getContext('2d');

        let width, height;
        let fishes = [];
        let bubbles = [];
        const fishCount = 15;
        const bubbleCount = 60;
        const colors = ['#FF5733', '#FFC300', '#DAF7A6', '#FF33FF', '#33FF57', '#33FFF5', '#FF3385'];

        function resize() {
            width = canvas.width = window.innerWidth;
            height = canvas.height = window.innerHeight;
        }

        window.addEventListener('resize', resize);
        resize();

        class Fish {
            constructor() {
                this.init();
            }

            init() {
                this.x = Math.random() * width;
                this.y = Math.random() * height;
                this.size = Math.random() * 20 + 15;
                this.speedX = (Math.random() - 0.5) * 3;
                this.speedY = (Math.random() - 0.5) * 1;
                this.color = colors[Math.floor(Math.random() * colors.length)];
                this.wiggle = 0;
                this.wiggleSpeed = Math.random() * 0.1 + 0.05;
                this.direction = this.speedX > 0 ? 1 : -1;
            }

            update() {
                this.x += this.speedX;
                this.y += this.speedY;
                this.wiggle += this.wiggleSpeed;

                // Bounce off walls
                if (this.x > width + 50) this.x = -50;
                if (this.mulX && this.x < -50) this.x = width + 50;
                if (this.x < -50) this.x = width + 50;
                if (this.x > width + 50) this.x = -50;

                // Boundary check for Y
                if (this.y < 0 || this.y > height) this.speedY *= -1;

                // Flip direction
                if (this.speedX > 0) this.direction = 1;
                if (this.speedX < 0) this.direction = -1;

                // Slight vertical drift
                this.y += Math.sin(this.wiggle) * 0.5;
            }

            draw() {
                ctx.save();
                ctx.translate(this.x, this.y);
                ctx.scale(this.direction, 1);

                // Tail
                ctx.beginPath();
                ctx.moveTo(-this.size, 0);
                ctx.lineTo(-this.size - (15 * Math.sin(this.wiggle)), -this.size/2);
                ctx.lineTo(-this.size - (15 * Math.sin(this.wiggle)), this.size/2);
                ctx.fillStyle = this.color;
                ctx.fill();

                // Body (Oval)
                ctx.beginPath();
                ctx.ellipse(0, 0, this.size, this.size/1.5, 0, 0, Math.PI * 2);
                ctx.fillStyle = this.color;
                ctx.fill();

                // Eye
                ctx.beginPath();
                ctx.arc(this.size/2, -this.size/4, this.size/5, 0, Math.PI * 2);
                ctx.fillStyle = 'white';
                ctx.fill();
                ctx.beginPath();
                ctx.arc(this.size/2 + 1, -this.size/4, this.size/10, 0, Math.PI * 2);
                ctx.fillStyle='black';
                ctx.fill();

                // Fin
                ctx.beginPath();
                ctx.ellipse(-this.size/4, -this.size/2, this.size/3, this.size/5, Math.PI/4, 0, Math.PI * 2);
                ctx.fillStyle = 'rgba(255,255,255,0.3)';
                ctx.fill();

                ctx.restore();
            }
        }

        class Bubble {
            constructor() {
                this.init();
            }

            init() {
                this.x = Math.random() * width;
                this.y = height + Math.random() * 100;
                this.radius = Math.random() * 6 + 2;
                this.speedY = Math.random() * 1.5 + 0.5;
                this.speedX = (Math.random() - 0.5) * 1;
                this.opacity = Math.random() * 0.5 + 0.1;
                this.wobble = Math.random() * Math.PI * 2;
            }

            update() {
                this.y -= this.speedY;
                this.x += Math.sin(this.wobble) * 0.5;
                this.wobble += 0.05;

                if (this.y < -20) {
                    this.init();
                }
            }

            draw() {
                ctx.beginPath();
                ctx.arc(this.x, this.y, this.radius, 0, Math.PI * 2);
                ctx.strokeStyle = `rgba(255, 255, 255, ${this.opacity})`;
                ctx.lineWidth = 1;
                ctx.stroke();
                
                // Small highlight inside bubble
                ctx.beginPath();
                ctx.arc(this.x - this.radius/3, this.y - this.radius/3, this.radius/4, 0, Math.PI * 2);
                ctx.fillStyle = `rgba(255, 255, 255, ${this.opacity + 0.2})`;
                ctx.fill();
            }
        }

        function createScene() {
            fishes = [];
            bubbles = [];
            for (let i = 0; i < fishCount; i++) {
                fishes.push(new Fish());
            }
            for (let i = 0; i < bubbleCount; i++) {
                bubbles.push(new Bubble());
            }
        }

        function animate() {
            // Clear canvas with a slight trail effect
            ctx.clearRect(0, 0, width, height);

            // Draw Bubbles
            bubbles.forEach(bubble => {
                bubble.update();
                bubble.draw();
            });

            // Draw Fish
            fishes.forEach(fish => {
                fish.update();
                fish.draw();
            });

            requestAnimationFrame(animate);
        }

        createScene();
        animate();

    </script>
</body>
</html>
```
