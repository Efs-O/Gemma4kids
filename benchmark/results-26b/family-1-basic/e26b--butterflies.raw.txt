    <style>
        :root {
            --sky-top: #87CEEB;

        body, html {
            margin: 0;

        #canvas-container {

        /* Flower Styling */

<script>
    const garden = document.getElementById('garden');
    const butterflyLayer = document.getElementById('butterfly-layer');
    const colors = ['#FF1493', '#00FF7F', '#FFD700', '#FF4500', '#9400D3', '#00BFFF', '#FF69B4'];

    // Generate Flowers
    function createFlowers() {
        const flowerCount = 15;
        for (let i = 0; i < flowerCount; i++) {
            const flower = document.createElement('div');
            flower.className = 'flower';
            
            const xPos = Math.random() * 100;
            const yPos = 85 + (Math.random() * 10);
            const petalColor = colors[Math.floor(Math.random() * colors.length)];
            
            flower.style.left = `${xPos}%`;
            flower.style.top = `${yPos}%`;
            
            // Create petals
            const petalCount = 5 + Math.floor(Math.random() * 3);
            for (let j = 0; j < petalCount; j++) {
                const petal = document.createElement('div');
                petal.className = 'petal';
                petal.style.backgroundColor = petalColor;
                petal.style.transform = `rotate(${j * (360/petalCount)}deg) translateY(-15px)`;
                flower.appendChild(petal);
            }

            const center = document.createElement('div');
            center.className = 'flower-center';
            flower.appendChild(center);

            garden.appendChild(flower);
        }
    }

    // Butterfly Class
    class Butterfly {
        constructor() {
            this.el = document.createElement('div');
            this.el.className = 'butterfly';
            
            this.color = colors[Math.floor(Math.random() * colors.length)];
            this.x = Math.random() * window.innerWidth;
            this.y = Math.random() * window.innerHeight;
            this.vx = (Math.random() - 0.5) * 3;
            this.vy = (Math.random() - 0.5) * 3;
            this.angle = 0;

            this.setupElements();
            butterflyLayer.appendChild(this.el);
        }

        setupElements() {
            const body = document.createElement('div');
            body.className = 'body';
            
            const leftWing = document.createElement('div');
            leftWing.className = 'wing left';
            leftWing.style.backgroundColor = this.color;
            
            const rightWing = document.createElement('div');
            rightWing.className = 'wing right';
            rightWing.style.backgroundColor = this.color;

            this.el.appendChild(leftWing);
            this.el.appendChild(rightWing);
            this.el.appendChild(body);
        }

        update() {
            // Movement physics
            this.x += this.vx;
            this.y += this.vy;

            // Add some "flutter" noise
            this.vx += (Math.random() - 0.5) * 0.5;
            this.vy += (Math.random() - 0.5) * 0.5;

            // Boundary check
            if (this.x < 0 || this.x > window.innerWidth) this.vx *= -1;
            if (this.y < 0 || this.y > window.innerHeight) this.vy *= -1;

            // Keep speed within limits
            const speed = Math.sqrt(this.vx * this.vx + this.vy * this.vy);
            const maxSpeed = 4;
            if (speed > maxSpeed) {
                this.vx = (this.vx / speed) * maxSpeed;
                this.vy = (this.vy / speed) * maxSpeed;
            }

            // Calculate rotation angle based on velocity
            this.angle = Math.atan2(this.vy, this.vx) * (180 / Math.PI);

            // Apply transform
            this.el.style.transform = `translate3d(${this.x}px, ${this.y}px, 0) rotate(${this.angle + 90}deg)`;
        }
    }

    let butterflies = [];

    function init() {
        createFlowers();
        for (let i = 0; i < 12; i++) {
            butterflies.push(new Butterfly());
        }
        animate();
    }

    function animate() {
        butterflies.forEach(b => b.update());
        requestAnimationFrame(animate);
    }

    window.onload = init;

    // Handle resize
    window.addEventListener('resize', () => {
        // Reset butterflies to prevent them from being stuck off-screen
        butterflyLayer.innerHTML = '';
        butterflies = [];
        for (let i = 0; i < 12; i++) {
            butterflies.push(new Butterfly());
        }
    });
</script>

</body>
