<script>
    const canvas = document.getElementById('canvas');
    const ctx = canvas.getContext('2d');

    let width, height;
    let balls = [];
    const gravity = 0.25;
    const friction = 0.98; // Air resistance/Energy loss on bounce
    const colors = ['#FF0055', '#00FFCC', '#FFCC00', '#FF00FF', '#00FF00', '#00FFFF', '#FFFFFF'];

    class Ball {
        constructor(x, y, radius, color, vx, vy) {
            this.x = x;
            this.y = y;
            this.radius = radius;
            this.color = color;
            this.vx = vx;
            this.vy = vy;
        }

        draw() {
            ctx.beginPath();
            ctx.arc(this.x, this.y, this.radius, 0, Math.PI * 2);
            ctx.fillStyle = this.color;
            // Add a glow effect
            ctx.shadowBlur = 15;
            ctx.shadowColor = this.color;
            ctx.fill();
            ctx.closePath();
            ctx.shadowBlur = 0; // Reset shadow for performance
        }

        update() {
            // Apply gravity
            this.vy += gravity;
            
            // Apply movement
            this.x += this.vx;
            this.y += this.vy;

            // Floor collision
            if (this.y + this.radius > height) {
                this.y = height - this.radius;
                this.vy *= -friction;
            }

            // Ceiling collision
            if (this.y - this.radius < 0) {
                this.y = this.radius;
                this.vy *= -friction;
            }

            // Right wall collision
            if (this.x + this.radius > width) {
                this.x = width - this.radius;
                this.vx *= -friction;
            }

            // Left wall collision
            if (this.x - this.radius < 0) {
                this.x = this.radius;
                this.vx *= -friction;
            }

            this.draw();
        }
    }

        resize();
        balls = [];
        const numberOfBalls = 15;
        for (let i = 0; i < numberOf
        for (let i = 0; i < numberOfBalls; i++) {
            const radius = Math.random() * 20 + 15;
            const x = Math.random() * (width - radius * 2) + radius;
            const y = Math.random() * (height - radius * 2) + radius;
            const color = colors[Math.floor(Math.random() * colors.length)];
            const vx = (Math.random() - 0.5) * 15;
            const vy = (Math.random() - 0.5) * 15;
            balls.push(new Ball(x, y, radius, color, vx, vy));
        }
    }

    function resize() {
        width = window.innerWidth;
        height = window.innerHeight;
        canvas.width = width;
        canvas.height = height;
    }

    function animate() {
        // Create a trailing effect by using a translucent rectangle instead of clearRect
        ctx.fillStyle = 'rgba(0, 0, 0, 0.15)';
        ctx.fillRect(0, 0, width, height);

        balls.forEach(ball => ball.update());
        requestAnimationFrame(animate);
    }

    window.addEventListener('resize', () => {
        resize();
        // Re-initialize balls on resize to keep them within bounds
        init();
    });

    // Click to add a ball
    window.addEventListener('mousedown', (e) => {
        const radius = Math.random() * 20 + 15;
        const color = colors[Math.floor(Math.random() * colors.length)];
        const vx = (Math.random() - 0.5) * 20;
        const vy = (Math.random() - 0.5) * 20;
        balls.push(new Ball(e.clientX, e.clientY, radius, color, vx, vy));
    });

    init();
    animate();
</script>
