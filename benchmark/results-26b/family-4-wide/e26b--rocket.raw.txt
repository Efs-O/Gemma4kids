```html
<!DOCTYPE html>
<html lang="en">
<head>
    <meta charset="UTF-8">
    <meta name="viewport" content="width=device-width, initial-scale=1.0">
    <title>Rocket Launch into Space</title>
    <style>
        body, html {
            margin: 0;
            padding: 0;
            width: 100%;
            height: 100%;
            overflow: hidden;
            background-color: #050510; /* Deep space blue */
        }

        #canvas-container {
            position: relative;
            width: 100vw;
            height: 100vh;
            display: flex;
            justify-content: center;
            align-items: center;
        }

        canvas {
            display: block;
        }

        /* The Rocket UI/Overlay */
        #ui-overlay {
            position: absolute;
            top: 20px;
            left: 20px;
            color: #00ffcc;
            font-family: 'Segoe UI', Tahoma, Geneva, Verdana, sans-serif;
            pointer-events: none;
            text-transform: uppercase;
            letter-spacing: 2px;
            text-shadow: 0 0 10px #00ffcc;
        }

        /* Shake animation for the rocket when thrust increases */
        @keyframes rocketShake {
            0% { transform: translate(0, 0); }
            25% { transform: async translate(1px, -1px); }
            50% { transform: translate(-1px, 1px); }
            75% { transform: translate(1px, 1px); }
            100% { transform: translate(0, 0); }
        }

        .launch-active {
            animation: rocketShake 0.1s infinite;
        }
    </style>
</head>
<body>

<div id="canvas-container">
    <div id="ui-overlay">
        System: <span id="status">Ready for Launch</span><br>
        Velocity: <span id="velocity">0</span> km/s
    </div>
    <canvas id="spaceCanvas"></canvas>
</div>

<script>
    const canvas = document.getElementById('spaceCanvas');
    const ctx = canvas.getContext('2d');
    const statusText = document.getElementById('status');
    const velocityText = document.getElementById('velocity');

    let width, height;
    let stars = [];
    let particles = [];
    let rocketY = 0;
    let launchProgress = 0; // 0 to 1
    let isLaunching = false;
    let speedMultiplier = 1;
    let frameCount = 0;

    // Configuration
    const STAR_COUNT = 200;
    const PARTICLE_COUNT = 50;

    class Star {
        constructor() {
            this.reset();
            this.y = Math.random() * height; // Distribute stars vertically
        }

        reset() {
            this.x = Math.random() * width;
            this.y = -10;
            this.size = Math.random() * 2 + 0.5;
            this.speed = (Math.random() * 0.5 + 0.2) * speedMultiplier;
            this.opacity = Math.random() * 0.8 + 0.2;
        }

        update() {
            this.y += this.speed * speedMultiplier;
            if (this.y > height) {
                this.reset();
            }
        }

        draw() {
            ctx.beginPath();
            ctx.arc(this.x, this.y, this.size, 0, Math.PI * 2);
            ctx.fillStyle = `rgba(255, 255, 255, ${this.opacity})`;
            ctx.fill();
        }
    }

    class Particle {
        constructor() {
            this.reset();
        }

        reset() {
            this.x = width / 2 + (Math.random() - 0.5) * 20;
            this.y = height / 2 + 40;
            this.vx = (Math.random() - 0.5) * 2;
            this.vy = Math.random() * 3 + 2;
            this.life = 1.0;
            this.decay = Math.random() * 0.02 + 0.01;
            this.color = Math.random() > 0.5 ? '#ff4500' : '#ffae00';
            this.size = Math.random() * 5 + 2;
        }

        update() {
            this.x += this.vx * speedMultiplier;
            this.y += this.vy * speedMultiplier;
            this.life -= this.decay;
            if (this.life <= 0) {
                this.reset();
            }
        }

        draw() {
            ctx.beginPath();
            ctx.arc(this.x, this.y, this.size, 0, Math.PI * 2);
            ctx.fillStyle = this.color.replace(')', `, ${this.life})`).replace('ff', 'ff'); // Simple alpha trick
            // Correct way to handle alpha in canvas
            ctx.globalAlpha = this.life;
            ctx.fillStyle = this.color;
            ctx.fill();
            ctx.globalAlpha = 1.0;
        }
    }

    function init() {
        width = window.innerWidth;
        height = window.innerHeight;
        canvas.width = width;
        canvas.height = height;

        stars = [];
        for (let i = 0; i < STAR_COUNT; i++) {
            stars.push(new Star());
        }

        particles = [];
        for (let i = 0; i < PARTICLE_COUNT; i++) {
            particles.push(new Particle());
        }

        // Trigger launch after 2 seconds
        setTimeout(() => {
            isLaunching = true;
            statusText.innerText = "Ascending...";
            statusText.style.color = "#ff4500";
        }, 2000);
    }

    function drawRocket(x, y) {
        const shakeX = isLaunching ? (Math.random() - 0.5) * 3 : 0;
        const shakeY = isLaunching ? (Math.random() - 0.5) * 3 : 0;
        
        ctx.save();
        ctx.translate(x + shakeX, y + shakeY);

        // 1. Exhaust Flame (Bottom)
        if (isLaunching) {
            const flameHeight = 60 + Math.random() * 40;
            const gradient = ctx.createLinearGradient(0, 0, 0, flameHeight);
            gradient.addColorStop(0, '#ffae00');
            gradient.addColorStop(0.5, '#ff4500');
            gradient.addColorStop(1, 'transparent');
            
            ctx.beginPath();
            ctx.moveTo(-15, 0);
            ctx.lineTo(15, 0);
            ctx.lineTo(0, flameHeight);
            ctx.fillStyle = gradient;
            ctx.fill();
        }

        // 2. Rocket Fins
        ctx.fillStyle = '#cc0000';
        ctx.beginPath();
        ctx.moveTo(-20, 20);
        ctx.lineTo(-20, -10);
        ctx.lineTo(-5, 0);
        ctx.fill();

        ctx.beginPath();
        ctx.moveTo(20, 20);
        ctx.lineTo(20, -10);
        ctx.lineTo(5, 0);
        ctx.fill();

        // 3. Rocket Body
        ctx.fillStyle = '#e0e0e0';
        ctx.fillRect(-15, -40, 30, 60);

        // 4. Nose Cone
        ctx.beginPath();
        ctx.moveTo(-15, -40);
        ctx.lineTo(15, -40);
        ctx.lineTo(0, -70);
        ctx.fillStyle = '#cc0000';
        ctx.fill();

        // 5. Window
        ctx.beginPath();
        ctx.arc(0, -20, 7, 0, Math.PI * 2);
        ctx.fillStyle = '#333';
        ctx.fill();
        ctx.strokeStyle = '#fff';
        ctx.lineWidth = 2;
        ctx.stroke();
        
        // Window reflection
        ctx.beginPath();
        ctx.arc(-3, -23, 2, 0, Math.PI * 2);
        ctx.fillStyle = 'rgba(255,255,255,0.5)';
        ctx.fill();

        ctx.restore();
    }

    function animate() {
        frameCount++;
        ctx.fillStyle = '#050510';
        ctx.fillRect(0, 0, width, height);

        // Update Speed logic
        if (isLaunching) {
            speedMultiplier += 0.001;
            launchProgress += 0.005;
            velocityText.innerText = (launchProgress * 28000).toFixed(0);
            
            // Add smoke/particles more aggressively during launch
            if (frameCount % 2 === 0) {
                particles.push(new Particle());
                if (particles.length > 100) particles.shift();
            }
        } else {
            velocityText.innerText = "0";
        }

        // Draw Stars
        stars.forEach(star => {
            star.update();
            star.draw();
        });

        // Draw Particles (Smoke/Fire)
        particles.forEach((p, index) => {
            p.update();
            p.draw();
            if (p.life <= 0) particles.splice(index, 1);
        });

        // Draw Rocket in center
        drawRocket(width / 2, height / 2);

        requestAnimationFrame(animate);
    }

    window.addEventListener('resize', () => {
        width = window.innerWidth;
        height = window.innerHeight;
        canvas.width = width;
        canvas.height = height;
    });

    init();
    animate();

</script>
</body>
</html>
```
