```html
<!DOCTYPE html>
<html lang="en">
<head>
    <meta charset="UTF-8">
    <meta name="viewport" content="width=device-width, initial-scale=1.0">
    <title>Autumn Leaves Animation</title>
    <style>
        body, html {
            margin: 0;
            padding: 0;
            width: 100%;
            height: 100%;
            overflow: hidden;
            background: linear-gradient(to bottom, #2c3e50, #e67e22, #d35400);
        }

        #scene {
            position: relative;
            width: 100vw;
            height: 100vh;
            display: flex;
            justify-content: center;
            align-items: center;
        }

        /* The Background Trees */
        .tree-silhouette {
            position: absolute;
            bottom: 0;
            width: 100%;
            height: 60%;
            background: linear-gradient(to top, #2d1b0a, transparent);
            z-index: 1;
        }

        .trunk {
            position: absolute;
            bottom: 0;
            width: 100px;
            height: 70%;
            background-color: #2d1b0a;
            z-index: 2;
            border-radius: 50px 50px 0 0;
        }

        .trunk-left { left: 10%; }
        .trunk-right { right: 15%; }

        /* The Leaves */
        .leaf {
            position: absolute;
            top: -50px;
            border-radius: 2px 10px 2px 10px;
            opacity: 0.8;
            z-index: 5;
            pointer-events: none;
        }

        /* Animation for falling and swaying */
        @keyframes fallAndSway {
            0% {
                transform: translateY(0) translateX(0) rotate(0deg);
                opacity: 0;
            }
            10% {
                opacity: 1;
            }
            90% {
                opacity: 1;
            }
            100% {
                transform: translateY(110vh) translateX(150px) rotate(720deg);
                opacity: 0;
            }
        }

        @keyframes sway {
            0%, 100% { transform: translateX(0px); }
            50% { transform: translateX(50px); }
        }

        /* Floating dust particles for atmosphere */
        .particle {
            position: absolute;
            background-color: rgba(255, 255, 255, 0.2);
            border-radius: 50%;
            z-index: 3;
            pointer-events: none;
        }

        @keyframes drift {
            0% { transform: translateY(0) translateX(0); }
            50% { transform: translateY(-50vh) translateX(30px); }
            100% { transform: translateY(-100vh) translateX(-30px); }
        }
    </style>
</head>
<body>

    <div id="scene">
        <div class="tree-silhouette"></div>
        <div class="trunk trunk-left"></div>
        <div class="trunk trunk-right"></div>
        <div id="leaf-container"></div>
        <div id="particle-container"></div>
    </div>

    <script>
        const leafContainer = document.getElementById('leaf-container');
        const particleContainer = document.getElementById('particle-container');
        const colors = ['#e74c3c', '#e67e22', '#f1c40f', '#d35400', '#c0392b', '#f39c12', '#7e5109'];
        
        const width = window.innerWidth;
        const height = window.innerHeight;

        /**
         * Creates a leaf element with randomized properties
         */
        function createLeaf() {
            const leaf = document.createElement('div');
            leaf.className = 'leaf';
            
            // Randomize appearance
            const size = Math.random() * 20 + 10;
            const color = colors[Math.floor(Math.random() * colors.length)];
            const startX = Math.random() * width;
            const duration = Math.random() * 8 + 7; // 7s to 15s
            const delay = Math.random() * -20; // Negative delay to start mid-animation
            const zIndex = Math.floor(Math.random() * 10);
            const scale = Math.random() * 0.5 + 0.5;

            // Apply styles
            leaf.style.width = `${size}px`;
            leaf.style.height = `${size * 0.7}px`;
            leaf.style.backgroundColor = color;
            leaf.style.left = `${startX}px`;
            leaf.style.zIndex = zIndex;
            leaf.style.animation = `fallAndSway ${duration}s linear ${delay}s infinite`;
            leaf.style.opacity = Math.random() * 0.5 + 0.5;
            
            // Layered depth: larger leaves feel closer (higher z-index, larger scale)
            if (zIndex > 7) {
                leaf.style.transform = `scale(${scale * 1.5})`;
                leaf.style.filter = 'blur(0.5px)';
            } else {
                leaf.style.transform = `scale(${scale * 0.7})`;
                leaf.style.filter = 'blur(1px)';
            }

            leafContainer.appendChild(leaf);
        }

        /**
         * Creates floating dust particles for depth
         */
        function createParticle() {
            const particle = document.createElement('div');
            particle.className = 'particle';
            
            const size = Math.random() * 4 + 2;
            const startX = Math.random() * width;
            const startY = Math.random() * height;
            const duration = Math.random() * 10 + 10;
            const delay = Math.random() * -10;

            particle.style.width = `${size}px`;
            particle.style.height = `${size}px`;
            particle.style.left = `${startX}px`;
            particle.style.top = `${startY}px`;
            particle.style.animation = `drift ${duration}s ease-in-out ${delay}s infinite`;

            particleContainer.appendChild(particle);
        }

        // Initialize the scene
        function init() {
            // Clear containers in case of resize
            leafContainer.innerHTML = '';
            particleContainer.innerHTML = '';

            // Create a high number of leaves for a full effect
            for (let i = 0; i < 80; i++) {
                createLeaf();
            }

            // Create particles for atmospheric depth
            for (let i = 0; i < 40; i++) {
                createParticle();
            }
        }

        // Handle window resize to maintain full viewport coverage
        window.addEventListener('resize', () => {
            init();
        });

        // Run on load
        init();

    </script>
</body>
</html>
```
