```html
<!DOCTYPE html>
<html lang="en">
<head>
    <meta charset="UTF-8">
    <meta name="viewport" content="width=device-width, initial-scale=1.0">
    <title>Fun Clown Juggling Animation</title>
    <style>
        body {
            margin: 0;
            padding: 0;
            overflow: hidden;
            background: radial-gradient(circle, #2e004f 0%, #000000 100%);
            display: flex;
            justify-content: center;
            align-items: center;
            height: 100vh;
        }

        #canvasContainer {
            position: relative;
            width: 100vw;
            height: 100vh;
        }

        canvas {
            display: block;
        }

        /* The Clown "Presence" - A stylized UI element for depth */
        #ui-overlay {
            position: absolute;
            top: 20px;
            left: 20px;
            pointer-events: none;
            font-family: 'Arial Black', sans-serif;
            color: #ffeb3b;
            font-size: 2rem;
            text-shadow: 3px 3px 0px #ff4081, 5px 5px 0px #000;
            animation: floatText 3s ease-in-out infinite;
            z-index: 10;
        }

        @keyframes floatText {
            0%, 100% { transform: translateY(0) rotate(-2deg); }
            50% { transform: translateY(-20px) rotate(2deg); }
        }

        /* Background Sparkles */
        .sparkle {
            position: absolute;
            width: 4px;
            height: 4px;
            background-color: white;
            border-radius: 50%;
            opacity: 0.5;
            animation: twinkle 2s infinite;
        }

        @keyframes twinkle {
            0%, 100% { opacity: 0.2; transform: scale(1); }
            50% { opacity: 1; transform: scale(1.5); }
        }
    </style>
</head>
<body>

    <div id="canvasContainer">
        <div id="ui-overlay">CIRCUS PARTY!</div>
        <canvas id="mainCanvas"></canvas>
    </div>

    <script>
        const canvas = document.getElementById('mainCanvas');
        const ctx = canvas.getContext('2d');
        let width, height;

        // Configuration
        const ballCount = 6;
        const confettiCount = 50;
        const balls = [];
        const confettiPieces = [];
        const colors = ['#FF0000', '#FFEB3B', '#00FF00', '#00BCD4', '#E91E63', '#FF9800'];

        class Ball {
            constructor(index) {
                this.radius = 25 + Math.random() * 10;
                this.reset(index);
            }

            reset(index) {
                // Give each ball a unique trajectory path (elliptical juggling)
                this.x = width / 2;
                this.y = height / 2;
                this.vx = Math.cos(index * (Math.PI * 2 / ballCount)) * 5;
                this.vy = -Math.sin(index * (Math.PI * 2 / ballCount)) * 8;
                this.color = colors[index % colors.length];
                this.gravity = 0.15;
                this.bounce = -0.95;
                this.trail = [];
            }

            update() {
                // Add trail
                this.trail.push({ x: this.x, y: this.y });
                if (this.trail.length > 10) this.trail.shift();

                this.vy += this.gravity;
                this.x += this.vx;
                this.y += this.vy;

                // Bounce off sides
                if (this.x + this.radius > width || this.x - this.radius < 0) {
                    this.vx *= -1;
                }

                // Bounce off bottom (Floor)
                if (this.y + this.radius > height) {
                    this.y = height - this.radius;
                    this.vy *= this.bounce;
                }

                // Bounce off top (Ceiling)
                if (this.y - this.radius < 0) {
                    this.y = this.radius;
                    this.vy *= this.bounce;
                }
            }

            draw() {
                // Draw trail
                ctx.beginPath();
                ctx.strokeStyle = this.color;
                ctx.lineWidth = 2;
                ctx.globalAlpha = 0.3;
                for (let i = 0; i < this.trail.length; i++) {
                    ctx.lineTo(this.trail[i].x, this.trail[i].y);
                }
                ctx.stroke();
                ctx.globalAlpha = 1.0;

                // Draw Ball
                ctx.beginPath();
                ctx.arc(this.x, this.y, this.radius, 0, Math.PI * 2);
                ctx.fillStyle = this.color;
                ctx.shadowBlur = 15;
                ctx.shadowColor = this.color;
                ctx.fill();
                
                // Shine effect
                ctx.beginPath();
                ctx.arc(this.x - this.radius * 0.3, this.y - this.radius * 0.3, this.radius * 0.2, 0, Math.PI * 2);
                ctx.fillStyle = 'rgba(255, 255, 255, 0.6)';
                ctx.fill();
                ctx.shadowBlur = 0;
            }
        }

        class Confetti {
            constructor() {
                this.x = Math.random() * width;
                this.y = Math.random() * height;
                this.size = Math.random() * 8 + 4;
                this.color = colors[Math.floor(Math.random() * colors.length)];
                this.speedX = (Math.random() - 0.5) * 2;
                this.speedY = Math.random() * 2 + 1;
                this.rotation = Math.random() * 360;
                this.rotationSpeed = (Math.random() - 0.5) * 5;
            }

            update() {
                this.y += this.speedY;
                this.x += this.speedX;
                this.rotation += this.rotationSpeed;

                if (this.y > height) {
                    this.y = -20;
                    this.x = Math.random() * width;
                }
            }

            draw() {
                ctx.save();
                ctx.translate(this.x, this.y);
                ctx.rotate(this.rotation * Math.PI / 180);
                ctx.fillStyle = this.color;
                ctx.globalAlpha = 0.7;
                ctx.fillRect(-this.size/2, -this.size/2, this.size, this.size);
                ctx.restore();
            }
        }

        function init() {
            width = window.innerWidth;
            height = window.innerHeight;
            canvas.width = width;
            canvas.height = height;

            balls.length = 0;
            for (let i = 0; i < ballCount; i++) {
                balls.push(new Ball(i));
            }

            confettiPieces.length = 0;
            for (let i = 0; i < confettiCount; i++) {
                confettiPieces.push(new Confetti());
            }
        }

        function drawClownFace() {
            const centerX = width / 2;
            const centerY = height / 2 + 50;

            // Face Base
            ctx.beginPath();
            ctx.arc(centerX, centerY, 80, 0, Math.PI * 2);
            ctx.fillStyle = "white";
            ctx.shadowBlur = 20;
            ctx.shadowColor = "rgba(255,255,255,0.5)";
            ctx.fill();
            ctx.shadowBlur = 0;

            // Eyes
            ctx.fillStyle = "black";
            ctx.beginPath();
            ctx.arc(centerX - 30, centerY - 10, 8, 0, Math.PI * 2);
            ctx.arc(centerX + 30, centerY - 10, 8, 0, Math.PI * 2);
            ctx.fill();

            // Red Nose
            ctx.beginPath();
            ctx.arc(centerX, centerY + 10, 20, 0, Math.PI * 2);
            ctx.fillStyle = "#ff0000";
            ctx.fill();

            // Smile
            ctx.beginPath();
            ctx.arc(centerX, centerY + 20, 40, 0.2 * Math.PI, 0.8 * Math.PI);
            ctx.lineWidth = 5;
            ctx.strokeStyle = "#ff4081";
            ctx.lineCap = "round";
            ctx.stroke();

            // Clown Hair (Colorful arcs)
            for (let i = 0; i < 8; i++) {
                ctx.beginPath();
                ctx.lineWidth = 15;
                ctx.strokeStyle = colors[i % colors.length];
                ctx.arc(centerX, centerY - 20, 100 + (i * 5), Math.PI, 0);
                ctx.stroke();
            }
        }

        function animate() {
            // Clear background with a slight fade for motion blur
            ctx.fillStyle = 'rgba(20, 0, 40, 0.3)';
            ctx.fillRect(0, 0, width, height);

            // 1. Draw Background Layer (Confetti)
            confettiPieces.forEach(p => {
                p.update();
                p.draw();
            });

            // 2. Draw Midground Layer (Clown)
            drawClownFace();

            // 3. Draw Foreground Layer (Juggling Balls)
            balls.forEach(ball => {
                ball.update();
                ball.draw();
            });

            requestAnimationFrame(animate);
        }

        // Event Listeners
        window.addEventListener('resize', () => {
            init();
        });

        // Start
        init();
        animate();

        // Create some decorative floating sparkles in the DOM
        for (let i = 0; i < 20; i++) {
            const sparkle = document.createElement('div');
            sparkle.className = 'sparkle';
            sparkle.style.left = Math.random() * 100 + 'vw';
            sparkle.style.top = Math.random() * 100 + 'vh';
            sparkle.style.animationDelay = Math.random() * 2 + 's';
            document.body.appendChild(sparkle);
        }

    </script>
</body>
</html>
```
