```html
<!DOCTYPE html>
<html lang="en">
<head>
    <meta charset="UTF-8">
    <meta name="viewport" content="width=device-width, initial-scale=1.0">
    <title>Vibrant Deep Sea Animation</title>
    <style>
        body, html {
            margin: 0;
            padding: 0;
            width: 100%;
            height: 100%;
            overflow: hidden;
            background: linear-gradient(to bottom, #001f3f, #0074D9, #001f3f);
        }

        #ocean {
            position: relative;
            width: 100vw;
            height: 100vh;
            overflow: hidden;
        }

        /* Light rays effect */
        .light-ray {
            position: absolute;
            top: -50%;
            left: -50%;
            width: 200%;
            height: 200%;
            background: radial-gradient(circle, rgba(255, 255, 255, 0.1) 0%, rgba(255, 255, 255, 0) 70%);
            pointer-events: none;
            animation: rotateRay 20s linear infinite;
        }

        @keyframes rotateRay {
            from { transform: rotate(0deg); }
            to { transform: rotate(360deg); }
        }

        /* Fish Styling */
        .fish {
            position: absolute;
            border-radius: 50% 50% 50% 50% / 60% 60% 40% 40%;
            z-index: 10;
            transition: transform 0.5s ease;
        }

        /* The fish tail */
        .fish::after {
            content: '';
            position: absolute;
            right: -15px;
            top: 25%;
            width: 20px;
            height: 50%;
            background: inherit;
            clip-path: polygon(0 0, 100% 50%, 0 100%);
        }

        /* Bubble Styling */
        .bubble {
            position: absolute;
            background: rgba(255, 255, 255, 0.3);
            border: 1px solid rgba(255, 255, 255, 0.5);
            border-radius: 50%;
            pointer-events: none;
            z-index: 5;
        }

        /* Animations */
        @keyframes swimRight {
            0% { transform: translateX(-200px) translateY(0px); }
            25% { transform: translateX(25vw) translateY(20px); }
            50% { transform: translateX(50vw) translateY(-20px); }
            75% { transform: translateX(75vw) translateY(20px); }
            100% { transform: translateX(110vw) translateY(0px); }
        }

        @keyframes swimLeft {
            100% { transform: translateX(110vw) translateY(0px); }
            75% { transform: translateX(75vw) translateY(-20px); }
            50% { transform: translateX(50vw) translateY(20px); }
            25% { transform: translateX(25vw) translateY(-20px); }
            0% { transform: translateX(-200px) translateY(0px); }
        }

        @keyframes rise {
            0% { transform: translateY(110vh) scale(0.5); opacity: 0; }
            20% { opacity: 0.8; }
            80% { opacity: 0.8; }
            100% { transform: translateY(-10vh) scale(1.2); opacity: 0; }
        }

    </string>
</head>
<body>

<div id="ocean">
    <div class="light-ray"></div>
    <!-- Elements generated by JavaScript -->
</div>

<script>
    const ocean = document.getElementById('ocean');
    const colors = ['#FF4136', '#FF851B', '#FFDC00', '#2ECC40', '#0074D9', '#B10DC9', '#FF00FF', '#7FDBFF'];

    function createFish(isForeground)
    {
        const fish = document.createElement('div');
        fish.className = 'fish';
        
        const size = isForeground ? (Math.random() * 30 + 40) : (Math.random() * 50 + 80);
        const color = colors[Math.floor(Math.random() * colors.length)];
        const duration = isForeground ? (Math.random() * 5 + 5) : (Math.random() * 15 + 20);
        const delay = Math.random() * -20; // Negative delay to start animation mid-way
        const direction = Math.random() > 0.5 ? 'swimRight' : 'swimLeft';
        const topPos = Math.random() * 100;
        const opacity = isForeground ? 1 : 0.6;
        const zIndex = isForeground ? 20 : 10;

        fish.style.width = size + 'px';
        fish.style.height = (size * 0.6) + 'px';
        fish.style.backgroundColor = color;
        fish.style.top = topPos + '%';
        fish.style.opacity = opacity;
        fish.style.zIndex = zIndex;
        fish.style.animation = `${direction} ${duration}s linear ${delay}s infinite`;
        
        // Add eye
        const eye = document.createElement('div');
        eye.style.position = 'absolute';
        eye.style.width = (size * 0.15) + 'px';
        eye.style.height = (size * 0.15) + 'px';
        eye.style.backgroundColor = 'white';
        eye.style.borderRadius = '50%';
        eye.style.top = '20%';
        eye.style.left = '20%';
        fish.appendChild(eye);

        const pupil = document.createElement('div');
        pupil.style.position = 'absolute';
        pupil.style.width = (size * 0.07) + 'px';
        pupil.style.height = (size * 0.07) + 'px';
        pupil.style.backgroundColor = 'black';
        pupil.style.borderRadius = '50%';
        pupil.style.top = '20%';
        pupil.style.left = '20%';
        eye.appendChild(pupil);

        ocean.appendChild(fish);
    }

    function createBubble()
    {
        const bubble = document.createElement('div');
        bubble.className = 'bubble';
        
        const size = Math.random() * 20 + 5;
        const leftPos = Math.random() * 100;
        const duration = Math.random() * 6 + 4;
        const delay = Math.random() * 5;

        bubble.style.width = size + 'px';
        bubble.style.height = size + 'px';
        bubble.style.left = leftPos + 'vw';
        bubble.style.bottom = '-50px';
        bubble.style.animation = `rise ${duration}s linear ${delay}s infinite`;

        ocean.appendChild(bubble);

        // Remove bubble from DOM after animation to keep performance high
        // Note: In a simple infinite CSS animation, they persist, 
        // but we can cleanup if we were generating them via setInterval.
        // For this static loop, we let CSS handle it.
    }

    function init()
    {
        // Set viewport dimensions
        const width = window.innerWidth;
        const height = window.innerHeight;

        // Create Background Fish (Slow, Large)
        for (let i = 0; i < 8; i++) {
            createFish(false);
        }

        // Create Midground Fish (Medium)
        for (let i = 0; i < 6; i++) {
            createFish(false);
        }

        // Create Foreground Fish (Fast, Small)
        for (let i = 0; i < 4; i++) {
            createFish(true);
        }

        // Create Bubbles
        for (let i = 0; i < 30; i++) {
            createBubble();
        }
    }

    // Handle window resize
    window.addEventListener('resize', () => {
        ocean.innerHTML = '<div class="light-ray"></div>';
        init();
    });

    init();
</script>

</body>
</html>
```
