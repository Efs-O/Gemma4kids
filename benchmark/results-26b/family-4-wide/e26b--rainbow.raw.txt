<script>
    const canvas = document.getElementById('rainCanvas');
    const ctx = canvas.getContext('2d');
    const cloudsLayer = document.createElement('div');
    document.getElementById('clouds-layer').appendChild(cloudsLayer);

    let width, height;
    let raindrops = [];
    let frame = 0;

    // Resize handling
    function resize() {
        width = window.innerWidth;
        height = window.innerHeight;
        canvas.width = width;
        canvas.height = height;
    }

    window.addEventListener('resize', resize);
    resize();

    // Raindrop Class
    class Raindrop {
        constructor() {
            this.reset();
        }

        reset() {
            this.x = Math.random() * width;
            this.y = Math.random() * -height;
            this.velocity = Math.random() * 10 + 10;
            this.length = Math.random() * 20 + 10;
            this.opacity = Math.random() * 0.5 + 0.2;
        }

        update(intensity) {
            this.y += this.velocity;
            if (this.y > height) {
                this.reset();
            }
        }

        draw(intensity) {
            ctx.beginPath();
            ctx.strokeStyle = `rgba(174, 194, 224, ${this.opacity * intensity})`;
            ctx.lineWidth = 2;
            ctx.moveTo(this.x, this.y);
            ctx.lineTo(this.x, this.y + this.length);
            ctx.stroke();
        }
    }

    // Initialize raindrops
    for (let i = 0; i < 300; i++) {
        raindrops.push(new Raindrop());
    }

    // Create Clouds
    function createClouds() {
        for (let i = 0; i < 8; i++) {
            const cloud = document.createElement('div');
            cloud.className = 'cloud';
            const size = Math.random() * 100 + 100;
            cloud.style.width = size + 'px';
            cloud.style.height = (size * 0.6) + 'px';
            cloud.style.top = Math.random() * 40 + '%';
            cloud.style.left = '-200px';
            
            // Randomize animation
            const duration = Math.random() * 20 + 20;
            const delay = Math.random() * -20;
            cloud.style.animationDuration = duration + 's';
            cloud.style.animationDelay = delay + 's';
            
            // Color variability
            const brightness = Math.random() * 30;
            cloud.style.filter = `blur(15px) brightness(${100 + brightness}%)`;
            
            cloudsLayer.appendChild(cloud);
        }
    }

    createClouds();

    // Animation Loop
    function animate() {
        ctx.clearRect(0, 0, width, height);
        
        frame++;

        // Calculate intensity: starts at 1, goes to 0 over 15 seconds
        // 15 seconds = roughly 900 frames at 60fps
        const intensity = Math.max(0, 1 - (frame / 900));

        // Update and draw raindrops
        raindrops.forEach(drop => {
            drop.update(intensity);
            drop.draw(intensity);
        });

        // Gradually thin out raindrops by removing them from the loop if needed
        // but for simplicity we just use intensity in the draw call
        
        requestAnimationFrame(animate);
    }

    animate();
</script>
