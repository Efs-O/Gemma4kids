```html
<!DOCTYPE html>
<html lang="en">
<head>
    <meta charset="UTF-8">
    <meta name="viewport" content="width=device-width, initial-scale$1.0">
    <title>Vivid Butterfly Garden</title>
    <style>
        body, html {
            margin: 0;
            padding: 0;
            width: 100%;
            height: 100%;
            overflow: hidden;
            background: linear-gradient(to bottom, #87CEEB, #E0F7FA, #90EE90);
        }

        #garden-container {
            position: relative;
            width: 100vw;
            height: 100vh;
        }

        #canvas-layer {
            position: absolute;
            top: 0;
            left: 0;
            z-index: 10;
        }

        .meadow {
            position: absolute;
            bottom: 0;
            width: 100%;
            height: 30%;
            background: linear-gradient(to top, #228B22, #32CD32);
            z-index: 1;
        }

        .flower {
            position: absolute;
            bottom: 20px;
            transform-origin: bottom center;
            animation: sway 4s ease-in-out infinite alternate;
            z-index: 5;
        }

        .flower-center {
            width: 20px;
            height: 20px;
            background-color: #FFD700;
            border-radius: 50%;
            position: relative;
            z-index: 2;
        }

        .petal {
            position: absolute;
            width: 15px;
            height: 25px;
            border-radius: 50%;
            top: -5px;
            left: 2.5px;
            z-index: 1;
        }

        @keyframes sway {
            from { transform: rotate(-5deg); }
            to { transform: rotate(5deg); }
        }

        /* Creating petal layers */
        .petal-1 { transform: rotate(0deg) translateY(-15px); }
        .petal-2 { transform: rotate(72deg) translateY(-15px); }
        .petal-3 { transform: rotate(144deg) translateY(-15px); }
        .petal-4 { transform: rotate(216deg) translateY(-15px); }
        .petal-5 { transform: rotate(288deg) translateY(-15px); }

    </style>
</head>
<body>

    <div id="garden-container">
        <div id="sky-layer"></div>
        <div class="meadow" id="meadow-layer"></div>
        <canvas id="canvas-layer"></canvas>
    </div>

    <script>
        const canvas = document.getElementById('canvas-layer');
        const ctx = canvas.getContext('2d');
        const meadowLayer = document.getElementById('meadow-layer');

        let width, height;
        let butterflies = [];
        const butterflyCount = 15;
        const flowerCount = 25;
        const colors = ['#FF1493', '#00BFFF', '#FF8C00', '#ADFF2F', '#FF00FF', '#00FF7F', '#FFFF00'];

        function resize() {
            width = window.innerWidth;
            height = window.innerHeight;
            canvas.width = width;
            canvas.height = height;
        }

        window.addEventListener('resize', resize);
        resize();

        // Create Flowers
        function createFlowers() {
            for (let i = 0; i < flowerCount; i++) {
                const flower = document.createElement('div');
                flower.className = 'flower';
                
                const color = colors[Math.floor(Math.random() * colors.length)];
                const posX = Math.random() * width;
                const posY = Math.random() * 40; // Random height from bottom
                const scale = 0.5 + Math.random() * 1.5;
                const delay = Math.random() * -5; // Randomize animation start

                flower.style.left = `${posX}px`;
                flower.style.bottom = `${posY}px`;
                flower.style.transform = `scale(${scale})`;
                flower.style.animationDelay = `${delay}s`;

                const center = document.createElement('div');
                center.className = 'flower-center';
                center.style.backgroundColor = '#FFD700';

                flower.appendChild(center);

                for (let j = 0; j < 5; j++) {
                    const petal = document.createElement('div');
                    petal.className = `petal petal-${j + 1}`;
                    petal.style.backgroundColor = color;
                    flower.appendChild(petal);
                }

                meadowLayer.appendChild(flower);
            }
        }

        class Butterfly {
            constructor() {
                this.init();
            }

            init() {
                this.x = Math.random() * width;
                this.y = Math.random() * height;
                this.size = 10 + Math.random() * 15;
                this.color = colors[Math.floor(Math.random() * colors.length)];
                this.vx = (Math.random() - 0.5) * 3;
                this.vy = (Math.random() - 0.5) * 3;
                this.wingFlap = 0;
                this.flapSpeed = 0.1 + Math.random() * 0.2;
                this.angle = Math.random() * Math.PI * 2;
            }

            update() {
                this.x += this.vx;
                this.y += this.vy;
                this.wingFlap += this.flapSpeed;

                // Bounce off walls
                if (this.x < 0 || this.x > width) this.vx *= -1;
                if (this.y < 0 || this.y > height) this.vy *= -1;

                // Subtle wandering
                this.vx += (Math.random() - 0.5) * 0.2;
                this.vy += (Math.random() - 0.5) * 0.2;

                // Limit speed
                const maxSpeed = 3;
                const currentSpeed = Math.sqrt(this.vx * this.vx + this.vy * this.vy);
                if (currentSpeed > maxSpeed) {
                    this.vx = (this.vx / currentSpeed) * maxSpeed;
                    this.vy = (this.vy / currentSpeed) * maxSpeed;
                }

                // Calculate rotation angle based on velocity
                this.angle = Math.atan2(this.vy, this.vx);
            }

            draw() {
                ctx.save();
                ctx.translate(this.x, this.y);
                ctx.rotate(this.angle + Math.PI/2);

                const flapWidth = Math.sin(this.wingFlap) * 1; // 1 is the scale factor for flap

                // Draw Wings
                ctx.fillStyle = this.color;
                ctx.globalAlpha = 0.8;

                // Left Wing
                ctx.beginPath();
                ctx.ellipse(-this.size * flapWidth, 0, this.size, this.size * 1.5, 0, 0, Math.PI * 2);
                ctx.fill();

                // Right Wing
                ctx.beginPath();
                ctx.ellipse(this.size * flapWidth, 0, this.size, this.size * 1.5, 0, 0, Math.PI * 2);
                ctx.fill();

                // Body
                ctx.fillStyle = '#333';
                ctx.beginPath();
                ctx.ellipse(0, 0, this.size * 0.2, this.size * 1.2, 0, 0, Math.PI * 2);
                ctx.fill();

                ctx.restore();
            }
        }

        function initButterflies() {
            for (let i = 0; i < butterflyCount; i++) {
                butterflies.push(new Butterfly());
            }
        }

        function animate() {
            ctx.clearRect(0, 0, width, height);

            butterflies.forEach(bf => {
                bf.update();
                bf.draw();
            });

            requestAnimationFrame(animate);
        }

        // Start everything
        createFlowers();
        initButterflies();
        animate();

    </script>
</body>
</html>
```
