<body>

    <canvas id="spaceCanvas"></canvas>

    <div id="rocket-container">
        <div class="nose-clone"></div> <!-- Hidden helper -->
        <div class="nose-cone"></div>
        <div class="rocket-body"></div>
        <div class="fin fin-left"></div>
        <div class="fin fin-right"></div>
        <div class="window"></div>
        <div class="flame"></div>
    </div>

    <script>
        const canvas = document.getElementById('spaceCanvas');
        const ctx = canvas.getContext('2d');

        let width, height;
        let stars = [];
        let shootingStars = [];
        let particles = [];

        class Star {
            constructor() {
                this.reset();
            }

            reset() {
                this.x = Math.random() * width;
                this.y = Math.random() * height;
                this.size = Math.random() * 2;
                this.opacity = Math.random();
                this.speed = Math.random() * 0.05;
                this.twinkleSpeed = Math.random() * 0.02;
            }

            update() {
                this.opacity += this.twinkleSpeed;
                if (this.opacity > 1 || this.opacity < 0) {
                    this.twinkleSpeed = -this.twinkle Rex(0.02);
                }
                // To keep opacity within bounds without complex logic:
                this.opacity = 0.3 + Math.abs(Math.sin(Date.now() * this.speed * 0.1));
            }

            draw() {
                ctx.beginPath();
                ctx.arc(this.x, this.y, this.size, 0, Math.PI * 2);
                ctx.fillStyle = `rgba(255, 255, 255, ${this.opacity})`;
                ctx.fill();
            }
        }

        class ShootingStar {
            constructor() {
                this.reset();
            }

            reset() {
                this.x = Math.random() * width;
                this.y = Math.random() * (height / 2);
                this.vx = Math.random() * 10 + 5;
                this.vy = Math.random() * 5 + 2;
                this.life = 100;
                this.active = false;
            }

            launch() {
                this.active = true;
                this.x = Math.random() * width;
                this.y = 0;
                this.life = 100;
            }

            update() {
                if (!this.active) return;
                this.x += this.vx;
                this.y += this.vy;
                this.life -= 2;
                if (this.life <= 0) this.active = false;
            }

            draw() {
                if (!this.active) return;
                ctx.beginPath();
                ctx.strokeStyle = `rgba(255, 255, 255, ${this.life / 100})`;
                ctx.lineWidth = 2;
                ctx.moveTo(this.x, this.runY(this.y));
                ctx.lineTo(this.x - this.vx * 2, this.y - this.vy * 2);
                ctx.stroke();
            }

            runY(val) { return val; } // helper
        }

        class ExhaustParticle {
            constructor(x, y) {
                this.x = x;
                this.y = y;
                this.vx = (Math.random() - 0.5) * 2;
                this.vy = Math.random() * 3 + 2;
                this.size = Math.random() * 8 + 2;
                this.life = 1.0;
                this.color = Math.random() > 0.5 ? '#FFD700' : '#FF4500';
            }

            update() {
                this.x += this.vx;
                this.y += this.vy;
                this.life -= 0.02;
                this.size *= 0.98;
            }

            draw() {
                ctx.beginPath();
                ctx.arc(this.x, this.y, this.size, 0, Math.PI * 2);
                ctx.fillStyle = this.color.replace(')', `, ${this.life})`).replace('rgb', 'rgba'); 
                // Simple color trick for opacity
                ctx.fillStyle = `rgba(255, 69, 0, ${this.life})`;
                ctx.fill();
            }
        }

        function init() {
            width = window.innerWidth;
            height = window.innerHeight;
            canvas.width = width;
            canvas.height = height;

            stars = [];
            for (let i = 0; i < 150; i++) {
                stars.push(new Star());
            }

            shootingStars = [];
            for (let i = 0; i < 3; i++) {
                let s = new ShootingStar();
                s.active = false;
                shootingStars.push(s);
            }
        }

        function animate() {
            ctx.clearRect(0, 0, width, height);

            // Update and draw stars
            stars.forEach(star => {
                star.update();
                star.draw();
            });

            // Update and draw shooting stars
            if (Math.random() < 0.01) {
                let s = shootingStars[Math.floor(Math.random() * shootingStars.length)];
                s.launch();
            }

            shootingStars.forEach(ss => {
                ss.update();
                ss.draw();
            });

            // Handle Particles (Exhaust)
            // We find the rocket position roughly via the animation loop
            // Since the rocket is CSS, we can't easily get its exact pixel pos without JS,
            // but we can simulate it based on the CSS animation timing.
            const time = Date.now() / 8000; 
            const progress = (time % 1); // 0 to 1
            const rocketY = (1.5 * height) - (progress * (2 * height));
            const rocketX = width / 2;

            // Spawn particles at the rocket's base
            // The rocket container is 120px high, starts at 150vh
            // We approximate the Y position based on the 8s loop
            const currentY = (height * 1.5) - (progress * (height * 2.5));
            
            for (let i = 0; i < 5; i++) {
                particles.push(new ExhaustParticle(width / 2, currentY + 80));
            }

            for (let i = particles.length - 1; i >= 0; i--) {
                particles[i].update();
                particles[i].draw();
                if (particles[i].life <= 0) {
                    particles.splice(i, 1);
                }
            }

            requestAnimationFrame(animate);
        }

        window.addEventListener('resize', init);

        init();
        animate();

    </script>
</body>
