```html
<!DOCTYPE html>
<html lang="en">
<head>
    <meta charset="UTF-8">
    <meta name="viewport" content="width=device-width, initial-scale=1.0">
    <title>Magic Carousel</title>
    <style>
        body {
            margin: 0;
            padding: 0;
            overflow: hidden;
            background: radial-gradient(circle, #ff0080, #6600cc, #000000);
            display: flex;
            justify-content: center;
            align-items: center;
            height: 100vh;
        }

        /* The carousel container rotates */
        .carousel-platform {
            position: relative;
            width: 600px;
            height: 400px;
            transform-style: preserve-3d;
            animation: rotateCarousel 15s linear infinite;
        }

        @keyframes rotateCarousel {
            from { transform: rotateY(0deg); }
            to { transform: rotateY(360deg); }
        }

        /* The floor of the carousel */
        .carousel-base {
            position: absolute;
            top: 50%;
            left: 50%;
            width: 500px;
            height: 500px;
            background: repeating-conic-gradient(
                #ff00ff 0deg 30deg,
                #00ffff 30deg 60deg,
                #ffff00 60deg 90deg,
                #00ff00 90deg 120deg
            );
            transform: translate(-50%, -50%) rotateX(80deg);
            border-radius: 50%;
            border: 10px solid #ffffff;
            box-shadow: 0 0 50px #ff00ff;
        }

        /* The central pole */
        .pole {
            position: absolute;
            top: 50%;
            left: 50%;
            width: 30px;
            height: 500px;
            background: linear-gradient(to right, #ffd700, #ffffff, #ffd700);
            transform: translate(-50%, -50%);
            z-index: -1;
        }

        /* Individual Horse Container */
        .horse-container {
            position: absolute;
            top: 50%;
            left: 50%;
            transform-style: preserve-3d;
        }

        /* The Horse Shape */
        .horse {
            position: relative;
            width: 60px;
            height: 80px;
            transform-style: preserve-3d;
        }

        /* Horse Body Parts */
        .horse-body {
            position: absolute;
            width: 100%;
            height: 40px;
            background: #ff4500;
            border-radius: 20px;
            box-shadow: 0 0 15px #ff4500;
        }

        .horse-neck {
            position: absolute;
            top: -30px;
            left: 10px;
            width: 25px;
            height: 40px;
            background: #ff4500;
            border-radius: 10px;
            transform: rotate(-20deg);
        }

        .horse-head {
            position: absolute;
            top: -40px;
            left: 0px;
            width: 35px;
            height: 25px;
            background: #ff8c00;
            border-radius: 15px 15px 5px 5px;
        }

        /* Animation for up and down movement */
        .up-down {
            animation: bounce 2s ease-in-out infinite alternate;
        }

        @keyframes bounce {
            from { transform: translateY(0px); }
            to { transform: translateY(-30px); }
        }

        /* Decorative lights */
        .light {
            position: absolute;
            width: 15px;
            height: 15px;
            border-radius: 50%;
            background: #fff;
            box-shadow: 0 0 20px #fff;
            animation: flicker 0.5s infinite alternate;
        }

        @keyframes flicker {
            from { opacity: 1; }
            to { opacity: 0.3; }
        }

        /* Sparkles in background */
        .sparkle {
            position: absolute;
            width: 5px;
            height: 5px;
            background: white;
            border-radius: 50%;
            animation: drift 10s linear infinite;
        }

        @keyframes drift {
            from { transform: translateY(100vh) translateX(0); opacity: 1; }
            to { transform: translateY(-10vh) translateX(100px); opacity: 0; }
        }
    </style>
</head>
<body>

    <div id="background-effects"></div>

    <div class="carousel-platform" id="carousel">
        <div class="pole"></div>
        <div class="carousel-base"></div>
        <!-- Horses will be injected here via JS -->
    </div>

    <script>
        const carousel = document.getElementById('carousel');
        const bgContainer = document.getElementById('background-effects');
        const colors = ['#ff00ff', '#00ffff', '#ffff00', '#ff4500', '#00ff00', '#ff0000'];
        const horseCount = 8;

        // Function to create a horse
        function createHorse(index) {
            const container = document.createElement('div');
            container.className = 'horse-container';
            
            // Arrange horses in a circle
            const angle = (index / horseCount) * 360;
            const radius = 220;
            
            // CSS transform to position horse around the center
            // We rotate the container, then translate out, then add a bounce animation
            container.style.transform = `rotateY(${angle}deg) translateX(${radius}px) rotateY(-${angle}deg)`;

            const horseWrapper = document.createElement('div');
            horseWrapper.className = 'horse up-down';
            // Stagger the bounce
            horseWrapper.style.animationDelay = `${index * 0.2}s`;

            const color = colors[index % colors.length];

            const body = document.createElement('div');
            body.className = 'horse-body';
            body.style.backgroundColor = color;
            body.style.boxShadow = `0 0 20px ${color}`;

            const neck = document.createElement('div');
            neck.className = 'horse-neck';
            neck.style.backgroundColor = color;

            const head = document.createElement('div');
            head.className = 'horse-head';
            head.style.backgroundColor = color;

            // Add a saddle
            const saddle = document.createElement('div');
            saddle.style.position = 'absolute';
            saddle.style.top = '5px';
            saddle.style.left = '10px';
            saddle.style.width = '40px';
            saddle.style.height = '15px';
            saddle.style.backgroundColor = '#4b0082';
            saddle.style.borderRadius = '5px';

            horseWrapper.appendChild(neck);
            horseWrapper.appendChild(head);
            horseWrapper.appendChild(body);
            body.appendChild(saddle);
            container.appendChild(horseWrapper);
            
            return container;
        }

        // Create the horses
        for (let i = 0; i < horseCount; i++) {
            carousel.appendChild(createHorse(i));
        }

        // Add decorative lights around the top of the pole
        for (let i = 0; i $i < 12; i++) {
            const light = document.createElement('div');
            light.className = 'light';
            const angle = (i / 12) * 360;
            const radius = 150;
            const x = Math.cos(angle * Math.PI / 180) * radius;
            const y = Math.sin(angle * Math.PI / 180) * radius;
            
            light.style.left = `calc(50% + ${x}px - 7px)`;
            light.style.top = `calc(50% + ${y}px - 7px)`;
            light.style.backgroundColor = colors[i % colors.length];
            light.style.boxShadow = `0 0 15px ${colors[i % colors.length]}`;
            carousel.appendChild(light);
        }

        // Create background sparkles
        for (let i = 0; i < 50; i++) {
            const sparkle = document.createElement('div');
            sparkle.className = 'sparkle';
            sparkle.style.left = Math.random() * 100 + 'vw';
            sparkle.style.top = Math.random() * 100 + 'vh';
            sparkle.style.animationDuration = (Math.random() * 5 + 5) + 's';
            sparkle.style.opacity = Math.random();
            bgContainer.appendChild(sparkle);
        }

    </script>
</body>
</html>
```
