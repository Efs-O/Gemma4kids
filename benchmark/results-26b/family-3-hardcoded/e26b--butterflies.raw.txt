    const meadow = document.getElementById('meadow');
    const butterflyLayer = document.getElementById('butterfly-layer');

    const colors = ['#FF007F', '#FF5733', '#C70039', '#900C3F', '#FFC300', '#DAF7A6', '#33FF57'];

    // Create Flowers
    function createFlowers() {
        const flowerCount = 15;
        for (let i = 0; i < flowerCount; i++) {
            const flower = document.createElement('div');
            flower.className = 'flower';
            
            const xPos = Math.random() * 90;
            const yPos = Math.random() * 20; // Spread across width
            const petalColor = colors[Math.floor(Math.random() * colors.length)];
            
            flower.style.left = `${xPos}vw`;
            flower.style.bottom = `${yPos + 10}vh`;
            flower.style.animationDelay = `${Math.random() * 2}s`;

            const stem = document.createElement('div');
            stem.className = 'stem';
            
            const center = document.createElement('div');
            center.className = 'flower-center';

            flower.appendChild(stem);
            flower.appendChild(center);

            // Create petals
            for (let j = 0; j < 5; j++) {
                const petal = document.createElement('div');
                petal.className = 'petal';
                petal.style.backgroundColor = petalColor;
                flower.appendChild(petal);
            }

            meadow.appendChild(flower);
        }
    }

    // Create Butterflies
    function createButterflies() {
        const butterflyCount = 8;
        for (let i = 0; i < butterflyCount; i++) {
            const container = document.createElement('div');
            container.className = 'butterfly-container';
            
            // Randomize flight path
            const duration = 10 + Math.random() * 15;
            const delay = Math.random() * -20;
            const startX = Math.random() * -20;
            
            container.style.animationDuration = `${duration}s`;
            container.style.animationDelay = `${delay}s`;
            
            const butterfly = document.createElement('div');
            butterfly.className = 'butterfly';

            const wingLeft = document.createElement('div');
            wingLeft.className = 'wing wing-left';
            const wingRight = document.createElement('div');
            wing                className = 'wing wing-right';
            const body = document.createElement('div');
            body.className = 'body';

            const wingColor = colors[Math.floor(Math.random() * colors.length)];
            wingLeft.style.backgroundColor = wingColor;
            wingRight.style.backgroundColor = wingColor;

            butterfly.appendChild(wingLeft);
            butterfly.appendChild(wingRight);
            butterfly.appendChild(body);
            container.appendChild(butterfly);
            butterflyLayer.appendChild(container);
        }
    }

    // Initialize
    createFlowers();
    createButterflies();

    // Add some extra "pollen" or sparkles
    function createSparkles() {
        for(let i=0; i<30; i++) {
            const sparkle = document.createElement('div');
            sparkle.style.position = 'absolute';
            sparkle.style.width = '4px';

            sparkle.style.opacity = Math.random();
            sparkle.style.zIndex = '5';
            sparkle.style.pointerEvents = 'none';
            document.body.appendChild(sparkle);
            
            // Random twinkling
            setInterval(() => {
                sparkle.style.opacity = Math.random();
            }, 500 + Math.random() * 1000);
        }
    }
