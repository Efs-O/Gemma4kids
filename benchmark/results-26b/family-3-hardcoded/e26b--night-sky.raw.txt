```html
<!DOCTYPE html>
<html lang="en">
<head>
    <meta charset="UTF-8">
    <meta name="viewport" content="width=device-width, initial-scale=1.0">
    <title>Starfield Animation</title>
    <style>
        body {
            margin: 0;
            padding: 0;
            overflow: hidden;
            background: radial-gradient(circle at bottom, #1a0033 0%, #000000 100%);
            width: 100vw;
            height: 100vh;
        }

        canvas {
            display: block;
        }

        .shooting-star {
            position: absolute;
            top: 0;
            left: 0;
            width: 100%;
            height: 100%;
            pointer-events: none;
        }

        /* Styling for a decorative nebula effect in the background */
        .nebula {
            position: absolute;
            width: 100%;
            height: 100%;
            background: 
                radial-gradient(circle at 20% 30%, rgba(255, 0, 255, 0.1) 0%, transparent 40%),
                radial-gradient(circle at 80% 70%, rgba(0, 255, 255, 0.1) 0%, transparent 40%);
            z-index: -1;
        }
    </style>
</head>
<body>
    <div class="nebula"></div>
    <div class="shooting-star" id="starContainer"></div>
    <canvas id="starCanvas"></canvas>

    <script>
        const canvas = document.getElementById('starCanvas');
        const ctx = canvas.getContext('2d');
        const starContainer = document.getElementById('starContainer');

        let width, height;
        const stars = [];
        const starCount = 150;

        // Resize canvas to fill viewport
        function resize() {
            width = window.innerWidth;
            height = window.innerHeight;
            canvas.width = width;
            canvas.height = height;
            initStars();
        }

        // Star object class
        class Star {
            constructor() {
                this.reset();
            }

            reset() {
                this.x = Math.random() * width;
                this.y = Math.random() * height;
                this.size = Math.random() * 2 + 0.5;
                this.opacity = Math.random();
                this.fadeSpeed = Math.random() * 0.02 + 0.005;
                this.growing = true;
                // Vibrant colors for stars: white, light blue, soft yellow, light pink
                const colors = ['#ffffff', '#e0ffff', '#fff0f5', '#fffacd'];
                this.color = colors[Math.floor(Math.random() * colors.length)];
            }

            update() {
                if (this.growing) {
                    this.opacity += this.fadeSpeed;
                    if (this.opacity >= 1) this.growing = false;
                } else {
                    this.opacity -= this.fadeSpeed;
                    if (this.opacity <= 0.1) this.growing = true;
                }
            }

            draw() {
                ctx.beginPath();
                ctx.arc(this.x, this.y, this.size, 0, Math.PI * 2);
                ctx.fillStyle = this.color;
                ctx.globalAlpha = this.opacity;
                ctx.fill();
            }
        }

        function initStars() {
            stars.length = 0;
            for (let i = 0; i < starCount; i++) {
                stars.push(new Star());
            }
        }

        // Create shooting star elements
        function createShootingStar() {
            const star = document.createElement('div');
            star.style.position = 'absolute';
            star.style.width = '2px';
            star.style.height = '2px';
            star.style.backgroundColor = '#ffffff';
            star.style.boxShadow = '0 0 10px 2px #00ffff, 0 0 20px 5px #ff00ff';
            star.style.borderRadius = '50%';
            star.style.opacity = '1';
            
            // Random start position
            const startX = Math.random() * width;
            const startY = Math.random() * (height / 2);
            star.style.left = startX + 'px';
            star.style.top = startY + 'px';

            starContainer.appendChild(star);

            // Animate via JS
            const duration = 1000 + Math.random() * 1000;
            const angle = Math.PI / 4 + (Math.random() * 0.5); // Diagonal downward
            const distance = Math.max(width, height);

            const animation = star.animate([
                { transform: 'translate(0, 0) rotate(0deg)', opacity: 1 },
                { transform: `translate(${Math.cos(angle) * distance}px, ${Math.sin(angle) * distance}px) rotate(${angle}rad)`, opacity: 0 }
            ], {
                duration: duration,
                easing: 'linear'
            });

            animation.onfinish = () => {
                star.remove();
            };
        }

        function animate() {
            // Clear canvas with a slight trail effect (though we use solid gradient)
            ctx.clearRect(0, 0, width, height);
            
            // Re-apply background gradient via canvas for better composition if needed, 
            // but here we just rely on CSS body background.
            
            stars.forEach(star => {
                star.update();
                star.draw();
            });

            requestAnimationFrame(animate);
        }

        // Event listeners
        window.addEventListener('resize', resize);

        // Initialization
        resize();
        animate();

        // Randomly trigger shooting stars
        setInterval(() => {
            if (Math.random() > 0.7) {
                createShootingStar();
            }
        }, 800);

    </script>
</body>
</html>
```
