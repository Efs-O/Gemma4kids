```html
<!DOCTYPE html>
<html lang="en">
<head>
    <meta charset="UTF-8">
    <meta name="viewport" content="width=device-width, initial-scale=1.0">
    <title>Neon Bounce</title>
    <style>
        body {
            margin: 0;
            padding: 0;
            overflow: hidden;
            background: radial-gradient(circle, #2b0044 0%, #000000 100%);
            display: flex;
            justify-content: center;
            align-items: center;
        }

        canvas {
            display: block;
        }

        #ui {
            position: absolute;
            top: 20px;
            left: 20px;
            color: #00ffcc;
            font-family: 'Segoe UI', Tahoma, Geneva, Verdana, sans-serif;
            font-weight: bold;
            font-size: 24px;
            text-shadow: 0 0 10px #00ffcc;
            pointer-events: none;
            text-transform: uppercase;
            letter-spacing: 2px;
        }
    </style>
</head>
<body>
    <div id="ui">Hyper Bounce</div>
    <canvas id="bounceCanvas"></canvas>

    <script>
        const canvas = document.getElementById('bounceCanvas');
        const ctx = canvas.getContext('2d');

        let width, height;
        let ball;

        class Ball {
            constructor(x, y, radius, color) {
                this.x = x;
                this.y = y;
                this.radius = radius;
                this.color = color;
                this.vy = 2; // Initial vertical velocity
                this.vx = 5; // Initial horizontal velocity
                this.gravity = 0.8;
                this.friction = 0.92; // Damping factor
                this.bounceStrength = -0.85; 
            }

            draw() {
                // Add a glowing effect
                ctx.shadowBlur = 30;
                ctx.shadowColor = this.color;
                
                ctx.beginPath();
                ctx.arc(this.x, this.y, this.radius, 0, Math.PI * 2);
                ctx.fillStyle = this.color;
                ctx.fill();
                ctx.closePath();

                // Reset shadow for other elements
                ctx.shadowBlur = 0;

                // Add a small highlight for 3D effect
                ctx.beginPath();
                ctx.arc(this.x - this.radius * 0.3, this.y - this.radius * 0.3, this.radius * 0.2, 0, Math.PI * 2);
                ctx.fillStyle = 'rgba(255, 255, 255, 0.6)';
                ctx.fill();
                ctx.closePath();
            }

            update() {
                // Apply gravity
                this.vy += this.gravity;
                
                // Update position
                this.y += this.vy;
                this.x += this.vx;

                // Floor collision
                if (this.y + this.radius > height) {
                    this.y = height - this.radius;
                    this.vy *= this.bounceStrength;
                    
                    // Add horizontal friction when hitting floor
                    this.vx *= this.friction;
                }

                // Ceiling collision
                if (this.y - this.radius < 0) {
                    this.y = this.radius;
                    this.vy *= -1;
                }

                // Wall collisions
                if (this.x + this.radius > width) {
                    this.x = width - this.radius;
                    this.vx *= -1;
                }

                if (this.x - this.radius < 0) {
                    this.x = this.radius;
                    this.vx *= -1;
                }

                // Prevent the ball from losing all momentum and getting stuck
                if (Math.abs(this.vy) < 1 && this.y > height - this.radius - 1) {
                    this.vy = -10; // Re-launch if it settles
                }
            }
        }

        function init() {
            width = window.innerWidth;
            height = window.innerHeight;
            canvas.width = width;
            canvas.height = height;

            // Create a large vibrant neon ball (100px diameter)
            ball = new Ball(width / 2, height / 2, 50, '#00ffcc');
        }

        function animate() {
            // Semi-transparent clear for trail effect
            ctx.fillStyle = 'rgba(0, 0, 0, 0.2)';
            ctx.fillRect(0, 0, width, height);

            // Draw floor line
            ctx.beginPath();
            ctx.moveTo(0, height - 2);
            ctx.lineTo(width, height - 2);
            ctx.strokeStyle = '#ff00ff';
            ctx.lineWidth = 4;
            ctx.stroke();
            ctx.closePath();

            ball.update();
            ball.draw();

            requestAnimationFrame(animate);
        }

        window.addEventListener('resize', () => {
            width = window.innerWidth;
            height = window.innerHeight;
            canvas.width = width;
            canvas.height = height;
            // Keep ball in bounds on resize
            if (ball.y + ball.radius > height) ball.y = height - ball.radius;
            if (ball.x + ball.radius > width) ball.x = width - ball.radius;
        });

        // Initial start
        init();
        animate();
    </script>
</body>
</html>
```
