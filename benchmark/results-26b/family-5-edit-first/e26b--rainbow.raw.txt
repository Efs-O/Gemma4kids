```html
<!DOCTYPE html>
<html lang="en">
<head>
    <meta charset="UTF-8">
    <meta name="viewport" content="width=device-width, initial-scale=1.0">
    <title>Rainbow After Rain Animation</title>
    <style>
        body, html {
            margin: 0;
            padding: 0;
            width: 100%;
            height: 100%;
            overflow: hidden;
            background-color: #2c3e50; /* Start with a dark stormy color */
        }

        canvas {
            display: block;
        }

        #ui-overlay {
            position: absolute;
            top: 20px;
            left: 50%;
            transform: translateX(-50%);
            font-family: 'Segoe UI', Tahoma, Geneva, Verdana, sans-serif;
            color: white;
            font-size: 1.5rem;
            text-shadow: 2px 2px 4px rgba(0,0,0,0.5);
            pointer-events: none;
            transition: opacity 2s;
            text-align: center;
        }
    </style>
</head>
<body>

    <div id="ui-overlay">A storm is passing...</div>
    <canvas id="animationCanvas"></canvas>

    <script>
        const canvas = document.getElementById('animationCanvas');
        const ctx = canvas.getContext('2d');
        const uiText = document.getElementById('ui-overlay');

        let width, height;
        let raindrops = [];
        let clouds = [];
        let frameCount = 0;
        
        // Animation States: 0 = Storm, 1 = Transition, 2 = Clear
        let currentState = 0; 
        let skyColor = { r: 44, g: 62, b: 80 }; // Dark Slate
        let targetSkyColor = { r: 44, g: 62, b: 80 };
        let rainbowOpacity = 0;
        let sunScale = 0;

        const RAIN_COUNT = 150;
        const CLOUD_COUNT = 6;

        function resize() {
            width = canvas.width = window.innerWidth;
            height = canvas.height = window.innerHeight;
        }

        window.addEventListener('resize', resize);
        resize();

        class RainDrop {
            constructor() {
                this.reset();
            }

            reset() {
                this.x = Math.random() * width;
                this.y = Math.random() * -height;
                this.speed = 10 + Math.random() * 15;
                this.len = 15 + Math.random() * 10;
                this.opacity = 0.4 + Math.random() * 0.4;
            }

            update() {
                this.y += this.speed;
                if (this.y > height) {
                    this.reset();
                }
            }

            draw() {
                ctx.beginPath();
                ctx.strokeStyle = `rgba(174, 194, 224, ${this.opacity})`;
                ctx.lineWidth = 1;
                ctx.moveTo(this.x, this.y);
                ctx.lineTo(this.x, this.y + this.len);
                ctx.stroke();
            }
        }

        class Cloud {
            constructor() {
                this.reset();
                this.x = Math.random() * width;
            }

            reset() {
                this.x = -300;
                this.y = Math.random() * (height * 0.4);
                this.size = 100 + Math.random() * 150;
                this.speed = 0.2 + Math.random() * 0.5;
                this.opacity = 0.3 + Math.random() * 0.5;
            }

            update() {
                this.x += this.speed;
                if (this.x > width + 300) {
                    this.reset();
                }
            }

            draw() {
                ctx.beginPath();
                const gradient = ctx.createRadialGradient(this.x, this.y, 0, this.x, this.y, this.size);
                const color = currentState === 0 ? `rgba(100, 100, 100, ${this.opacity})` : `rgba(255, 255, 255, ${this.opacity})`;
                gradient.addColorStop(0, color);
                gradient.addColorStop(1, 'transparent');
                ctx.fillStyle = gradient;
                ctx.arc(this.x, this.y, this.size, 0, Math.PI * 2);
                ctx.fill();
            }
        }

        // Initialize
        for (let i = 0; i < RAIN_COUNT; i++) raindrops.push(new RainDrop());
        for (let i = 0; i < CLOUD_COUNT; i++) clouds.push(new Cloud());

        const rainbowColors = [
            'rgba(255, 0, 0,',    // Red
            'rgba(255, 127, 0,',  // Orange
            'rgba(255, 255, 0,',  // Yellow
            'rgba(0, 255, 0,',    // Green
            'rgba(0, 0, 255,',    // Blue
            'rgba(75, 0, 130,',   // Indigo
            'rgba(148, 0, 211,'   // Violet
        ];

        function updateLogic() {
            frameCount++;

            // Timeline Management
            if (frameCount === 300) {
                currentState = 1; // Start Transition
                targetSkyColor = { r: 135, g: 206, b: 235 }; // Sky Blue
                uiText.innerText = "The clouds are parting...";
                uiText.style.opacity = 1;
            }

            if (frameCount === 600) {
                currentState = 2; // Clear
                uiText.innerText = "A beautiful rainbow appears!";
                setTimeout(() => { uiText.style.opacity = 0; }, 3000);
            }

            // Smoothly transition sky color
            const lerp = 0.005;
            skyColor.r += (targetSkyColor.r - skyron) * lerp; // Note: fix variable typo below
            skyColor.r += (targetSkyColor.r - skyColor.r) * lerp;
            skyColor.g += (targetSkyColor.g - skyColor.g) * lerp;
            skyColor.b += (targetSkyColor.b - skyColor.b) * lerp;

            // Transition Rain Opacity
            if (currentState >= 1) {
                raindrops.forEach(r => r.opacity *= 0.99);
            }

            // Rainbow Appearance
            if (currentState >= 1) {
                if (rainbowOpacity < 0.5) rainbowOpacity += 0.002;
            }

            // Sun Appearance
            if (currentState === 2) {
                if (sunScale < 1) sunScale += 0.005;
            }

            // Update Elements
            raindrops.forEach(r => r.update());
            clouds.forEach(c => c.update());
        }

        function drawRainbow() {
            if (rainbowOpacity <= 0) return;

            const centerX = width / 2;
            const centerY = height * 1.2; // Start arc below screen
            const baseRadius = Math.min(width, height) * 0.4;

            ctx.save();
            ctx.globalAlpha = rainbowOpacity;
            
            rainbowColors.forEach((color, index) => {
                ctx.beginPath();
                ctx.lineWidth = 15;
                ctx.strokeStyle = color + ' 0.4)';
                ctx.arc(centerX, centerY, baseRadius + (index * 15), Math.PI, 0);
                ctx.stroke();
            });
            ctx.restore();
        }

        function drawSun() {
            if (sunScale <= 0) return;

            const sunX = width - 100;
            const sunY = 100;
            
            ctx.save();
            ctx.translate(sunX, sunY);
            ctx.scale(sunScale, sunScale);
            
            const gradient = ctx.createRadialGradient(0, 0, 0, 0, 0, 60);
            gradient.addColorStop(0, 'rgba(255, 255, 0, 1)');
            gradient.addColorStop(0.5, 'rgba(255, 165, 0, 0.8)');
            gradient.addColorStop(1, 'rgba(255, 255, 0, 0)');

            ctx.beginPath();
            ctx.fillStyle = gradient;
            ctx.arc(0, 0, 60, 0, Math.PI * 2);
            ctx.fill();
            
            ctx.restore();
        }

        function animate() {
            // 1. Draw Background
            ctx.fillStyle = `rgb(${skyColor.r}, ${skyColor.g}, ${skyColor.b})`;
            ctx.fillRect(0, 0, width, height);

            // 2. Update & Draw Clouds
            clouds.forEach(c => c.draw());

            // 3. Update & Draw Rain
            if (currentState === 0) {
                raindrops.forEach(r => r.draw());
            } else {
                // If transition, draw faded rain
                raindrops.forEach(r => {
                    if (r.opacity > 0.05) r.draw();
                });
            }

            // 4. Draw Rainbow
            drawRainbow();

            // 5. Draw Sun
            drawSun();

            // 6. Logic
            updateLogic();

            requestAnimationFrame(animate);
        }

        animate();
    </script>
</body>
</html>
```
