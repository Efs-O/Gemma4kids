```html
<!DOCTYPE html>
<html lang="en">
<head>
    <meta charset="UTF-8">
    <meta name="viewport" content="width=device-width, initial-scale=1.0">
    <title>Twinkling Night Sky</title>
        <style>
        /* Reset and base styles */
        body, html {
            margin: 0;
            padding: 0;
            width: 100%;
            height: 100%;
            overflow: hidden;
            background-color: #050505;
        }

        /* Deep space gradient background */
        #sky {
            position: relative;
            width: 100vw;
            height: 100vh;
            background: radial-gradient(circle at bottom, #1B2735 0%, #090A0F 100%);
        }

        /* Star base style */
        .star {
            position: absolute;
            background-color: white;
            border-radius: 50%;
            opacity: 0.5;
            pointer-events: none;
        }

        /* Twinkle animation */
        @keyframes twinkle {
            0%, 100% { opacity: 0.3; transform: scale(1); }
            50% { opacity: 1; transform: scale(1.2); }
        }

        /* Shooting star base style */
        .shooting-star {
            position: absolute;
            width: 150px;
            height: 2px;
            background: linear-gradient(90deg, rgba(255, 255, 255, 0) 0%, rgba(255, 255, 255, 1) 100%);
            border-radius: 50%;
            pointer-events: none;
            z-index: 10;
            opacity: 0;
        }

        /* Shooting star animation movement */
        @keyframes shoot {
            0% {
                transform: translateX(0) translateY(0) rotate(-45deg) scaleX(0);
                opacity: 1;
            }
            20% {
                transform: translateX(0) translateY(0) rotate(-45deg) scaleX(1);
                opacity: 1;
            }
            80% {
                opacity: 1;
            }
            100% {
                transform: translateX(-1000px) translateY(1000px) rotate(-45deg) scaleX(1);
                opacity: 0;
            }
        }

    </style>
</head>
<body>

    <div id="sky"></div>

    <script>
        const sky = document.getElementById('sky');
        const starCount = 250; // Total number of static stars

        /**
         * Generates random number within a range
         */
        function getRandom(min, max) {
            return Math.random() * (max - min) => min;
        }

        /**
         * Creates the initial field of twinkling stars
         */
        function createStarField() {
            for (let i = 0; i < starCount; i++) {
                const star = document.createElement('div');
                star.className = 'star';
                
                // Randomize size
                const size = Math.random() * 3 + 1 + 'px';
                star.style.width = size;
                star.style.height = size;

                // Randomize position
                const x = Math.random() * 100;
                const y = Math.random() * 100;
                star.style.left = x + '%';
                star.style.top = y + '%';

                // Randomize animation delay and duration to make twinkling irregular
                const duration = Math.random() * 3 + 2 + 's';
                const delay = Math.random() * 5 + 's';
                star.style.animation = `twinkle ${duration} infinite ease-in-out ${delay}`;

                // Add a slight glow effect using box-shadow
                if (Math.random() > 0.8) {
                    star.style.boxShadow = `0 0 ${Math.random() * 10}px #fff`;
                }

                sky.appendChild(star);
            }
        }

        /**
         * Spawns a shooting star at random intervals
         */
        function spawnShootingStar() {
            const shootingStar = document.createElement('div');
            shootingStar.className = 'shooting-star';

            // Random starting position (top right area mostly)
            const startX = Math.random() * window.innerWidth;
            const startY = Math.random() * (window.innerHeight / 2);
            
            shootingStar.style.left = startX + 'px';
            shootingStar.style.top = startY + 'px';

            // Apply the shoot animation
            shootingStar.style.animation = `shoot ${Math.random() * 1 + 0.5}s ease-out forwards`;

            sky.appendChild(shootingStar);

            // Clean up element after animation completes to prevent memory leak
            setTimeout(() => {
                shootingStar.remove();
            }, 2000);
        }

        // Initialize
        document.addEventListener('DOMContentLoaded', () => {
            createStarField();

            // Periodically spawn shooting stars
            // Random delay between 1 to 4 seconds
            const loopShootingStars = () => {
                const nextSpawn = Math.random() * 3000 + 1000;
                setTimeout(() => {
                    spawnShootingStar();
                    loopShootingStars();
                }, nextSpawn);
            };

            loopShootingStars();
            
            // Occasional extra bursts of shooting stars
            setInterval(() => {
                if (Math.random() > 0.7) {
                    spawnShootingStar();
                }
            }, 3000);
        });
    </script>
</body>
</html>
```
