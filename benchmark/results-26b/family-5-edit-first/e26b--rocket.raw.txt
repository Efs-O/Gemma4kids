    <meta charset="UTF-8">
    <meta name="viewport" content="width=device-width, initial-scale=1.0">
    <title>Rocket Launch Animation</title>
    <style>
        :root {
            --space-bg: #050510;
            --rocket-body: #f0f0f0;
            --rocket-fin: #e63946;
            --rocket-window: #a8dadc;
            --flame-color: #ffb703;
            --flame-secondary: #fb8500;
        }

    /* Reset and full viewport setup */
        body, html {
            margin: 0;
            padding: 0;
            width: 100%;
            height: 100%;
            overflow: hidden;
            background-color: var(--space-bg);
            display: flex;
            justify-content: center;
            align-items: center;
        }

        canvas {
            position: absolute;
            top: 0;
            left: 0;
            z-index: 1;
        }

        /* The Rocket Container handles the vertical "launch" movement */
        #rocket-container {
            position: relative;
            z-index: 10;
            display: flex;
            flex-direction: column;
            align-items: center;
            bottom: -20%; /* Start partially off-screen */
            transition: transform 4s cubic-bezier(0.5, 0, 0.5, 1);
            animation: launchMovement 10s forwards;
        }

        /* Shake animation for the ignition phase */
        @keyframes shake {
            0% { transform: translateX(0); }
            25% { transform: translateX(-2px); }
            50% { transform: translateX(2px); }
            75% { transform: translateX(-2px); }
            100% { transform: translateX(0); }
        }

        /* Movement of the rocket moving upwards through space */
        @keyframes launchMovement {
            0% { transform: translateY(0px); }
            20% { transform: translateY(0px); } /* Wait for ignition */
            40% { transform: translateY(-100px); } /* Start lifting */
            100% { transform: translateY(-150vh); } /* Fly off screen */
        }

        .rocket-body-main {
            width: 50px;
            height: 100px;
            background-color: var(--rocket-body);
            border-radius: 50% 50% 10% 10%;
            position: relative;
            box-shadow: inset -10px 0 15px rgba(0,0,0,0.1);
            animation: shake 0.1s infinite;
            animation-delay: 2s; /* Start shaking after 2 seconds */
        }

        /* Rocket Window */
        .window {
            width: 20px;
            height: 20px;
            background-color: var(--rocket-window);
            border: 3px solid #333;
            border-radius: 50%;
            position: absolute;
            top: 25px;
            left: 15px;
        }

        /* Rocket Fins */
        .fin {
            width: 30px;
            height: 50px;
            background-color: var(--rocket-fin);
            position: absolute;
            bottom: 0;
            z-index: -1;
        }

        .fin-left {
            left: -20px;
            border-radius: 100% 0 0 0;
            transform: rotate(-15deg);
        }

        .fin-right {
            right: -20px;
            border-radius: 0 100% 0 0;
            transform: rotate(15deg);
        }

        /* The Engine Flame (Animated via JS particles, but CSS provides the core glow) */
        #flame-glow {
            width: 40px;
            height: 40px;
            background: radial-gradient(circle, var(--flame-color), var(--flame-secondary), transparent);
            position: absolute;
            bottom: -30px;
            border-radius: 50%;
            opacity: 0;
            filter: blur(5px);
            animation: ignite 1s forwards;
            animation-delay: 2s;
        }

        @keyframes ignite {
            0% { opacity: 0; transform: scale(0.5); }
            100% { opacity: 1; transform: scale(1.5); }
        }

        /* UI Overlay */
        #status-text {
            position: absolute;
            top: 20px;
            left: 20px;
            color: white;
            font-family: 'Segoe UI', Tahoma, Geneva나, sans-serif;

<body>

    <div id="status-text">Status: Ready for Launch</div>

    <canvas id="spaceCanvas"></canvas>

    <div id="rocket-container">
        <div id="flame-glow"></div>
        <div class="rocket-body-main">
            <div class="window"></div>
            <div class="fin fin-left"></div>
            <div class="fin fin-right"></div>
        </div>
    </div>

    <script>
        const canvas = document.getElementById('spaceCanvas');
        const ctx = canvas.getContext('2d');
        const statusText = document.getElementById('status-text');

        let width, height;
        let stars = [];
        let particles = [];
        let launchStarted = false;
        let speedMultiplier = 1;

        // Configuration
        const STAR_COUNT = 400;
        const PARTICLE_COUNT = 50;

        class Star {
            constructor() {
                this.init();
            }

            init() {
                this.x = Math.random() * width;
                this.y = Math.random() * height;
                this.size = Math.random() * 2;
                this.speed = Math.random() * 0.5 + 0.1;
                this.brightness = Math.random();
            }

            update() {
                // Stars move downwards to simulate rocket moving upwards
                this.y += this.speed * speedMultiplier;

                // If star goes off bottom, reset to top
                if (this.y > height) {
                    this.y = 0;
                    this.x = Math.random() * width;
                }
            }

            draw() {
                ctx.fillStyle = `rgba(255, 255, 255, ${this.brightness})`;
                ctx.beginPath();
                ctx.arc(this.x, this.y, this.size, 0, Math.PI * 2);
                ctx.fill();
            }
        }

        class Particle {
            constructor() {
                this.reset();
            }

            reset() {
                this.x = width / 2; 
                // We position particles relative to the screen center 
                // but we'll animate them based on the rocket's visual position
                this.y = height * 0.7; 
                this.vx = (Math.random() - 0.5) * 4;
                this.vy = Math.random() * 5 + 2;
                this.life = 1;
                this.decay = Math.random() * 0.02 + 0.01;
                this.color = Math.random() > 0.5 ? '#ffb703' : '#fb850    0';
                this.size = Math.random() * 8 + 4;
            }

            update() {
                this.x += this.vx;
                this.y += this.vy;
                this.life -= this.decay;

                if (this.life <= 0) {
                    this.reset();
                }
            }

            draw() {
                ctx.globalAlpha = this.life;
                ctx.fillStyle = this.color;
        // Use a glow effect for particles
                ctx.shadowBlur = 10;
                ctx.shadowColor = this.color;
                ctx.beginPath();
                ctx.arc(this.x, this.y, this.size, 0, Math.PI * 2);
                ctx.fill();
                ctx.shadowBlur = 0;
                ctx.globalAlpha = 1;
            }
        }

        function resize() {
            width = canvas.width = window.innerWidth;
            height = canvas.height = window.innerHeight;
            initStars();
        }

        function initStars() {
            stars = [];
            for (let i = 0; i < STAR_COUNT; i++) {
                stars.push(new Star());
            }
        }

        function initParticles() {
            particles = [];
            for (let i = 0; i < PARTICLE_COUNT; i++) {
                particles.push(new Particle());
            }
        }

        function animate() {
            // Background Clear
            ctx.fillStyle = '#050510';
            ctx.fillRect(0, 0, width, height);

            // Update Speed based on "Time" (simulating acceleration)
            // We use the timestamp to drive the "speed" of the stars
            const time = Date.now() * 0.001;
            
            // Increase star speed after 2 seconds (ignition)
            if (time > 2) {
                speedMultiplier = Math.min(20, 1 + (time - 2) * 2);
                statusText.innerText = "Status: Ascending...";
                statusText.style.color = "#fb8500";
            }

            // Update and Draw Stars
            stars.forEach(star => {
                star.update();
                star.draw();
            });

            // Update and Draw Particles (Flame)
            // Only show particles after ignition (2s)
            if (time > 2) {
                particles.forEach(p => {
                    p.update();
                    p.draw();
                });
            }

            requestAnimationFrame(animate);
        }

        // Event Listeners
        window.addEventListener('resize', resize);

        // Start App
        resize();
        initParticles();
        animate();

        // Trigger status change delay
        setTimeout(() => {
            statusText.innerText = "Status: Ignition...";
            statusText.style.color = "#e63946";
        }, 1500);

    </script>
</body>
</html>
