```html
<!DOCTYPE html>
<html lang="en">
<head>
    <meta charset="UTF-8">
    <meta name="viewport" content="width=device-width, initial-scale=1.0">
    <title>Colorful Underwater World</title>
    <style>
        /* Ensure the canvas covers the entire viewport and prevent scrolling */
        body, html {
            margin: 0;
            padding: 0;
            width: 100%;
            height: 100%;
            overflow: hidden;
            background-color: #001a33; /* Deep ocean blue */
        }

        canvas {
            display: block;
        }
    </style>
</head>
<body>

    <canvas id="oceanCanvas"></canvas>

    <script>
        /**
         * Underwater Animation
         * Features: 
         * - Procedurally generated colorful fish with swimming motion
         * - Rising bubbles with varying sizes and speeds
         * - Dynamic gradient background
         */

        const canvas = document.getElementById('oceanCanvas');
        const ctx = canvas.getContext('2d');

        let width, height;
        let fishArray = [];
        let bubbleArray = [];

        // Configuration
        const FISH_COUNT = 15;
        const BUBBLE_COUNT = 40;
        const COLORS = [
            '#FF5733', // Vibrant Orange
            '#FFC300', // Bright Yellow
            '#DAF7A6', // Lime Green
            '#FF33FF', // Hot Pink
            '#33FFCE', // Cyan
            '#FF3385', // Magenta
            '#75FF33'  // Bright Green
        ];

        /**
         * Resize function to handle window resizing
         */
        function resize() {
            width = window.innerWidth;
            height = window.innerHeight;
            canvas.width = width;
            canvas.height = height;
            
            // Re-initialize elements so they stay within viewport bounds
            initScene();
        }

        /**
         * Bubble Class
         * Represents a single rising bubble
         */
        class Bubble {
            constructor() {
                this.reset();
                // Start at random heights initially so they don't all spawn at the bottom at once
                this.y = Math.random() * height;
            }

            reset() {
                this.x = Math.random() * width;
                this.y = height + 50;
                this.radius = Math.random() * 10 + 2;
                this.speedY = Math.random() * 1.5 + 0.5;
                this.speedX = Math.sin(Math.random() * Math.PI) * 0.5;
                this.opacity = Math.random() * 0.5 + 0.1;
            }

            update() {
                this.y -= this.speedY;
                this.x += Math.sin(this.y / 50) * 0.5; // Wobble effect

                // If bubble floats off top, reset to bottom
                if (this.y < -50) {
                    this.reset();
                }
            }

            draw() {
                ctx.beginPath();
                ctx.arc(this.x, this.y, this.radius, 0, Math.PI * 2);
                ctx.fillStyle = `rgba(255, 255, 255, ${this.opacity})`;
                ctx.fill();
                // Add a small shine highlight
                ctx.beginPath();
                ctx.arc(this.x - this.radius * 0.3, this.y - this.radius * 0.3, this.radius * 0.2, 0, Math.PI * 2);
                ctx.fillStyle = 'rgba(255, 255, 255, 0.8)';
                ctx.fill();
            }
        }

        /**
         * Fish Class
         * Represents a swimming fish
         */
        class Fish {
            constructor() {
                this.reset();
            }

            reset() {
                this.x = Math.random() * width;
                this.y = Math.random() * height;
                this.size = Math.random() * 20 + 20;
                this.color = COLORS[Math.floor(Math.random() * COLORS.length)];
                this.speedX = (Math.random() * 2 + 1) * (Math.random() > 0.5 ? 1 : -1);
                this.speedY = (Math.random() - 0.5) * 1;
                this.wiggle = 0;
                this.wiggleSpeed = Math.random() * 0.1 + 0.05;
            }

            update() {
                this.x += this.speedX;
                this.y += Math.sin(this.wiggle) * 0.5; // Slight vertical wave
                this.wiggle += this.wiggleSpeed;

                // Wrap around screen
                if (this.speedX > 0 && this.x > width + 50) {
                    this.x = -50;
                    this.y = Math.random() * height;
                } else if (this.speedX < 0 && this.x < -50) {
                    this.x = width + 50;
                    this.y = Math.random() * height;
                }
            }

            draw() {
                ctx.save();
                ctx.translate(this.x, this.y);
                
                // Flip fish if swimming left
                if (this.speedX < 0) {
                    ctx.scale(-1, 1);
                }

                // Draw Tail
                ctx.beginPath();
                ctx.moveTo(-this.size, 0);
                ctx.lineTo(-this.size - (15 * Math.sin(this.wiggle)), -this.size/2);
                ctx.lineTo(-this.size - (15 * Math.sin(this.wiggle)), this.size/2);
                ctx.closePath();
                ctx.fillStyle = this.color;
                ctx.fill();

                // Draw Body (Ellipse)
                ctx.beginPath();
                ctx.ellipse(0, 0, this.size, this.size / 1.8, 0, 0, Math.PI * 2);
                ctx.fillStyle = this.color;
                ctx.fill();

                // Draw Eye
                ctx.beginPath();
                ctx.arc(this.size * 0.5, -this.size * 0.2, this.size * 0.15, 0, Math.PI * 2);
                ctx.fillStyle = 'white';
                ctx.fill();
                ctx.beginPath();
                ctx.arc(this.size * 0.5, -this.size * 0.2, this.size * 0.07, 0, Math.PI * 2);
                ctx.fillStyle='black';
                ctx.fill();

                ctx.restore();
            }
        }

        function initScene() {
            fishArray = [];
            bubbleArray = [];

            // Create Fish
            for (let i = 0; i < FISH_COUNT; i++) {
                fishArray.push(new Fish());
            }

            // Create Bubbles
            for (let i = 0;
         i < BUBBLE_COUNT; i++) {
                bubbleArray.push(new Bubble());
            }
        }

        /**
         * Main Animation Loop
         */
        function animate() {
            // Create a beautiful deep sea gradient
            const gradient = ctx.createLinearGradient(0, 0, 0, height);
            gradient.addColorStop(0, '#005f73'); // Lighter teal top
            gradient.addColorStop(1, '#001219'); // Darker navy bottom
            ctx.fillStyle = gradient;
            ctx.fillRect(0, 0, width, height);

            // Update and draw bubbles
            bubbleArray.forEach(bubble => {
                bubble.update();
                bubble.draw();
            });

            // Update and draw fish
            fishArray.forEach(fish => {
                fish.update();
                fish.draw();
            });

            requestAnimationFrame(animate);
        }

        // Event Listeners
        window.addEventListener('resize', resize);

        // Kickoff
        resize();
        animate();

    </script>
</body>
</html>
```
