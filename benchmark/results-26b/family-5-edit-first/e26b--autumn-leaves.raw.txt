```html
<!DOCTYPE html>
<html lang="en">
<head>
    <meta charset="UTF-8">
    <meta name="viewport" content="width=device-width, initial-scale=1.0">
    <title>Autumn Leaves Animation</title>
    <style>
        /* Reset and full viewport setup */
        body, html {
            margin: 0;
            padding: 0;
            width: 100%;
            height: 100%;
            overflow: hidden;
            background: linear-gradient(to bottom, #ffecd2 0%, #fcb69f 100%);
        }

        canvas {
            display: block;
        }

        /* Overlay text for a nice touch */
        .overlay {
            position: absolute;
            top: 50%;
            left: 50%;
            transform: translate(-50%, -50%);
            font-family: 'Segoe UI', Tahoma, Geneva, Verdana, sans-serif;
            font-size: 3rem;
            color: rgba(139, 69, 19, 0.3);
            pointer-events: none;
            user-select: none;
            text-transform: uppercase;
            letter-spacing: 10px;
        }
    </style>
</head>
<body>

    <div class="overlay">Autumn</div>
    <canvas id="leafCanvas"></canvas>

    <script>
        const canvas = document.getElementById('leafCanvas');
        const ctx = canvas.getContext('2d');

        let width, height;
        let leaves = [];
        const leafCount = 80;
        
        // Vivid Autumn Colors
        const colors = [
            '#FF4500', // OrangeRed
            '#FF8C00', // DarkOrange
            '#FFD700', // Gold
            '#B22222', // FireBrick
            '#E97451', // Burnt Sienna
            '#DAA520', // Goldenrod
            '#FF6347'  // Tomato
        ];

        /**
         * Resizes canvas to fit window
         */
        function resize() {
            width = window.innerWidth;
            height = window.innerHeight;
            canvas.width = width;
            canvas.height = height;
            initLeaves();
            drawTrees();
        }

        /**
         * Represents a single falling leaf
         */
        class Leaf {
            constructor() {
                this.reset(true);
            }

            /**
             * Initialize or reset leaf position and properties
             * @param {boolean} isInitial - If true, start from top, otherwise start from bottom
             */
            reset(isInitial = false) {
                this.x = Math.random() * width;
                // If initial, scatter everywhere; if resetting, start from top
                this.y = isInitial ? Math.random() * height : -20;
                
                this.size = Math.random() * 15 + 10;
                this.color = colors[Math.floor(Math.random() * colors.length)];
                this.speedY = Math.random() * 1 + 0.5;
                this.speedX = Math.random() * 1 - 0.5;
                this.rotation = Math.random() * Math.PI * 2;
                this.rotationSpeed = Math.random() * 0.02 - 0.01;
                
                // Oscillation properties for "swaying" effect
                this.swingAmplitude = Math.random() * 2;
                this.swingFrequency = Math.random() * 0.02;
                this.swingAngle = Math.random() * Math.PI;
            }

            update() {
                this.y += this.speedY;
                this.x += Math.sin(this.swingAngle) * this.swingAmplitude;
                this.swingAngle += this.swingFrequency;
                this.rotation += this.rotationSpeed;

                // If leaf goes off screen bottom, reset to top
                if (this.y > height + 20) {
                    this.reset(false);
                }
            }

            draw() {
                ctx.save();
                ctx.translate(this.x, this.y);
                ctx.rotate(this.rotation);
                ctx.beginPath();
                
                // Draw a simple leaf shape (an ellipse with a point)
                ctx.fillStyle = this.color;
                ctx.moveTo(0, 0);
                ctx.bezierCurveTo(this.size, -this.size, this.size, this.size, 0, this.size * 1.5);
                ctx.bezierCurveTo(-this.size, this.size, -this.size, -this.size, 0, 0);
                
                ctx.fill();
                
                // Add a small vein in the middle
                ctx.strokeStyle = 'rgba(0,0,0,0.1)';
                ctx.lineWidth = 1;
                ctx.beginPath();
                ctx.moveTo(0, 0);
                ctx.lineTo(0, this.size * 1.5);
                ctx.stroke();

                ctx.restore();
            }
        }

        /**
         * Draws stylized tree trunks at the bottom corners
         */
        function drawTrees() {
            ctx.fillStyle = '#5D4037'; // Dark brown
            
            // Left tree trunk
            ctx.beginPath();
            ctx.moveTo(0, height);
            ctx.quadraticCurveTo(width * 0.1, height * 0.7, width * 0.2, height);
            ctx.fill();

            // Right tree trunk
            ctx.beginPath();
            ctx.moveTo(width, height);
            ctx.quadraticCurveTo(width * 0.9, height * 0.7, width * 0.8, height);
            ctx.fill();
        }

        function initLeaves() {
            leaves = [];
            for (let i = 0; i < leafCount; i++) {
                leaves.push(new Leaf());
            }
        }

        function animate() {
            // Clear canvas with a slight fade effect (creating motion trails)
            ctx.clearRect(0, 0, width, height);

            // Re-draw trees every frame in case of resize (logic simplified)
            // In a high-perf app, you'd draw trees to a static background layer
            drawTrees();

            leaves.forEach(leaf => {
                leaf.update();
                leaf.draw();
            });

            requestAnimationFrame(animate);
        }

        // Event Listeners
        window.addEventListener('resize', resize);

        // Initialize
        resize();
        animate();

    </script>
</body>
</html>
```
