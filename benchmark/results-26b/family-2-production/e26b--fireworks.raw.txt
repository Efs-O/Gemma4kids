        const canvas = document.getElementById('fireworksCanvas');
        const ctx = canvas.getContext('2d');

        let width, height;
        function resize() {
            width = window.innerWidth;
            height = window.innerHeight;
            canvas.width = width;
            canvas.height = height;
        }

        window.addEventListener('resize', resize);
        resize();

        class Particle {
            constructor(x, y, color, velocityX, velocityY) {
                this.x = x;
                this.y = y;
                this.color = color;
                this.velocityX = velocityX;
                this.velocityY = velocityY;
                this.alpha = 1;
                this.friction = 0.96;
                this.gravity = 0.15;
            }

            draw() {
                ctx.save();
                ctx.globalAlpha = this.alpha;
                ctx.beginPath();
                ctx.arc(this.x, this.y, 2, 0, Math.PI * 2);
                ctx.fillStyle = this.color;
                ctx.fill();
                ctx.restore();
            }

            update() {
                this.velocityX *= this.friction;
                this.velocityY *= this.friction;
                this.velocityY += this.gravity;
                this.x += this.velocityX;
                this.y += this.velocityY;
                this.alpha -= 0.012;
            }
        }

        class Firework {
            constructor(targetX, targetY) {
                this.x = Math.random() * width;
                this.y = height;
                this.targetX = targetX;
                this.targetY = targetY;
                this.distanceToTarget = Math.hypot(this.targetX - this.x, this.targetY - this.y);
                this.distanceTraveled = 0;
                
                const angle = Math.atan2(this.targetY - this.y, this.targetX - this.x);
                this.velocity = {
                    x: Math.cos(angle) * 12,
                    y: Math.sin(angle) * 12
                };
                
                this.particles = [];
                this.exploded = false;
                this.hue = Math.floor(Math.random() * 360);
                this.color = `hsl(${this.hue}, 100%, 60%)`;
            }

            explode() {
                const particleCount = 60;
                for (let i = 0; i < particleCount; i++) {
                    const angle = (Math.PI * 2 / particleCount) * i;
                    const speed = Math.random() * 6 + 2;
                    this.particles.push(new Particle(
                        this.targetX, 
                        this.targetY, 
                        `hsl(${this.hue}, 100%, 60%)`,
                        Math.cos(angle) * speed,
                        Math.sin(angle) * speed
                    ));
                }
            }

            update() {
                if (!this.exploded) {
                    this.x += this.velocity.x;
                    this.y += this.velocity.y;
                    
                    this.distanceTraveled = Math.hypot(this.x - (this.targetX - (this.velocity.x * 100)), this.y - (this.targetY - (this.velocity.y * 100)));
                    
                    // Check if it reached target area
                    if (Math.abs(this.x - this.targetX) < 10 && Math.abs(this.y - this.targetY) < 10) {
                        this.exploded = true;
                        this.explode();
                    }
                }

                this.particles.forEach((particle, index) => {
                    particle.update();
                    if (particle.alpha <= 0) {
                        this.particles.splice(index, 1);
                    }
                });
            }

            draw() {
                if (!this.exploded) {
                    ctx.beginPath();
                    ctx.arc(this.x, this.y, 3, 0, Math.PI * 2);
                    ctx.fillStyle = 'white';
                    ctx.fill();
                }

                this.particles.forEach(particle => particle.draw());
            }
        }

        let fireworks = [];

        function animate() {
            ctx.fillStyle = 'rgba(15, 12, 41, 0.2)';
            ctx.fillRect(0, 0, width, height);

            fireworks.forEach((firework, index) => {
                firework.update();
                firework.draw();
                if (firework.exploded && firework.particles.length === 0) {
                    fireworks.splice(index, 1);
                }
            });

            requestAnimationFrame(animate);
        }

        // Launch a firework every now and then automatically
        setInterval(() => {
            if(fireworks.length < 5) {
                fireworks.push(new Firework(Math.random() * width, Math.random() * (height/2)));
            }
        }, 1500);

        window.addEventListener('mousedown', (e) => {
            fireworks.push(new Firework(e.clientX, e.clientY));
        });
        
        window.addEventListener('touchstart', (e) => {
            fireworks.push(new Firework(e.touches[0].clientX, e.touches[0].clientY));
        });

        animate();
    </script>
</body>
</html>
